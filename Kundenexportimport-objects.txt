function TExpImpTableObj.AddChildTable(AChildTable: String): TExpImpTableObj;
begin
  if FChildTables.IndexOf(AChildTable) < 0 then
    FChildTables.Add(AChildTable)
  else
   raise Exception.Create('Interner Fehler: ' + AChildTable + ' wurde bereits hinzugefügt!!');
  result := self;
end;

function TExpImpTableObj.SelectIdForExport(AQry: TSQLQuery; ASelectedBy, AFieldName: String): Boolean;
begin
  result := false;
  if not FCanDoExport then
    exit;

  if AFieldName = '' then
    AFieldName := FIdFeldName;

  result := SelectIdForExport(AQry.fieldbyName(AFieldName).AsLongId, ASelectedBy)
end;

function TExpImpTableObj.SelectIdForExport(AId: TLongId; ASelectedBy: String): Boolean;
begin
  result := false;
  if not FCanDoExport then
    exit;

  if AId.IsNull then
    exit;

  if FIdsSelektiert.Contains(AId) then
    exit;

  FIdsSelektiert.Add(AId);
  if FOwner.ExportOptionen.IsExportXmlDebug then
  begin
    FIdsSelektiertDebug.AddObject(AId, ASelectedBy);
    TfrmLib32Debug.AddText(TableName + ': ' + aId.ToStr + ' hinzugefügt wg. ' + ASelectedBy);
  end;
  result := true;
end;

function TExpImpTableObj.CanDoImport(ATable: TTAbleObj; AKundenImport: IKundenImport): Boolean;

function TExpImpTableObj.CheckObSchonExportiert(ATableName: String; Qry: TSQLQuery;
  AMndKndIdListe: TelSimpleLongIdList): Boolean;
  var ColumnsRef: TStrings;
    i: Integer;
begin
  result := false;
  if not CanDoExport then
    exit;

  if ATableName = TableName then
    result := SelectIdForExport(qry.FieldByName(DefaultFieldName).AsLongId, GetSqlDebugInfo(qry))
  else
  begin
    result := false;
    ColumnsRef := TStringList.Create;
    try
      if GetRefColumns(ATableName, ColumnsRef) then
      begin
         for i := 0 to ColumnsRef.Count - 1 do
           if SelectIdForExport(qry.FieldByName(ColumnsRef[i]).AsLongId, GetSqlDebugInfo(qry)) then
             result := true;
       end;
    finally
      ColumnsRef.Free;
    end;
  end;
end;

procedure TExpImpTableObj.Clear;
begin
  FIdsSelektiert.Clear;
  FIdsPrepared.Clear;
  FIdsChildSelected.Clear;
  if FOwnsTableIdMapping then
    FTableIdMapping.Clear;
end;

constructor TExpImpTableObj.Create(AOwner: TExpImpObjectManager; ATable: String; ATableIdMapping: TelStringObjectMap<TTableIdMapping>; AIdentifier: String = '');
var idx: Integer;
     seq: String;

begin
  if AIdentifier = '' then
    AIdentifier := ATable;
  FParents := TelObjectList<TExpImpTableObj>.create(false);
  FOwner := AOwner;
  FAutoPrepareParentIds := false;
  FAutoSelectChildIds := true;

  FTableName := ATable;
  FCanDoImport := true;
  FCanDoExport := true;

  seq := '';
  FDTable := TDBModelUtils.GetTables.Find(TableName);
  if FDTable = nil then
    raise EInvalidOperation.Create('Tabelle ' + ATable + ' nicht in DB-Modell gefunden');

  if FDTable.PKFieldsCount <> 1 then
    raise EInvalidOperation.Create('Tabelle ' + ATable + ' hat mehr als einen PK');

  FIdentifier := AIdentifier;
  for idx := 0 to FDTable.Columns.Count - 1 do
    if FDTable.Columns[idx].PrimaryKey then
    begin
      FIdFeldName := FDTable.Columns[idx].Name;
      break;
    end;

  seq :=  DBModel.FindSequence(FDTable.Schema, FDTable.Name);
  if (ATableIdMapping <> nil) then
    FTableIdMapping := ATableIdMapping.Values[ATable]
  else
    FTableIdMapping := nil;

  FOwnsTableIdMapping := ATableIdMapping = nil;
  if FTableIdMapping = nil then
  begin
    FTableIdMapping := TTableIdMapping.Create(ATable, FIdFeldName, Seq);
    if ATableIdMapping <> nil  then
      ATableIdMapping.AddObject(ATable, FTableIdMapping);
  end;

  FIdsSelektiert := TelSimpleLongIdList.Create;
  FIdsChildSelected := TelSimpleLongIdList.Create;
  FChildTables := TStringList.Create;;
  FIdsPrepared := TelSimpleLongIdList.Create;
  FImportValues := TelObjectList<TTAbleObj>.Create(false);
  FIdsSelektiertDebug := TelLongIdStringMap.Create;

  FImportNachlauf := TelObjectList<TTAbleObj>.Create(false);
end;

destructor TExpImpTableObj.Destroy;
begin
  FIdsSelektiertDebug.Free;
  FIdsSelektiert.Free;
  FChildTables.Free;
  FIdsPrepared.Free;
  FIdsChildSelected.Free;
  if FOwnsTableIdMapping then
    FTableIdMapping.Free;
  FImportValues.Free;
  FImportNachlauf.Free;
  FParents.Free;
  inherited;
end;

procedure TExpImpTableObj.DoImportNacharbeiten(AConnection: TecConnection; AKundenImport: IKundenImport);
var
  i,j: Integer;
  dbColumn: TDBModelColumn;
  tableDts: TTableObj;
  tabCol: TColumnObj;
  id, orgValue: int64;
  sql, pkValue: String;
begin
  for i := 0 to FImportNachlauf.Count - 1 do
  begin
    tableDts := FImportNachlauf[i];
    sql := '';
    for j := 0 to tableDts.Columns.Count - 1 do
    begin
      tabCol := tableDts.Columns[j];
      dbColumn := FDTable.Columns.Find(tabCol.Columnname);

      if dbColumn.PrimaryKey then
      begin
        orgValue := StrToInt64Def(tabCol.Value, NULL_INTEGER);
        if (TableIdMapping.IdMapping.TryGetValue(orgValue, id)) then
        begin
          pkValue := IntToStr(id);
          continue;
        end;

        if FOwner.FImportOptionen.FehlerIgnorieren then
          AKundenImport.AddLog('DoImportNacharbeiten: PkSpalte "' + tabCol.Columnname + '" mit Wert "' + tabCol.Value + '" ist nicht auflösbar --> Kein Update auf FKs')
        else
          raise Exception.Create('DoImportNacharbeiten: PkSpalte "' + tabCol.Columnname + '" mit Wert "' + tabCol.Value + '" ist nicht auflösbar --> Kein Update auf FKs');
        sql := '';
        break;
      end;

      if (tabCol.Value = '') then
        continue;

      if not TableIdMapping.IsRelevant(tableDts.Tablename, tabCol.Columnname) then
        continue;

      if not IsSelbstreferenzierendeSpalte(dbColumn) then
        continue;

      orgValue := StrToInt64Def(tabCol.Value, NULL_INTEGER);
      if (TableIdMapping.IdMapping.TryGetValue(orgValue, id)) then
      begin
        if (sql <> '') then
          sql := sql + ', ';
        sql := sql + dbColumn.Name + ' = ' + IntToStr(id)  + CR_LF;
        continue;
      end;

      if FOwner.FImportOptionen.FehlerIgnorieren then
        AKundenImport.AddLog('DoImportNacharbeiten: Spalte "' + tabCol.Columnname + '" mit Wert "' + tabCol.Value + '" ist nicht auflösbar --> NULL bleibt')
      else
        raise Exception.Create('DoImportNacharbeiten: Spalte "' + tabCol.Columnname + '" mit Wert "' + tabCol.Value + '" ist nicht auflösbar');
    end;

    if (sql <> '') then
    begin
      sql := 'UPDATE CREATOR.' + FDTable.Name + CR_LF +
             ' SET ' + sql +
             'WHERE ' + FIdFeldName + ' = ' + pkValue;
      AConnection.ExecuteDirect(sql);
    end;
  end;
end;

function TExpImpTableObj.DoInitialSelect(AConnection: TecConnection; ASql, AIdParamName: String;
  AIdListe: TelSimpleLongIdList; ADoMore: TVerarbeiteQueryProc): Integer;
var qry: TSqlQuery;
    i: integer;
begin
  result := 0;
  qry := AConnection.CreateSQLQuery;
  try
    qry.SQL.Text := ASql;
    qry.DBPrepare;
    if not AnsiContainsText(Asql, FIdFeldName) then
    begin
      if DebugHook > 0 then
        raise Exception.Create('DoInitialSelect, ' + TableName + ': Das Feld ' + FIdFeldName + ' ist nicht im Sql enthalten: ' +  CR_LF + ASql);
      exit;
    end;

    for i := 0 to AIdListe.Count - 1 do
    begin
      qry.ParamByName(AIdParamName).AsLongId := AIdListe[i];
      try
        qry.Open;
        while not qry.Eof do
        begin
          if SelectIdForExport(qry.FieldByName(FIdFeldName).AsLongId, ASql + ' [' + AIdParamName + '=' + AIdListe[i].ToStr + ']') then
            inc(result);
          if assigned(ADoMore) then
            ADoMore(qry);
          qry.Next;
        end;
      finally
        qry.Close;
      end;
    end;
  finally
    qry.Free;
  end;
end;

function TExpImpTableObj.DoInitialSelect(AConnection: TecConnection; ASql, AIdParamName: String;
  AIdListe: TelSimpleIntegerList; ADoMore: TVerarbeiteQueryProc): Integer;
var i: integer;
    lIdListe:  TelSimpleLongIdList;
begin
  lIdListe := TelSimpleLongIdList.Create(AIdListe.Count);
  try
    for i := 0 to AIdListe.Count - 1 do
      lIdListe.Add(TLongId.From(AIdListe[i]));
    result := DoInitialSelect(AConnection, ASql, AIdParamName, lIdListe, ADOmORE);
  finally
    lIdListe.Free;
  end;
end;

function TExpImpTableObj.GetNewColumnValue(ATable: TTAbleObj; AColumn:TColumnObj; AKundenImport: IKundenImport;
    Var ANewValue: String; AForInsert: Boolean):Boolean;
var id, orgValue: int64;
    dbColumn: TDBModelColumn;

begin
  result := False;
  ANewValue := AColumn.Value;
  if AColumn.Value = '' then
    exit;

  orgValue := StrToInt64Def(AColumn.Value, NULL_INTEGER);
  if orgValue = NULL_INTEGER then
    exit;

  if TableIdMapping.IsRelevant(ATable.Tablename, AColumn.Columnname) then
  begin
    if (TableIdMapping.IdMapping.TryGetValue(orgValue, id)) then
    begin
      result := true;
      ANewValue := IntToStr(id);
      exit;
    end;

    if not AForInsert then
      exit;

    dbColumn := FDTable.Columns.Find(AColumn.Columnname);
    if IsSelbstreferenzierendeSpalte(dbColumn) then
    begin
      if not dbColumn.Nullable then
         raise Exception.Create('GetNewColumnValue: Selbstreferenzierte Spalte "' + AColumn.Columnname + '" ist nicht Nullable');
      if FImportNachlauf.IndexOf(ATable) < 0 then
          FImportNachlauf.Add(ATable);
      ANewValue := '';
      result := true;
      exit;
    end;

    if FOwner.FImportOptionen.FehlerIgnorieren then
      AKundenImport.AddLog('GetNewColumnValue: Spalte "' + AColumn.Columnname + '" mit Wert "' + AColumn.Value + '" ist nicht auflösbar --> NULL importiert')
    else
      raise Exception.Create('GetNewColumnValue: Spalte "' + AColumn.Columnname + '" mit Wert "' + AColumn.Value + '" ist nicht auflösbar');
  end;
end;

procedure TExpImpTableObj.SortImportValues;

var orgValue: int64;
    tabIdx, colIdx: Integer;
    table: TTAbleObj;
    col: TColumnObj;
    columnsRef: TStrings;
    refTables: TelObjectList<TTAbleObj>;

begin
  columnsRef := TStringList.Create;
  refTables := TelObjectList<TTAbleObj>.Create(false);
  try
    if GetRefColumns(TableName, columnsRef) then
    begin
      for tabIdx := FImportValues.Count - 1 downto 0 do
      begin
        table := FImportValues[tabIdx];
        for colIdx := 0 to columnsRef.Count - 1 do
        begin
          col := table.Columns[colIdx];
          orgValue := StrToInt64Def(col.Value, NULL_INTEGER);
          if orgValue <> NULL_INTEGER then
          begin
            refTables.Add(table);
            FImportValues.Remove(table);
            break;
          end;
        end;
      end;
    end;

    FImportValues.SortList(
          function(Item1, Item2: Pointer): Integer
          var
            id1, id2: Int64;
          begin
            Result := 0;
            if (TObject(Item1) is TTAbleObj) and (TObject(Item2) is TTAbleObj) then begin
              id1 := StrToInt64Def(TTAbleObj(Item1).GetColumnByName(FIdFeldName).Value, NULL_INTEGER);
              id2 := StrToInt64Def(TTAbleObj(Item2).GetColumnByName(FIdFeldName).Value, NULL_INTEGER);
              Result := CompareValue(id1, id2);
            end;
          end);

    if refTables.Count > 0 then
    begin
      RefTables.SortList(function(Item1, Item2: Pointer): Integer
            var
              id1, id2: Int64;
            begin
              Result := 0;
              if (TObject(Item1) is TTAbleObj) and (TObject(Item2) is TTAbleObj) then begin
                id1 := StrToInt64Def(TTAbleObj(Item1).GetColumnByName(FIdFeldName).Value, NULL_INTEGER);
                id2 := StrToInt64Def(TTAbleObj(Item2).GetColumnByName(FIdFeldName).Value, NULL_INTEGER);
                Result := CompareValue(id1, id2);
              end;
            end);

      FImportValues.AddRange(refTables);
    end;

  finally
    refTables.Free;
    columnsRef.Free;
  end;
end;

function TExpImpTableObj.GetRefColumns(ATableName: string; AColumnsRef: TStrings): Boolean;
var i: Integer;
begin
  result := false;
  for i := 0 to FDTable.ReferencedBy.Count - 1 do
    if FDTable.ReferencedBy[i].FKTableSchema + '.' + FDTable.ReferencedBy[i].FKTable = ATableName then
    begin
      if length(FDTable.ReferencedBy[i].FKFieldnames) <> 1 then
          raise EInvalidOperation.Create('length(FDTable.ReferencedBy[i].FKFieldnames) <> 1  wird nicht unterstützt!');
      AColumnsRef.Add(FDTable.ReferencedBy[i].FKFieldnames[0]);
      result := true;
    end;
end;

function TExpImpTableObj.GetSqlDebugInfo(AQuery: TSqlQuery): String;
var
  I: Integer;
begin
  if FOwner.ExportOptionen.IsExportXmlDebug then
  begin
    result := AQuery.SQL.Text;
    for I := 0 to AQuery.Params.Count - 1 do
      result := result + CR_LF + AQuery.Params[i].Name + ' = ' + VarToStr(AQuery.Params[i].Value);
  end
   else result := '';
end;

function TExpImpTableObj.HasImportValues: Boolean;
var idx: Integer;
begin
  result := false;
  if not FCanDoImport then
   exit;

  for idx := 0 to FImportValues.Count -1 do
    if not FImportValues[idx].ImportErledigt then
    begin
      result := true;
      break;
    end;
end;

function TExpImpTableObj.IsSelbstreferenzierendeSpalte(AColumn: TDBModelColumn): Boolean;
var
  ref: TDBModelRef;
  i, j: Integer;
begin
  result := false;
  if AColumn = nil then
    exit;

  for i := 0 to FDTable.ReferencesTo.Count - 1 do
  begin
    ref := FDTable.ReferencesTo[i];
    if ref.FKTableSchema + '.' + ref.FKTable <> FTableName then
      continue;
    for j := 0 to high(ref.FKSourceFieldnames) do
      if (ref.FKSourceFieldnames[j] = AColumn.name) then
      begin
        result := true;
        exit;
      end;
  end;
end;

function TExpImpTableObj.PrepareParentId(AConnection: TecConnection;
      AId: TLongId; AParent: TExpImpTableObj; AColumnName: String): boolean;
var qry: TSQLQuery;
begin
  result := false;
  qry := AConnection.CreateSQLQuery;
  try
    qry.SQL.Text := 'SELECT ' + AColumnName +
                    ' FROM ' + TableName + ' WHERE ' + FIdFeldName + ' = :IDFELD';
    qry.DBPrepare;
    qry.ParamByName('IDFELD').AsLongId := AId;
    qry.Open();
    while not qry.Eof do
    begin
      if AParent.SelectIdForExport(qry.Fields[0].AsLongId,TableName + ' / ' + FIdFeldName + ' = ' + AId.ToStr) then
        result := True;
      qry.Next;
    end;
  finally
    qry.Free;
  end;
end;

procedure TExpImpTableObj.PreparePksForImport(AConnection: TecConnection; ATable: TTAbleObj; AKundenImport: IKundenImport);
var id, orgValue: int64;
    col: TColumnObj;
    dbCol: TDBModelColumn;
begin
  // PK-Ids setzen
  col := ATable.GetColumnByName(FIdFeldName, true);
  orgValue := StrToInt64Def(col.Value, NULL_INTEGER);
  if orgValue = NULL_INTEGER then
     raise Exception.Create(TableName + ': PK-Spalte ' + FIdFeldName + ' leer!!');

  dbCol := FDTable.Columns.Find(col.Columnname);
  if (dbCol = nil) then
     raise Exception.Create(TableName + ': PK-Spalte ' + FIdFeldName + ' nicht im DB-Modell gefunden!!');

  if not TableIdMapping.IdMapping.TryGetValue(orgValue, id) then
  begin
    // Neue ID generieren
    id := ConnectionPool.GetSequenceValue(TableIdMapping.SequenceName);
    TableIdMapping.IdMapping.Add(orgValue, id);
  end;
end;

function TExpImpTableObj.PrepareId(AConnection: TecConnection; AKundenExport: IKundenExport; AId: TLongId; AExpImpManager: TExpImpObjectManager): boolean;
var refIdx: Integer;
    parent: TExpImpTableObj;
begin
  result := false;
  if FAutoPrepareParentIds then
    for refIdx := 0 to FDTable.ReferencesTo.Count-1 do
    if length(FDTable.ReferencesTo[refIdx].FKSourceFieldnames) = 1 then
    begin
      parent := AExpImpManager.GetItemByIdentifier('CREATOR.' + FDTable.ReferencesTo[refIdx].FKTable);
      if (parent <> nil) then
      begin
        if parent.ChildTables.IndexOf(TableName) >= 0 then
        begin
          if PrepareParentId(AConnection, aid, parent, FDTable.ReferencesTo[refIdx].FKSourceFieldnames[0]) then
              result := True;
        end;
      end;
    end;
end;

function TExpImpTableObj.PrepareSelectedIds(AConnection: TecConnection; AKundenExport: IKundenExport;
  AExpImpManager: TExpImpObjectManager): Boolean;
var idx: Integer;
    id : TLongId;
begin
  result := false;
  if not CanDoExport then
    exit;


  for idx := 0 to FIdsSelektiert.Count - 1 do
    if not FIdsPrepared.Contains(FIdsSelektiert[idx]) then
    begin
      id := FIdsSelektiert[idx];
      if PrepareId(AConnection, AKundenExport, id, AExpImpManager) then
        result := true;
    end;
  // IDS selektieren, damit müssten wir 90% der Selbstreferenzierten ID-Sachen erledigt kriegen
  FIdsSelektiert.Sort(false);
end;

function TExpImpTableObj.SelectIds(AConnection: TecConnection; AExpImpManager: TExpImpObjectManager):boolean;
var idx: Integer;
    child: TExpImpTableObj;
    baseSql, sql: String;
    idsChildSelects,
    idsExport: TelSimpleLongIdList;
    qry: TSqlQuery;
begin
  result := false;
  if not CanDoExport then
    exit;

  result := false;
  if FIdsSelektiert.Count = 0 then
    exit;

  if FChildTables.Count = 0 then
    exit;

  if AutoSelectChildIds then
  begin
    idsExport := TelSimpleLongIdList.Create();;
    idsChildSelects := TelSimpleLongIdList.Create();;
    qry := AConnection.CreateSQLQuery;
    try
      for idx := 0 to FIdsSelektiert.Count - 1 do
        if not FIdsChildSelected.Contains(FIdsSelektiert[idx]) then
          idsChildSelects.Add(FIdsSelektiert[idx]);

      if idsChildSelects.Count > 0 then
      begin
        for idx := 0 to FChildTables.Count - 1 do
        begin
          idsExport.AddRange(idsChildSelects);
          child := AExpImpManager.GetItemByIdentifier(FChildTables[idx]);
          if child = nil then
             raise EInvalidOperation.Create('Kein TExpImpTableObj für ' + FChildTables[idx] + ' gefunden!');
          baseSql := 'SELECT ' + child.DefaultFieldName + ',' + DefaultFieldName + CR_LF +
                 ' FROM ' + child.TableName + CR_LF +
                 ' WHERE ' + DefaultFieldName + ' ' + BASISIDLISTTAG;
          sql :=  GetSqlMitIds(baseSql, idsExport);
          while sql <> '' do
          begin
            qry.SQL.Text := sql;
            qry.Open();
            while not qry.Eof do
            begin
              if child.SelectIdForExport(qry, 'SelectId/Child by ' + DefaultFieldName + ' = ' + qry.FieldByName(DefaultFieldName).AsString) then
                  result := true;
              qry.Next;
            end;
            sql := GetSqlMitIds(baseSql, idsExport);
          end;
        end;
        FIdsChildSelected.AddRange(idsChildSelects);
      end
    finally
      idsExport.Free;
      idsChildSelects.Free;
      qry.Free;
    end;
  end;
end;

function TExpImpTableObj.SelectOther( AQuery: TSqlQuery; ATableName, AFieldName: String; AConnection: TecConnection; AKundenExport: IKundenExport; AExpImpManager: TExpImpObjectManager): boolean;
var el: TExpImpTableObj;
begin
  if AQuery.FieldByName(AFieldName).IsNull then
  begin
    result := false;
    exit;
  end;

  el := AExpImpManager.GetItemByIdentifier(ATableName);
  if el = nil then
  begin
    result := false;
    exit;
  end;
  result := el.PrepareId(AConnection, AKundenExport, AQuery.FieldByName(AFieldName).AsLongId, AExpImpManager);
end;

procedure TExpImpTableObj.SelectExportIds(AConnection: TecConnection; ASql: string);
var qry: TSqlQuery;
begin
  if not FCanDoExport then
    exit;
  qry := AConnection.CreateSQLQuery;
  try
    qry.SQL.Text := ASql;
    qry.Open();
    while not qry.Eof do
    begin
      SelectIdForExport(qry, TableName + ' / SelectExportIds / ' + ASql, '');
      qry.Next;
    end;
  finally
    qry.Free;
  end;
end;

{ TExpImpObjectManager }

function TExpImpObjectManager.AddTable(ATable: String; ATableIdMapping: TelStringObjectMap<TTableIdMapping>): TExpImpTableObj;
begin
  if GetItemByIdentifier(ATable) <> nil then
    raise EInvalidOperation.Create('TExpImpObjectManager.AddTable: Es existiert bereits ein Eintrag mit ' + ATable);
  result := TExpImpTableObj.Create(self, ATable, ATableIdMapping);
  add(result)
end;

procedure TExpImpObjectManager.CheckChilds;
var idx, idx2: Integer;
begin
  for idx := 0 to Count - 1 do
    for idx2 := 0 to Items[idx].FChildTables.Count - 1 do
      if GetItemByIdentifier(Items[idx].FChildTables[idx2]) = nil  then
        raise EInvalidOperation.Create('TExpImpObjectManager.CheckChilds / ' + Items[idx].TableName +  ': Kein Eintrag zu Child ' + Items[idx].FChildTables[idx2] + ' gefunden!');
end;

destructor TExpImpObjectManager.Destroy;
begin
  if FOwnsExportOptionen then
    FExportOptionen.Free;

  if FOwnsImportOptionen then
    FImportOptionen.Free;
  inherited;
end;

function TExpImpObjectManager.AddTable(ATableObj: TExpImpTableObj): TExpImpTableObj;
begin
  if GetItemByIdentifier(ATableObj.Identifier) <> nil then
    raise EInvalidOperation.Create('TExpImpObjectManager.AddTable: Es existiert bereits ein Eintrag mit ' + ATableObj.Identifier);
  add(ATableObj);

  if ATableObj is TExportEdNachrichtEpv then
    ObjEdepv.RegisterEpvArtObj(ATableObj, TExportEdNachrichtEpv(ATableObj).FEpvArt);
  result := ATableObj;
end;

function TExpImpObjectManager.GetItemByIdentifier(AIdentifier: String): TExpImpTableObj;
var idx: Integer;
begin
  result := nil;
  for idx := 0 to Count - 1 do
    if items[idx].Identifier = AIdentifier then
    begin
      result := items[idx];
      break;

procedure TExpImpObjectManager.FillParents;
var idx, idx2: Integer;
    master, child: TExpImpTableObj;

begin
  for idx := 0 to Count - 1 do
    items[idx].Parents.Clear;

  for idx := 0 to Count - 1 do
  begin
    master := items[idx];
    if (master.ChildTables.Count > 0) then
    begin
      for idx2 := 0 to master.ChildTables.Count - 1 do
      begin
        child := GetItemByIdentifier(Master.ChildTables[idx2]);
        if (child <> nil) and (child.Parents.IndexOf(master) <0) then
          child.Parents.Add(master);
      end;
    end;
  end;
end;

  var child: TExpImpTableObj;
      idx: Integer;
  begin
    for idx := 0 to AMaster.ChildTables.Count -1 do
    begin
      child := GetItemByIdentifier(AMaster.ChildTables[idx]);
      if child <> nil then
      begin
        // nur Disablen, wenn child keine anderen master hat!
        if child.Parents.Count = 1 then
          child.CanDoExport := false;
      end;
    end;
  end;

var idx: Integer;
begin
  for idx := 0 to Count - 1 do
    if not items[idx].CanDoExport and (items[idx].ChildTables.Count > 0) then
      DisableChilds(items[idx]);
end;

procedure TExpImpObjectManager.InitTables(ATables: TTablesObj);
var idx,  tabIdx: Integer;
begin
  for tabIdx := 0 to ATables.Childs.Count - 1 do
    for idx := 0 to Count - 1 do
     items[idx].AddImportValues(TTAbleObj(ATables.Childs.Items[tabIdx]));
end;

function TExportEDINV.PrepareId(AConnection: TecConnection; AKundenExport: IKundenExport; AId: TLongId; AExpImpManager: TExpImpObjectManager): boolean;
begin
  result := inherited PrepareId(AConnection, AKundenExport, AId, AExpImpManager);
  if FSelectEdNachricht = nil then
  begin
     FSelectEdNachricht := AConnection.CreateSQLQuery;
     FSelectEdNachricht.SQL.Text := 'SELECT DAT_ID, INV_ID, INV_STORNOINV_ID, INV_SENDER_ILN, INV_EMPF_ILN, MSC_ID FROM ' + EDINV + ' WHERE INV_ID = :INV_ID OR INV_STORNOINV_ID = :INV_ID';

    if not FSelectEdNachricht.Eof then

      if not FSelectEdNachricht.FieldByName('INV_STORNOINV_ID').IsNull and
         SelectIdForExport(FSelectEdNachricht.FieldByName('INV_STORNOINV_ID').AsLongId, 'TExportEDINV.PrepareId, ' + AId.ToStr + ' / INV_STORNOINV_ID') then

         SelectIdForExport(FSelectEdNachricht.FieldByName('INV_ID').AsLongId, 'TExportEDINV.PrepareId, ' + AId.ToStr + ' / INV_ID') then
         result := true;

      if AExpImpManager.GetItemByIdentifier(EDDAT).SelectIdForExport(FSelectEdNachricht.FieldByName('DAT_ID').AsLongId, 'TExportEDINV.PrepareId, ' + AId.ToStr + ' / DAT_ID') then
         result := true;

      if AExpImpManager.GetItemByIdentifier(EDMSC).SelectIdForExport(FSelectEdNachricht.FieldByName('MSC_ID').AsLongId, 'TExportEDINV.PrepareId, ' + AId.ToStr + ' / MSC_ID') then
         result := true;

       AKundenExport.AddToMprdList(FSelectEdNachricht.FieldByName('INV_SENDER_ILN').AsString);
       AKundenExport.AddToMprdList(FSelectEdNachricht.FieldByName('INV_EMPF_ILN').AsString);
    end;
  finally
    FSelectEdNachricht.Close;
  end;

  if FSelectEdremp = nil then
  begin
     FSelectEdremp := AConnection.CreateSQLQuery;
     FSelectEdremp.SQL.Text := 'SELECT DISTINCT REM_ID FROM ' + EDREMP + ' WHERE INV_ID = :INV_ID';
     FSelectEdremp.DBPrepare;
  end;

  FSelectEdremp.ParamByName('INV_ID').AsLongId := AId;
  try
    FSelectEdremp.Open();
    while not FSelectEdremp.Eof do
    begin
      if AExpImpManager.GetItemByIdentifier(EDREM).SelectIdForExport(FSelectEdremp.FieldByName('REM_ID').AsLongId, 'TExportEDINV.PrepareId, ' + AId.ToStr + ' / REM_ID') then
         result := true;
      FSelectEdremp.Next;
    end;
  finally
    FSelectEdremp.Close;
  end;
end;

function TExportEdNachrichtEpv.PrepareId(AConnection: TecConnection; AKundenExport: IKundenExport;
  AId: TLongId; AExpImpManager: TExpImpObjectManager): boolean;
begin
  result := inherited PrepareId(AConnection, AKundenExport, AId, AExpImpManager);
  if FEpvArt >= 0 then
  begin
    if FSelectEdepv = nil then
    begin
       FSelectEdepv := AConnection.CreateSQLQuery;
       FSelectEdepv.SQL.Text := 'SELECT EPR_ID, EPV_ID FROM ' + EDEPV + ' WHERE EDI_ID = :EDI_ID' +
                 IfThen(FEpvArt >= 0, ' AND EPV_ART = :EPV_ART', '');
       FSelectEdepv.DBPrepare;
    end;

    FSelectEdepv.ParamByName('EDI_ID').AsLongId := AId;
    FSelectEdepv.ParamByName('EPV_ART').AsInteger := FEpvArt;
    try
      FSelectEdepv.open;
      while not FSelectEdepv.Eof do
      begin

{ TTableIdMapping }

procedure TTableIdMapping.Clear;
begin
  IdMapping.Clear;
end;

constructor TTableIdMapping.Create(ATableName, AIdFieldName, ASequenceName: String);
begin
  FTableName := ATableName;
  FIdFieldName := AIdFieldName;
  FSequenceName := ASequenceName;
  FIdMapping := TIdMapping.Create;
  FDTable := TDBModelUtils.GetTables.Find(TableName);
  if FDTable = nil then
    raise EInvalidOperation.Create('Tabelle ' + ATableName + ' nicht in DB-Modell gefunden');
end;

destructor TTableIdMapping.Destroy;
begin
  IdMapping.Free;
end;

function TTableIdMapping.IsRelevant(ATableName, AColumnName: String): Boolean;
var i: Integer;
begin
  result := false;
  if (ATableName = TableName) and (AColumnName = IdFieldName) then
  begin
    result := true;
    exit;
  end;
  if FDTable <> nil then
  begin
    for i := 0 to FDTable.ReferencedBy.Count - 1 do
      if FDTable.ReferencedBy[i].FKTableSchema + '.' + FDTable.ReferencedBy[i].FKTable = ATableName then
      begin
        if length(FDTable.ReferencedBy[i].FKFieldnames) <> 1 then
            raise EInvalidOperation.Create('length(FDTable.ReferencedBy[i].FKFieldnames) <> 1  wird nicht unterstützt!');
        if (AColumnName = FDTable.ReferencedBy[i].FKFieldnames[0]) then
        begin
          result := true;
          break;
        end;
      end;
  end;
end;

        result := GetNewColumnValue(ATable, ATable.GetColumnByName('EDI_ID'), AKundenImport, newValue, false);
        // SonderLocke: Nur Importieren, falls "Übersetzbar"
      end;
    end
    else
      result := False;
  end;
end;

function TExportEdepv.GetNewColumnValue(ATable: TTAbleObj; AColumn: TColumnObj;
  AKundenImport: IKundenImport; Var ANewValue: String; AForInsert: Boolean): Boolean;
var column: TColumnObj;
    orgId, id: Int64;
    art: Integer;
begin
  result := inherited GetNewColumnValue(ATable, AColumn, AKundenImport, ANewValue, AForInsert);
  if result then
    exit;

  if ATable.Tablename <> TableName then
    exit;

  if AColumn.Columnname <> 'EDI_ID' then
    exit;

  orgId := StrToIntDef(AColumn.Value, NULL_INTEGER);
  id := orgId;
  column := ATable.GetColumnByName('EPV_ART');
  art := StrToIntDef(column.Value, -1);
  if FEpvArt2Obj.ContainsKey(art) then
     result := FEpvArt2Obj.Values[art].TableIdMapping.IdMapping.TryGetValue(orgId, id);

  if result then
    ANewValue := IntToStr(id);
end;

procedure TExportEdepv.RegisterEpvArtObj(AObj: TExpImpTableObj; AEpvArt: Integer);
begin
  if (AObj = nil) or (AEpvArt < 0) then exit;

  if FEpvArt2Obj.ContainsKey(AEpvArt) then
    raise EInvalidArgument.Create('RegisterEpvArtObj: Es existiert bereits ein Eintrag für die EpvArt ' + IntTostr(AEpvArt));
  FEpvArt2Obj.Add(AEpvArt, AObj)
end;
