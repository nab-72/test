function LoadXmlFromFile(const FileName: string): IXMLDocument;
var
  OldDisableMemoryManagerStacktraces: Boolean;
begin
  Result := TelXmlDocument.Create(True, True, True, True);
  OldDisableMemoryManagerStacktraces := SetDisableMemoryManagerStacktraces(True);
  try
    Result.LoadFromFile(FileName);
  finally
    SetDisableMemoryManagerStacktraces(OldDisableMemoryManagerStacktraces);
  end;
end;

procedure SetParamValueByType(ADBParam: TParam; const AValue: string; ADataType: TFieldType);
var
  Int64Value: Int64;
begin
  if AValue = '' then
  begin
    ClearDBParam(ADBParam, ADataType);
    Exit;
  end;

  case ADataType of
    ftUnknown, ftString, ftWideString, ftFixedChar, ftFixedWideChar:
      ADBParam.AsString := AValue;
    ftShortint, ftByte, ftSmallint, ftWord, ftInteger, ftLongWord, ftLargeint:
      SetDBParam(ADBParam, StrToInt64(AValue));
    ftMemo, ftWideMemo, ftOraClob: // CLOB
      ADBParam.AsString := AValue;
    ftDate:
      SetDBParamDateTime(ADBParam, StrToDate(AValue));
    ftDateTime, ftTimeStamp, ftOraTimeStamp:
      SetDBParamDateTime(ADBParam, StrToDateTime(AValue));
    ftFloat, ftCurrency, ftExtended:
      SetDBParamFloat(ADBParam, StrToFloat(AValue));
    ftBCD, ftFMTBcd:
      if TryStrToInt64(AValue, Int64Value) then
        SetDBParam(ADBParam, Int64Value)
      else
        ADBParam.AsFMTBCD := StrToBcd(AValue);
  end;
end;

procedure TStammdatenExportImportManager.AddUsedValue(AMndId: Integer; const AColumnName: string; ADataType: TFieldType; const AValue: String);
var tab: TStammdatenTab;
begin
  if Trim(AValue) = '' then
    exit;

  if FAufloesbareStammdatenMap.TryGetValue(AColumnName, tab) then
    tab.AddUsedValue(AMndId, AValue)
  else
  if TurnusStammdatenRec.CanResolve(AColumnName) then
    TurnusStammdatenRec.AddUsedValue(AMndId, AColumnName, ADataType, AValue)
   else
  if TurnusStammdatenAbl.CanResolve(AColumnName) then
    TurnusStammdatenAbl.AddUsedValue(AMndId, AColumnName, ADataType, AValue)
   else
  if TurnusStammdatenAbt.CanResolve(AColumnName) then
    TurnusStammdatenAbt.AddUsedValue(AMndId, AColumnName, ADataType, AValue)
   else
  if TurnusStammdatenVorausZ.CanResolve(AColumnName) then
    TurnusStammdatenVorausZ.AddUsedValue(AMndId, AColumnName, ADataType, AValue)
end;

function TStammdatenExportImportManager.CanResolve(AColumnName: String): Boolean;
begin
  result := FAufloesbareStammdatenMap.ContainsKey(AColumnName);

  if not result then
    result := TurnusStammdatenRec.CanResolve(AColumnName);
  if not result then
    result := TurnusStammdatenAbl.CanResolve(AColumnName);
  if not result then
    result := TurnusStammdatenAbt.CanResolve(AColumnName);
  if not result then
    result := TurnusStammdatenVorausZ.CanResolve(AColumnName);
end;

procedure TStammdatenExportImportManager.Clear;
var idx: Integer;
begin
  for idx := 0 to FAufloesbareStammdatenList.Count - 1 do
    TStammdatenTab(FAufloesbareStammdatenList[idx]).Clear;

  FTurnusStammdatenRec.Clear;
  FTurnusStammdatenAbt.Clear;
  FTurnusStammdatenAbl.Clear;
  FTurnusStammdatenVorausZ.Clear;
end;

constructor TStammdatenExportImportManager.Create(AConnection: TecConnection; ATables: TTablesObj; AStammdatenWieExport: Boolean;
      AStandardStammdaten: Boolean; AIsImport: Boolean; AStammdatenMitFallback: Boolean; AFehlerIgnorieren: Boolean;

begin
  FConnection := AConnection;
  FAufloesbareStammdatenMap := TelStringObjectMap<TStammdatenTab>.Create;
  FAufloesbareStammdatenMap.OwnsObjects := false;
  FAufloesbareStammdatenList := TObjectList.Create(true);
  FStammdatenWieExport := AStammdatenWieExport;
  FStandardStammdaten := AStandardStammdaten;
  FIsImport := AIsImport;
  FNbtStammdaten := nil;
  FStammdatenMitFallback := AStammdatenMitFallback;
  FFehlerIgnorieren := AFehlerIgnorieren;
  FDummyStammdatenAnlegen := ADummyStammdatenAnlegen;
  FTables := ATables;

  FTurnusStammdatenRec := TTurnusStammdaten.Create(self, 0);
  FTurnusStammdatenAbt := TTurnusStammdaten.Create(self, 1);
  FTurnusStammdatenAbl := TTurnusStammdaten.Create(self, 2);
  FTurnusStammdatenVorausZ := TTurnusStammdaten.Create(self, 3);
end;

function TStammdatenExportImportManager.GetDefaultValue(AColumnName, AAltId: String; AMndId: Integer): String;
var tab: TStammdatenTab;
begin
  result := '';
  if FAufloesbareStammdatenMap.TryGetValue(AColumnName, tab) then
    result := tab.GetDefaultValue(AAltId, AMndId);
end;

function TStammdatenExportImportManager.GetValue(AColumnName, AAltId: String): String;
var tab: TStammdatenTab;
begin
  result := '';
  if FStammdatenWieExport then
    result := AAltId // Keine Übersetzung!!
  else
  if FAufloesbareStammdatenMap.TryGetValue(AColumnName, tab) then
  begin
    if not tab.ValuesResolved then
      tab.Resolve(Connection, MandantId);
    result := tab.GetResolvedValue(AAltId);
  end
   else
  if TurnusStammdatenRec.CanResolve(AColumnName) then
    result := TurnusStammdatenRec.GetResolvedValue(AColumnName, AAltId)
   else
  if TurnusStammdatenAbl.CanResolve(AColumnName) then
    result := TurnusStammdatenAbl.GetResolvedValue(AColumnName, AAltId)
   else
  if TurnusStammdatenAbt.CanResolve(AColumnName) then
    result := TurnusStammdatenAbt.GetResolvedValue(AColumnName, AAltId)
   else
  if TurnusStammdatenVorausZ.CanResolve(AColumnName) then
    result := TurnusStammdatenVorausZ.GetResolvedValue(AColumnName, AAltId)
end;

function TStammdatenExportImportManager.GetValueWithDefaultFallback(AColumnName, AAltId: String): String;
begin
  result := GetValue(AColumnName, AAltId);
  if (result = '') and StandardStammdaten then
    result := GetDefaultValue(AColumnName, AAltId, MandantId);
end;

function TStammdatenExportImportManager.LoadExportData(ADocRoot: IXMLNode; AExportMndId: Integer): Boolean;
var cNode, node: IXMLNode;
    idx, idx2: Integer;
    tabName: String;
    tab: TStammdatenTab;
begin
  Clear;
  result := false;
  if ADocRoot = nil then
    exit;

  node := TXmlUtils.GetAndCheckChildXmlNode(ADocRoot, TAG_STAMMDATEN, false);
  if node = nil then
    exit;

  result := true;
  if not node.HasChildNodes then
   exit;

  for idx := 0 to node.ChildNodes.Count -1 do
  begin
   cNode := node.ChildNodes[idx];
   if cNode.NodeName <> TAG_RESOLVE_TABLE then
     continue;

   if not cNode.HasAttribute(TAG_RESOLVE_TABLENAME) then
     Continue;

    tabName := cNode.Attributes[TAG_RESOLVE_TABLENAME];
    if TurnusStammdatenRec.TableName = tabName then
      TurnusStammdatenRec.LoadExportData(cNode, AExportMndId)
    else
    if TurnusStammdatenAbl.TableName = tabName then
      TurnusStammdatenAbl.LoadExportData(cNode, AExportMndId)
    else
    if TurnusStammdatenAbt.TableName = tabName then
      TurnusStammdatenAbl.LoadExportData(cNode, AExportMndId)
    else
    if TurnusStammdatenRec.TableName = tabName then
      TurnusStammdatenVorausZ.LoadExportData(cNode, AExportMndId)
    else
    for idx2 := 0 to FAufloesbareStammdatenList.Count - 1 do
    begin
      tab := TStammdatenTab(FAufloesbareStammdatenList[idx2]);
      if tab.TableName = tabName then
      begin
        tab.LoadExportData(cNode, AExportMndId);
        break;
      end;
    end;

  end;
end;

procedure TStammdatenExportImportManager.RegisterStammdaten(AStammdatenTab: TStammdatenTab);
var tab: TStammdatenTab;
begin
  if FAufloesbareStammdatenMap.TryGetValue(AStammdatenTab.KeyColName, tab) then
  begin
    if (tab.TableName <> AStammdatenTab.TableName) or
       (tab.FBaseExportSql <> AStammdatenTab.FBaseExportSql) then
        raise EInvalidOperation.Create('Interner Fehler, ' + AStammdatenTab.KeyColName + ' wurde bereits per RegisterAufloesbareStammdaten übergeben (Tabelle: ' + Tab.FTableName + ')');
    exit;
  end;

  FAufloesbareStammdatenMap.Add(AStammdatenTab.KeyColName, AStammdatenTab);
  FAufloesbareStammdatenList.Add(AStammdatenTab)
end;

procedure TStammdatenExportImportManager.Resolve;

      raise EInvalidOperation.Create('Interner Fehler, ' + ASpaltenNameAlternativ + '  ist bereits für ' + ASdTableName + ' registriert');

procedure TStammdatenTab.PreSelectOtherUsedStammdatenValues(AConnection: TecConnection);
var qry: TSQLQuery;
    idx, cIdx: Integer;
begin
  // ggf. für alle UsedValues die auf andere Stammdaten zeigen (z.B. LPR.NBT_ID)
  // AddUsedStammdatenValues aufrufen!
  if not HasUsedValues or not FAutoPreSelectOtherUsedStammdatenValues then
    exit;
  qry := AConnection.CreateSQLQuery;
  try
    qry.SQL.Text := GetExportSql(false);
    qry.DBPrepare;
    for Idx := 0 to FUsedValues.Count - 1 do
    begin
      SetParamValueByType(qry.ParamByName(KEY_VALUE), GetUsedValue(idx), KeyColDataType);
      if MitMndIdSpalte then
        qry.ParamByName(KEY_MND_ID).AsInteger := GetUsedValueMndId(idx);
      try
        qry.Open();
        if not qry.Eof then
          for cIdx := 0 to qry.Fields.Count - 1 do
            if (qry.Fields[cIdx].FieldName <> FKeyColName) and not qry.Fields[cIdx].IsNull then
              FParent.AddUsedValue(GetUsedValueMndId(idx), qry.Fields[cIdx].FieldName, qry.Fields[cIdx].DataType, qry.Fields[cIdx].AsString);
      finally
        qry.Close();
      end;
    end;
  finally
    qry.free;
  end;
end;

end;

procedure TStammdatenTab.WriteExportData(AConnection: TecConnection;  AFileStream: TStream);
var qry: TSQLQuery;
    idx, cIdx: Integer;
begin
  if not HasUsedValues then
    exit;
  TXmlUtils.writeXmlStr(AFileStream,'  <' +  TAG_RESOLVE_TABLE + ' ' + TAG_RESOLVE_TABLENAME + '="' + Tablename + '">');
  qry := AConnection.CreateSQLQuery;
  try
    qry.SQL.Text := GetExportSql(false);
    qry.DBPrepare;
    for Idx := 0 to FUsedValues.Count - 1 do
    begin
      SetParamValueByType(qry.ParamByName(KEY_VALUE), GetUsedValue(idx), KeyColDataType);
      if MitMndIdSpalte then
        qry.ParamByName(KEY_MND_ID).AsInteger := GetUsedValueMndId(idx);
      try
        qry.Open();
        if not qry.Eof then
        begin
          TXmlUtils.writeXmlStr(AFileStream, '    <' + TAG_RESOLVE_TABLE_ITEM + '>');
          TXmlUtils.writeXmlStr(AFileStream, '    <' + TAG_EXPORT_MND + '>' + IntToStr(GetUsedValueMndId(idx)) + '</' + TAG_EXPORT_MND + '>');
          for cIdx := 0 to qry.Fields.Count - 1 do
            TXmlUtils.writeXmlStr(AFileStream, '    <' + qry.Fields[cIdx].FieldName + '>' + TXmlUtils.ReplaceBackend2Xml(qry.Fields[cIdx].AsString, qry.Fields[cIdx].DataType, false) + '</' + qry.Fields[cIdx].FieldName + '>');

          TXmlUtils.writeXmlStr(AFileStream, '    </' + TAG_RESOLVE_TABLE_ITEM + '>');
        end;
      finally
        qry.Close();
      end;
    end;
  finally
    qry.free;
  end;
  TXmlUtils.writeXmlStr(AFileStream, '  </' + TAG_RESOLVE_TABLE + '>');
end;

class function TXmlUtils.ReplaceBackend2Xml(AValueStr: String; AFieldType: TFieldType = ftString; AQuoteStr:Boolean = true): String;
var value: String;
begin
  value := AValueStr;
  if (AFieldType = ftString) then
  begin
    if AQuoteStr then
      value := QuotedStr(value);
  end
  else if ((AFieldType = ftBCD)
        or (AFieldType = ftFMTBcd)) then
  begin
    value := ReplaceStr(value, ',', '.');
  end
  else if ((AFieldType = ftTimeStamp) and (value <> '')) then
  begin
    if (Length(value) = 19) then
      value := 'TO_DATE(' + QuotedStr(value) + ', ' + QuotedStr('DD.MM.YYYY HH24:MI:SS') + ')'
    else
      value := 'TO_DATE(' + QuotedStr(value) + ', ' + QuotedStr('DD.MM.YYYY') + ')';
  end;
  value := Stringreplace(value, '&', '&amp;', [rfReplaceAll]);
  value := Stringreplace(value, 'ö', '&#246;', [rfReplaceAll]);
  value := Stringreplace(value, 'ä', '&#228;', [rfReplaceAll]);
  value := Stringreplace(value, 'ü', '&#252;', [rfReplaceAll]);
  value := Stringreplace(value, 'Ö', '&#214;', [rfReplaceAll]);
  value := Stringreplace(value, 'Ä', '&#196;', [rfReplaceAll]);
  value := Stringreplace(value, 'Ü', '&#220;', [rfReplaceAll]);
  value := Stringreplace(value, '>', '&gt;', [rfReplaceAll]);
  value := Stringreplace(value, '<', '&lt;', [rfReplaceAll]);
  value := Stringreplace(value, 'ß', '&szlig;', [rfReplaceAll]);
  result := value;
end;

class function TXmlUtils.ReplaceXml2Backend(AValueStr: String; AEntferneHochkomma: Boolean = true): String;
begin
  Result := Stringreplace(AValueStr, '&amp;', '&', [rfReplaceAll]);
  Result := Stringreplace(Result, '&#246;', 'ö', [rfReplaceAll]);
  Result := Stringreplace(Result, '&#228;', 'ä', [rfReplaceAll]);
  Result := Stringreplace(Result, '&#252;', 'ü', [rfReplaceAll]);
  Result := Stringreplace(Result, '&#214;', 'Ö', [rfReplaceAll]);
  Result := Stringreplace(Result, '&#196;', 'Ä', [rfReplaceAll]);
  Result := Stringreplace(Result, '&#220;', 'Ü', [rfReplaceAll]);
  Result := Stringreplace(Result, '&gt;', '>', [rfReplaceAll]);
  Result := Stringreplace(Result, '&lt;', '<',  [rfReplaceAll]);
  Result := Stringreplace(Result, '&szlig;', 'ß',  [rfReplaceAll]);
  if AEntferneHochkomma then
    result := EntferneHochkomma(result);
end;

class procedure TXmlUtils.WriteXmlStr(AFileStream: TStream; AXmlStr: String; ANewLine:Boolean = true);
var
  Bytes: TBytes;
begin
  if (ANewLine) then
    AXmlStr := CR_LF + AXmlStr;

  Bytes := TEncoding.Default.GetBytes(AXmlStr);
  if Bytes <> nil then
    AFileStream.Write(Bytes[0], Length(Bytes));
end;

class function TXmlUtils.GetAndCheckChildXmlNode(AXmlParent: IXMLNode; ALocalNameChild: String; ARequired: Boolean = true):IXMLNode;
begin
  if not AXmlParent.HasChildNodes then
  begin
   if ARequired then
     raise EFilerError.Create('Ungültige XML-Datei, <' + ALocalNameChild + '> erwartet');
    result := nil;
    exit;
  end;

  try
    result := AXmlParent.ChildNodes.FindNode(ALocalNameChild)
  except
    result := nil;
  end;

  if (result = nil) or (result.LocalName <> ALocalNameChild) then
  begin
    if ARequired then
      raise EFilerError.Create('Ungültige XML-Datei, Tag "' + ALocalNameChild + '" erwartet');
    result := nil;
  end;
end;

function TExportEckdaten.LoadExportData(ADocRoot: IXMLNode): Boolean;
Const
  TAG_EXPORT_UBA = 'ExportMndUnbundlingsicht'; // Outdatet

var nodeEckdaten, node, kfgNode : IXMLNode;
    currMnd, kfgBeschreibung, kfgWert: String;
    mndInfo: TExportMndInfo;
begin
  result := false;
  FExportAlleMndIds.Clear;
  Clear(false);

  if (ADocRoot = nil) then
    exit;

  nodeEckdaten := TXmlUtils.GetAndCheckChildXmlNode(ADocRoot, TAG_ECKDATEN, false);
  if (nodeEckdaten = nil) then
    exit;

  FExportVonWilkenInternerUmgebung := nodeEckdaten.ChildNodes[TAG_INTERNER_WILKEN_KUNDE] <> nil;
  if nodeEckdaten.HasAttribute(TAG_XML_VERSIONSNUMMER) then
    FExportXmlVersion := TXmlUtils.ReplaceXml2Backend(nodeEckdaten.Attributes[TAG_XML_VERSIONSNUMMER]);

  node := TXmlUtils.GetAndCheckChildXmlNode(nodeEckdaten, TAG_ENERGYVERSION, false);
  if node <> nil then
    FEnergyExportVersion := TXmlUtils.ReplaceXml2Backend(node.Text);

  node := TXmlUtils.GetAndCheckChildXmlNode(nodeEckdaten, TAG_EXPORT_MND, false);
  if node <> nil then
  begin
    FExportMndId := StrToIntDef(node.Text, NULL_INTEGER);
    MndInfo := TExportMndInfo.Create(ExportMndId);
    try
      node := TXmlUtils.GetAndCheckChildXmlNode(nodeEckdaten, TAG_EXPORT_MND_INFO, false);
      if node = nil then
      begin
        node := TXmlUtils.GetAndCheckChildXmlNode(nodeEckdaten, TAG_EXPORT_UBA, false);
        if node <> nil then
          mndinfo.Unbundlingart := StrToIntDef(node.Text, NULL_INTEGER);
        FExportMndInfo.AddObject(mndInfo.MndId, mndInfo);
        MndInfo := nil;
      end;
    finally
      FreeAndNil(MndInfo);
    end;
  end;

  node := TXmlUtils.GetAndCheckChildXmlNode(nodeEckdaten, TAG_EXPORT_TIME, false);
  if node <> nil then
    FExportZeitpunkt := StrToDateTimeDef(node.Text, NULL_DATE);

  node := TXmlUtils.GetAndCheckChildXmlNode(nodeEckdaten, TAG_EXPORT_MNDS_KOMPLETT, false);
  if node <> nil then
    FExportAlleMndIds.CommaText := TXmlUtils.ReplaceXml2Backend(node.Text);

  node := TXmlUtils.GetAndCheckChildXmlNode(nodeEckdaten, TAG_EXPORT_MIT_PRD, false);
  if node <> nil then
    FExportMitPrd := StrToIntDef(node.Text, NULL_INTEGER);

  node := TXmlUtils.GetAndCheckChildXmlNode(nodeEckdaten, TAG_INITIAL_KNDIDS, false);
  if node <> nil then
    FInitialKndIdCommaList := TXmlUtils.ReplaceXml2Backend(node.Text);

  node := TXmlUtils.GetAndCheckChildXmlNode(nodeEckdaten, TAG_EXPORT_IDENTIFIER, false);
  if node <> nil then
    FExportIdentifier := TXmlUtils.ReplaceXml2Backend(node.Text);

  node := TXmlUtils.GetAndCheckChildXmlNode(nodeEckdaten, TAG_EXPORT_MIT_RECHNUNGSDATEN, false);
  if node <> nil then
  begin
    FExportMitRechnungsdaten.CommaText := TXmlUtils.ReplaceXml2Backend(node.Text);
    FExportMitRechnungsdatenVorhanden := true;
  end
  else
    FExportMitRechnungsdatenVorhanden := false;

  node := TXmlUtils.GetAndCheckChildXmlNode(nodeEckdaten, TAG_EXPORT_ESM_MND, false);
  if node <> nil then
  begin
    FExportEsmMnd := StrToIntDef(TXmlUtils.ReplaceXml2Backend(node.Text), NULL_INTEGER);
    FExportEsmMndVorhanden := true;
  end
  else
    FExportEsmMndVorhanden := false; // Unbekannt, nicht exportiert

  node := nodeEckdaten.ChildNodes.First;
  while (node <> nil) do
  begin
    if node.LocalName = TAG_EXPORT_MND_INFO then
    begin
      if not FExportMndInfo.TryGetValue(StrToIntDef(node.Attributes['MND_ID'], NULL_INTEGER), mndInfo) then
      begin
        mndInfo := TExportMndInfo.Create(StrToIntDef(node.Attributes['MND_ID'], NULL_INTEGER));
        FExportMndInfo.AddObject(mndInfo.MndId, mndInfo);
      end;

      MndInfo.Unbundlingart := StrToIntDef(node.Attributes['UNBUNDLING'], NULL_INTEGER);
      MndInfo.MandantName := node.Attributes['MANDANTNAME'];
    end
     else
    if node.LocalName = TAG_EXPORT_KFG_INFO then
    begin
      currMnd := node.Attributes['MND_ID'];
      FKonfigInfos.Add(CR_LF + 'Konfiguration Export-Mandant ' + currMnd);
      FKonfigInfos.add('------------------------------------------');
      kfgNode := node.ChildNodes.First;
      while kfgNode <> nil do
      begin
        kfgBeschreibung := FKonfigExportMatchCodes.Values[kfgNode.Attributes['MATCHCODE']];
        if kfgBeschreibung = '' then
          kfgBeschreibung := kfgNode.Attributes['MATCHCODE'];
        if VarIsNull(kfgNode.Attributes['WERT']) then
          kfgWert := ''
        else
          kfgWert := kfgNode.Attributes['WERT'];
        FKonfigInfos.add('- ' + kfgBeschreibung + ': ' + kfgWert);
        kfgNode := kfgNode.NextSibling;
      end;
    end;
    node := node.NextSibling;
  end;

  node :=  TXmlUtils.GetAndCheckChildXmlNode(ADocRoot, TGpAdrInfoExportManager.TAG_GPINFOS, false);
  FExportHasGpnInfos := (node <> nil) and (node.HasChildNodes);

  node :=  TXmlUtils.GetAndCheckChildXmlNode(ADocRoot, TGpAdrInfoExportManager.TAG_ABNINFOS, false);
  FExportHasAbnInfos := (node <> nil) and (node.HasChildNodes);

  result := true;
end;

function TEdiCodeStammdaten.LoadExportData(ANode: IXMLNode; AExportMndId: Integer): Boolean;
var idx: Integer;
    ediCode: String;
    msg: String;
    other, vgl : TEdiCodeData;
begin
  result := false;
  if ANode = nil then
    exit;
  if not ANode.HasAttribute(TAG_RESOLVE_TABLENAME) then
    exit;
  if ANode.Attributes[TAG_RESOLVE_TABLENAME] <> TableName then
    exit;
  Clear;
  result := true;
  if not ANode.HasChildNodes then
   exit;

  for idx := 0 to ANode.ChildNodes.Count - 1 do
  begin
    if ANode.ChildNodes[idx].HasAttribute(KEY_MND_ID) and
       (ANode.ChildNodes[idx].Attributes[KEY_MND_ID] <> IntToStr(AExportMndId)) then
    begin
      continue;
    end;
    ediCode := ANode.ChildNodes[idx].Text;

    other := nil;
    for vgl in FResolveDataMap.ValueCollection do
      if vgl.EdiCode = ediCode then
      begin
        other := vgl;
        break;
      end;

    if other <> nil then
    begin
      if other.IdAlt = ANode.ChildNodes[idx].Attributes[KEY_PK_VALUE] then
        continue; // Doppelter Datensatz, eigentlich auch Datenschrott, aber das können wir ignorieren

      msg := 'Datenfehler ' + FTableName + ': ' +
          ' Der EdiCode ' + ediCode + ' wurde mit unterschiedlichen IDs exportiert: ' + ANode.ChildNodes[idx].Attributes[KEY_PK_VALUE] + ' und ' +
                 other.IdAlt + '. Das ist ein Datenfehler auf dem Altsystem!';

      FParent.Log.Add(msg);
      if not FParent.FFehlerIgnorieren then
          raise EInvalidOperation.Create(msg);
    end;
    FResolveDataMap.AddObject(ANode.ChildNodes[idx].Attributes[KEY_PK_VALUE], TEdiCodeData.Create(ediCode, ANode.ChildNodes[idx].Attributes[KEY_PK_VALUE]));
  end;
end;

procedure TEdiCodeStammdaten.Resolve(AConnection: TecConnection; AMndId: Integer);
var idx: Integer;
    qry: TSQLQuery;
begin
  if FInResolving then
    exit;
  StartResolving;
  qry := AConnection.CreateSQLQuery;
  try
    qry.SQL.Text := GetExportSql(true);
    if MitMndIdSpalte then
      qry.SQL.Text := AddBedingung(qry.SQL.Text, 'MND_ID = ' + IntToStr(Mandant_Id));
    qry.SQL.Text := AddBedingung(qry.SQL.Text, FEdiCodeColum + ' = :' + KEY_EDICODE);
    for idx := 0 to FResolveDataMap.Count - 1 do
    begin
      if FResolveDataMap.Objects[idx].IdResolved <> '' then
         Continue;
      qry.ParamByName(KEY_EDICODE).AsString := FResolveDataMap.Objects[idx].EdiCode;
      qry.Open;
      try
        if not qry.Eof then
        begin
          FResolveDataMap.Objects[idx].IdResolved := qry.FieldByName(KEY_PK_VALUE).AsString;
          AddResolvedValue(FResolveDataMap.Objects[idx].IdAlt, FResolveDataMap.Objects[idx].IdResolved);
        end;
      finally
        qry.Close;
      end;
    end;
  finally
    qry.Free;
    EndResolving;
  end;
end;

procedure TEdiCodeStammdaten.WriteExportData(AConnection: TecConnection; AFileStream: TStream);
var qry: TSQLQuery;
    idx: Integer;
begin
  if not HasUsedValues then
    exit;
  TXmlUtils.writeXmlStr(AFileStream,'  <' +  TAG_RESOLVE_TABLE + ' ' + TAG_RESOLVE_TABLENAME + '="' + Tablename + '">');
  qry := AConnection.CreateSQLQuery;
  try
    qry.SQL.Text := GetExportSql(false);
    qry.DBPrepare;
    for Idx := 0 to FUsedValues.Count - 1 do
    begin
      SetParamValueByType(qry.ParamByName(KEY_VALUE), GetUsedValue(idx), KeyColDataType);
      if MitMndIdSpalte then
        qry.ParamByName(KEY_MND_ID).AsInteger := GetUsedValueMndId(idx);
      try
        qry.Open();
        if not qry.Eof then
          TXmlUtils.writeXmlStr(AFileStream,
          '    <' + TAG_RESOLVE_TABLE_ITEM + ' ' +  KEY_PK_VALUE + ' = "' + qry.FieldByName(KEY_PK_VALUE).AsString + '"' +
          ifThen(MitMndIdSpalte, ' ' + KEY_MND_ID + ' = "' + qry.ParamByName(KEY_MND_ID).AsString + '"', '') +  '>' + qry.FieldByName(KEY_EDICODE).AsString + '</' + TAG_RESOLVE_TABLE_ITEM + '>');
      finally
        qry.Close();
      end;
    end;
  finally
    qry.free;
  end;
  TXmlUtils.writeXmlStr(AFileStream, '  </' + TAG_RESOLVE_TABLE + '>');
end;

  msg, ediCode, Bezeichnung: String;
  other, vgl: TEdiCodeMitBezData;
begin
  result := false;
  if ANode = nil then
    exit;
  if not ANode.HasAttribute(TAG_RESOLVE_TABLENAME) then
    exit;
  if ANode.Attributes[TAG_RESOLVE_TABLENAME] <> TableName then
    exit;
  Clear;
  result := true;
  if not ANode.HasChildNodes then
   exit;

  for idx := 0 to ANode.ChildNodes.Count - 1 do
  begin
    if ANode.ChildNodes[idx].HasAttribute(KEY_MND_ID) and
       (ANode.ChildNodes[idx].Attributes[KEY_MND_ID] <> IntToStr(AExportMndId)) then
    begin
      continue;
    end;
    ediCode := ANode.ChildNodes[idx].Text;
    Bezeichnung := QuotedStr('IMP ' + ediCode);

    if ((ANode.ChildNodes[idx].HasAttribute(KEY_BEZEICHNUNG)) and
        (ANode.ChildNodes[idx].Attributes[KEY_BEZEICHNUNG] <> NULL_STRING)) then
      Bezeichnung := TXmlUtils.ReplaceXml2Backend(ANode.ChildNodes[idx].Attributes[KEY_BEZEICHNUNG], False);

    other := nil;
    for vgl in FResolveDataMap.ValueCollection do
      if vgl.EdiCode = ediCode then
      begin
        other := vgl;
        break;
      end;

    if other <> nil then
    begin
      if other.IdAlt = ANode.ChildNodes[idx].Attributes[KEY_PK_VALUE] then
        continue; // Doppelter Datensatz, eigentlich auch Datenschrott, aber das können wir ignorieren

      msg := 'Datenfehler ' + FTableName + ': ' +
          ' Der EdiCode ' + ediCode + ' wurde mit unterschiedlichen IDs exportiert: ' + ANode.ChildNodes[idx].Attributes[KEY_PK_VALUE] + ' und ' +
                 other.IdAlt + '. Das ist ein Datenfehler auf dem Altsystem!';

      FParent.Log.Add(msg);
      if not FParent.FFehlerIgnorieren then
          raise EInvalidOperation.Create(msg);
    end;

    FResolveDataMap.AddObject(ANode.ChildNodes[idx].Attributes[KEY_PK_VALUE], TEdiCodeMitBezData.Create(ediCode, ANode.ChildNodes[idx].Attributes[KEY_PK_VALUE], Bezeichnung));
  end;

procedure TEdiCodeMitBezStammdaten.Resolve(AConnection: TecConnection; AMndId: Integer);
var
  idx: Integer;
  qry: TSQLQuery;
  ImpBezeichnung: String;
begin
  if FInResolving then
    exit;
  StartResolving;
  qry := AConnection.CreateSQLQuery;
  try
    qry.SQL.Text := GetExportSql(true);

    if MitMndIdSpalte then
      qry.SQL.Text := AddBedingung(qry.SQL.Text, 'MND_ID = ' + IntToStr(Mandant_Id));

    qry.SQL.Text := AddBedingung(qry.SQL.Text, FEdiCodeColum + ' = :' + KEY_EDICODE);
    for idx := 0 to FResolveDataMap.Count - 1 do
    begin
      if FResolveDataMap.Objects[idx].IdResolved <> '' then
         Continue;
      qry.ParamByName(KEY_EDICODE).AsString := FResolveDataMap.Objects[idx].EdiCode;
      qry.Open;
      try
        if not qry.Eof then
        begin
          FResolveDataMap.Objects[idx].IdResolved := qry.FieldByName(KEY_PK_VALUE).AsString;
          AddResolvedValue(FResolveDataMap.Objects[idx].IdAlt, FResolveDataMap.Objects[idx].IdResolved);
        end
        else
        begin
          // FResolveDataMap.Objects[idx].Bezeichnung ist bereits ohne Hochkommas
          ImpBezeichnung := QuotedStr('IMP ' + FResolveDataMap.Objects[idx].Bezeichnung);

          // Falls nicht vorhanden - AHK anlegen
          if (FEdiCodeColum = 'AHK_EDICODE') then
          begin
            FResolveDataMap.Objects[idx].IdResolved := IntToStr(ConnectionPool.GetSequenceValue(SEQ_ENCOMMON));
            AConnection.ExecuteDirect('INSERT INTO ' + ENAHK +
                    '(AHK_ID, MND_ID, AHK_BEZEICHNUNG, AHK_EDICODE, AHK_ANGELEGT)' + CR_LF +
                    'VALUES ' + CR_LF +
                    '(' + FResolveDataMap.Objects[idx].IdResolved + ', ' + IntToStr(Mandant_ID) + ', '
                    + ImpBezeichnung + ', ' + QuotedStr(FResolveDataMap.Objects[idx].EdiCode) + ', ' + 'SYSDATE' + ')');
            AddResolvedValue(FResolveDataMap.Objects[idx].IdAlt, FResolveDataMap.Objects[idx].IdResolved);
          end;
        end;
      finally
        qry.Close;
      end;
    end;
  finally
    qry.Free;
    EndResolving;
  end;
end;

function TSztpStammdaten.GetDefaultValue(AIdAlt: string; AMndId: Integer): string;
var idx: Integer;
    sztp: TSztpData;
    qry: TsqlQuery;
    eneResolved : String;
begin
  result := FDefaultValue;
  for idx := 0 to FResolveDataList.Count - 1 do
  begin
     sztp := FResolveDataList[idx];
     if sztp.SztpNrAlt = AIdAlt then
     begin
       //Erste SZTP-Nr mit gleicher Energieart wählen
       qry := ConnectionPool.MainConnection.CreateSQLQuery;
       try
          eneResolved := Parent.GetValueWithDefaultFallback('ENE_ID', IntToStr(sztp.EneId));

                          ' AND ENE_ID = ' + eneResolved + CR_LF +
                          ' ORDER BY SZTP_AKTIV DESC, SZTP_EDIMASTER DESC, SZTP_NR DESC';
          qry.Open;
          if not qry.Eof then
            result := qry.FieldByName('SZTP_NR').AsString;
       finally
         qry.Free;
       end;
       exit;
     end;
  end;
end;

function TSztpStammdaten.LoadExportData(ANode: IXMLNode; AExportMndId: Integer): Boolean;
var idx, idx2: Integer;
    node, nodeCol: IXMLNode;
    data: TSztpData;
    value: String;
begin
  result := false;
  if ANode = nil then
    exit;
  if not ANode.HasAttribute(TAG_RESOLVE_TABLENAME) then
    exit;
  if ANode.Attributes[TAG_RESOLVE_TABLENAME] <> TableName then
    exit;
  Clear;
  result := true;
  if not ANode.HasChildNodes then
   exit;

  for idx := 0 to ANode.ChildNodes.Count - 1 do
  begin
    node := ANode.ChildNodes[idx];
    if ExportMndIdAbweichend(node, AExportMndId) then
      continue;
    data := TSztpData.Create;
    for idx2 := 0 to node.ChildNodes.Count - 1 do
    begin
      nodeCol := node.ChildNodes[idx2];
      value := TXmlUtils.ReplaceXml2Backend(nodeCol.Text);
      if nodeCol.LocalName = 'MND_ID' then
        data.MndId := value
      else
      if nodeCol.LocalName = 'SZTP_NR' then
        data.SztpNrAlt := value
      else
      if nodeCol.LocalName = 'ENE_ID' then
        data.EneId := StrToIntDef(value, NULL_INTEGER)
      else
      if nodeCol.LocalName = 'SZTP_TYPCODE' then
        data.TypCode := value
      else
      if nodeCol.LocalName = 'SZTP_GROESSECODE' then
        data.GroesseCode := value
      else
      if nodeCol.LocalName = 'SZTP_UNTERTYPCODE' then
        data.UnterTypCode := value
    end;
    if (data.MndId <> '') and (StrToIntDef(data.MndId, AExportMndId) <> AExportMndId) then
    begin
      data.Free;
      continue;
    end;

    FResolveDataList.add(data);
  end;
end;

procedure TSztpStammdaten.Resolve(AConnection: TecConnection; AMndId: Integer);
var idx: Integer;
    qry: TSQLQuery;
    sztp: TSztpData;
    eneResolved: String;
begin
  if FInResolving then
    exit;

  StartResolving;
  qry := AConnection.CreateSQLQuery;
  try
    for idx := 0 to FResolveDataList.Count - 1 do
    begin
      sztp := FResolveDataList[idx];
      if sztp.SztpNrResolved <> '' then
         Continue;
       eneResolved := '';
       if sztp.EneId <> NULL_INTEGER then
         eneResolved := Parent.GetValueWithDefaultFallback('ENE_ID', IntToStr(sztp.EneId));
       qry.SQL.Text := 'SELECT SZTP_NR ' + CR_LF +
                       ' FROM ' + ENSZTP + CR_LF +
                       ' WHERE MND_ID = ' + IntToStr(AMndId) + CR_LF +
                       '  AND ENE_ID ' + IfThen(sztp.EneId = NULL_INTEGER, 'IS NULL', '=' + eneResolved) + CR_LF +
                       '  AND SZTP_TYPCODE = ' + QuotedStr(sztp.TypCode) + CR_LF +
                       IfThen(sztp.GroesseCode <> '', 'AND SZTP_GROESSECODE = ' + QuotedStr(sztp.GroesseCode) + CR_LF, '') +
                       IfThen(sztp.UnterTypCode <> '', 'AND SZTP_UNTERTYPCODE = ' + QuotedStr(sztp.UnterTypCode) + CR_LF, '') +
                       ' ORDER BY SZTP_EDIMASTER DESC, SZTP_AKTIV DESC';
      qry.Open;
      try
        if not qry.Eof then
        begin
          sztp.SztpNrResolved := qry.FieldByName('SZTP_NR').AsString;
          AddResolvedValue(sztp.SztpNrAlt, sztp.SztpNrResolved);
        end;
      finally
        qry.Close;
      end;
    end;
  finally
    qry.Free;
    EndResolving;
  end;
end;

function TExpImpStammdaten.LoadExportData(ANode: IXMLNode; AExportMndId: Integer): Boolean;
var
  idx, idx2: Integer;
  node, nodeCol: IXMLNode;
  expImpdata: TExpImpData;
  value: String;
begin
  Result := False;
  if ANode = nil then
    Exit;
  if not ANode.HasAttribute(TAG_RESOLVE_TABLENAME) then
    Exit;
  if ANode.Attributes[TAG_RESOLVE_TABLENAME] <> TableName then
    Exit;

  Clear;
  Result := True;

  // prüft ob ResolvableTable Spalten hat
  if not ANode.HasChildNodes then
  begin
    Parent.AddLog('Auflösungsfehler: Resolvingtabelle "' + TableName + '" hat keine Spalten zum Auflösen');
    exit;
  end;

  for idx := 0 to ANode.ChildNodes.Count - 1 do
  begin
    node := ANode.ChildNodes[idx];
    if ExportMndIdAbweichend(node, AExportMndId) then
      continue;

    expImpdata := TExpImpData.Create;
    expImpdata.ExpTableName := TableName;
    for idx2 := 0 to node.ChildNodes.Count - 1 do
    begin
      nodeCol := node.ChildNodes[idx2];
      value := TXmlUtils.ReplaceXml2Backend(nodeCol.Text);
      if (value = NULL_STRING) then
      begin
        Continue;
      end;
      if (nodeCol.LocalName = 'MND_ID') then
      begin
        expImpdata.MndIdOrig := value;
      end
      else if (nodeCol.LocalName = 'ENE_ID') then
      begin
        expImpdata.EneId := StrToIntDef(value, NULL_INTEGER);
      end
      else if ((nodeCol.LocalName = 'KAG_EDICODE') or (nodeCol.LocalName = 'NNV_EDICODE') or
                (nodeCol.LocalName = 'TMS_CODE') or (nodeCol.LocalName = 'PRS_EDICODE') or
                (nodeCol.LocalName = 'NBT_NR') or (nodeCol.LocalName = 'LPR_EDICODE') or
                (nodeCol.LocalName = 'ZRT_EDICODE') or (nodeCol.LocalName = 'RGZ_EIC_NR') or
                (nodeCol.LocalName = 'BIG_BILGEBIET_EIC') or (nodeCol.LocalName = 'MGB_CODE_NR') or
                (nodeCol.LocalName = 'BIK_EIC_NR') or (nodeCol.LocalName = 'NET_NR') or
                (nodeCol.LocalName = 'SPE_EDICODE') or (nodeCol.LocalName = 'ASTP_SCHLUESSEL') or
                (nodeCol.LocalName = 'AAB_CODE')) then
      begin
        expImpdata.EdiCode := value;
        expImpdata.EdiColumnName := nodeCol.LocalName;
      end
      // Alte Stammdaten IDs aus anderen Tabellen merken um diese in den Auflösungselekts
      // zu verwenden. Siehe TExpImpStammdaten.GetStammdatenSelect
      else if ((nodeCol.LocalName = 'NBT_ID') and (expImpdata.ExpTableName <> ENNBT)) then
      begin
        expImpdata.NbtIdAlt := value;
      end
      else if ((nodeCol.LocalName = 'NBT_ID_NEU') and (expImpdata.ExpTableName = ENNWS)) then
      begin
        expImpdata.NbtIdAlt := value;
      end
      else if ((nodeCol.LocalName = 'RGZ_ID') and (expImpdata.ExpTableName <> ENRGZ)) then
      begin
        expImpdata.RgzIdAlt := value;
      end
      else if ((nodeCol.LocalName = 'MGB_ID') and (expImpdata.ExpTableName <> ENMGB)) then
      begin
        expImpdata.MgbIdAlt := value;
      end
      else if ((nodeCol.LocalName = 'IS_GRUNDZ_KME') and (expImpdata.ExpTableName = ENMBT)) then
      begin
        expImpdata.IsGrundKme := value;
      end
      else if ((nodeCol.LocalName = 'IS_GRUNDZ_MBT') and (expImpdata.ExpTableName = ENMBT)) then
      begin
        expImpdata.IsGrundMbt := value;
      end
      else if ((nodeCol.LocalName = 'SPE_KZ_ENTNAHME') and (expImpdata.ExpTableName = ENSPE)) then
      begin
        expImpdata.SpeKzEntnahme := value;
      end
      else if ((nodeCol.LocalName = 'SPE_KZ_MESSUNG') and (expImpdata.ExpTableName = ENSPE)) then
      begin
        expImpdata.SpeKzMessung := value;
      end
      else if ((nodeCol.LocalName = 'NBT_EIGEN') or (nodeCol.LocalName = 'MBT_EIGEN') or
                (nodeCol.LocalName = 'LIF_EIGEN') or (nodeCol.LocalName = 'NET_EIGEN')) then
      begin
        expImpdata.IsEigenValue := value;
        expImpdata.IsEigenColumnName := nodeCol.LocalName;
      end
      else if ((nodeCol.LocalName = 'NBT_VERBUNDILN') or (nodeCol.LocalName = 'LIF_VERBUNDILN')) then
      begin
        expImpdata.VerbundIlnValue := value;
        expImpdata.VerbundIlnColumnName := nodeCol.LocalName;
      end
      else if (((nodeCol.LocalName = 'NBT_ID') and (expImpdata.ExpTableName = ENNBT)) or
                ((nodeCol.LocalName = 'RGZ_ID') and (expImpdata.ExpTableName = ENRGZ)) or
                ((nodeCol.LocalName = 'MGB_ID') and (expImpdata.ExpTableName = ENMGB))) then
      begin
        expImpdata.ExpImpIdAlt := value;
        expImpdata.IdColumnName := nodeCol.LocalName;
        expImpdata.IsLoaded := True;
      end
      else if ((nodeCol.LocalName = 'KAG_ID') or (nodeCol.LocalName = 'NNV_NR') or
                (nodeCol.LocalName = 'TMS_ID') or (nodeCol.LocalName = 'PRS_NR') or
                (nodeCol.LocalName = 'LPR_ID') or (nodeCol.LocalName = 'ZRT_ID') or
                (nodeCol.LocalName = 'BIG_ID') or (nodeCol.LocalName = 'MBT_ID') or
                (nodeCol.LocalName = 'LIF_ID') or (nodeCol.LocalName = 'BIK_ID') or
                (nodeCol.LocalName = 'NET_ID') or (nodeCol.LocalName = 'SPE_NR') or
                (nodeCol.LocalName = 'ASTP_ID') or (nodeCol.LocalName = 'NBT_ID') or
                (nodeCol.LocalName = 'AAB_ID')) then
      begin
        expImpdata.ExpImpIdAlt := value;
        expImpdata.IdColumnName := nodeCol.LocalName;
        expImpdata.IsLoaded := True;
      end
      else if (nodeCol.LocalName = 'AAB_ART') then
      begin
        expImpdata.IntFeld1 := value;
        expImpdata.IntFeld1ColumnName := nodeCol.LocalName;
      end
      else if (nodeCol.LocalName = 'ASTP_STANDARD') then
      begin
        expImpdata.IntFeld1 := QuotedStr(value);
        expImpdata.IntFeld1ColumnName := nodeCol.LocalName;
      end
      else if (nodeCol.LocalName = 'AAB_BEREICH') then
      begin
        expImpdata.IntFeld2 := value;
        expImpdata.IntFeld2ColumnName := nodeCol.LocalName;
      end;
    end;

    if (expImpdata.MndIdOrig <> '') and (StrToIntDef(expImpdata.MndIdOrig, AExportMndId) <> AExportMndId) then
    begin
      expImpdata.Free;
      continue;
    end;

    FResolveExpImpDataList.Add(expImpdata);
  end;
end;

procedure TExpImpStammdaten.Resolve(AConnection: TecConnection; AMndId: Integer);
var idx: Integer;
    netDummy, eneIdResolved, idResolved: String;
    qry: TSQLQuery;
    expImpdata: TExpImpData;
    mitEdicode, mitLifVerbung, isLike, ohneImsbKme: Boolean;

begin
  if FInResolving then
    exit;
  StartResolving;
  qry := AConnection.CreateSQLQuery;
  try

    // Liste befüllt in TExpImpStammdaten.LoadExportData
    // Falls Count=0 ist, muss die XML Datei geprüft werden. Es gab Fall, dass die ResolvableTable keine Spalten hatte.
    for idx := 0 to FResolveExpImpDataList.Count - 1 do
    begin
      expImpdata := FResolveExpImpDataList[idx];

      // Datei ist noch nicht eingelesen und kann nicht importiert werden
      if (not expImpdata.IsLoaded) then
      begin
        continue;
      end;

      if (expImpdata.ExpImpIdResolved <> '') then
      begin
        Continue;
      end;

      if ((expImpdata.MndIdOrig = NULL_STRING) and (Self.MitMndIdSpalte)) then
      begin
        Continue;
      end;

      if (expImpdata.ExpTableName = NULL_STRING) then
      begin
        Continue;
      end;

      if (expImpdata.ExpTableName = ENNBT) then
      begin
        ResolveNbt(qry, expImpData, AConnection);
        Continue;
      end;


      mitEdicode := True;
      mitLifVerbung := True;
      isLike := False;
      GetStammdatenSelect(qry, expImpdata, mitEdicode, mitLifVerbung, isLike);

      qry.Open;
      try
        if (not qry.Eof) then
        begin
          expImpdata.ExpImpIdResolved := qry.FieldByName(expImpdata.IdColumnName).AsString;
          AddResolvedValue(expImpdata.ExpImpIdAlt, expImpdata.ExpImpIdResolved);
          Parent.AddLog('Stammdaten aufgelöst im 1. Schritt: ' + expImpdata.IdColumnName + ' alt ' + expImpdata.ExpImpIdAlt +
                                      ' wurde durch ' + expImpdata.ExpImpIdResolved + ' aufgelöst');
        end
        else
        begin
          mitEdicode := False;
          mitLifVerbung := False;

          if (expImpdata.ExpTableName = ENMGB) then
          begin
            isLike := True;
          end;

          if (expImpdata.ExpTableName = ENSPE) then
          begin
            mitEdicode := True;
            isLike := True;
          end;

          if (expImpdata.ExpTableName = ENASTP) then
          begin
            if (expImpdata.IntFeld1 <> '1') then
            begin
              expImpdata.ExpImpIdResolved := LegeAstpAn(AConnection, expImpdata.ExpImpIdAlt);
              if (expImpdata.ExpImpIdResolved <> NULL_STRING) then
              begin
                AddResolvedValue(expImpdata.ExpImpIdAlt, expImpdata.ExpImpIdResolved);
              end
              else
              begin
                DoResolveError('Fehler an ENASTP: ASTP_SCHLUESSEL "' + expImpdata.EdiCode + '" in ' + expImpdata.ExpTableName
                            + ' im Mandant = ' + expImpdata.MndIdOrig + ' wurde wurde nicht exportiert und kann nicht angelegt werden.');
              end;
            end
            else
            begin
              DoResolveError('Fehler an ENASTP: ASTP_SCHLUESSEL "' + expImpdata.EdiCode + '" in ' + expImpdata.ExpTableName
                            + ' im Mandant = ' + expImpdata.MndIdOrig + ' konnte nicht aufgelöst werden');
            end;
          end;

          GetStammdatenSelect(qry, expImpdata, mitEdicode, mitLifVerbung, isLike);
          qry.Open;

          if (not qry.Eof) then
          begin
            expImpdata.ExpImpIdResolved := qry.FieldByName(expImpdata.IdColumnName).AsString;
            AddResolvedValue(expImpdata.ExpImpIdAlt, expImpdata.ExpImpIdResolved);
            Parent.AddLog('Stammdaten aufgelöst im 2. Schritt: ' + expImpdata.IdColumnName
                                          + ' alt ' + expImpdata.ExpImpIdAlt
                                          + ' wurde durch ' + expImpdata.ExpImpIdResolved + ' aufgelöst');
          end
           else
          begin
            if (expImpdata.ExpTableName = ENBIG) and parent.StandardStammdaten then
            begin
               if (expImpdata.RgzIdAlt <> '') then
               begin
                 idResolved := parent.GetValueWithDefaultFallback('RGZ_ID', expImpdata.RgzIdAlt);
                 if (idResolved <> '') then
                 begin
                   qry.SQL.Text := 'SELECT BIG_ID FROM ' + ENBIG + ' WHERE RGZ_ID = ' + idResolved + ' ORDER BY BIG_ID ASC';
                   qry.Open();
                   if not qry.Eof then
                   begin
                     Parent.AddLog('Ersetzte BIG_ID ' + expImpdata.ExpImpIdAlt + ' durch ' + qry.Fields[0].AsString + ' (Passend zu Rgz)');
                     Parent.AddDefaultValue('BIG_ID', expImpdata.ExpImpIdAlt, qry.Fields[0].AsString);
                     continue;
                   end;
                   qry.SQL.Text := 'SELECT BIG_ID FROM ' + ENBIG + ' WHERE MND_ID = ' + IntToStr(Parent.MandantId) + ' ORDER BY BIG_ID ASC';
                   qry.Open();
                   if not qry.Eof then
                   begin
                     Parent.AddLog('Ersetzte BIG_ID ' + expImpdata.ExpImpIdAlt + ' durch ' + qry.Fields[0].AsString + ' (Erste BIG_ID im Mandant)');
                     Parent.AddDefaultValue('BIG_ID', expImpdata.ExpImpIdAlt, qry.Fields[0].AsString);
                     continue;
                   end;
                 end;
               end;
            end
            else if ((expImpdata.ExpTableName = ENMBT) and parent.StandardStammdaten) then
            begin
              // Fallback für MBT. Falls MBT in beiden Schritten nicht gefunden wurde, wird es versuchen
              // einen MBT ohne mbt_id_imsb zu finden.
              ohneImsbKme := True;
              GetStammdatenSelect(qry, expImpdata, mitEdicode, mitLifVerbung, isLike, ohneImsbKme);
              qry.Open;
              if (not qry.Eof) then
              begin
                expImpdata.ExpImpIdResolved := qry.FieldByName(expImpdata.IdColumnName).AsString;
                AddResolvedValue(expImpdata.ExpImpIdAlt, expImpdata.ExpImpIdResolved);
                Parent.AddLog('MBT aufgelöst im 3. Schritt: MBT_ID alt ' + expImpdata.ExpImpIdAlt +
                              ' wurde durch ' + expImpdata.ExpImpIdResolved + ' OHNE imbt_id_imsb = mbt_id ' +
                              ' in NBT aufgelöst! ');
                continue;
              end
              else if (Parent.FStammdatenMitFallback) then
              begin
                // Fallback für MBT. Es wird dann der ältesten Lastprofil für MND und ENE genommen
                eneIdResolved := Parent.GetValueWithDefaultFallback('ENE_ID', IntToStr(expImpdata.EneId));
                qry.SQL.Clear;
                qry.SQL.Text := 'SELECT MBT_ID FROM ' + ENMBT + ' WHERE ENE_ID = ' + eneIdResolved +
                              '  AND mnd_id = ' + IntToStr(Parent.MandantId) + ' ORDER BY MBT_ID ASC';
                qry.Open();
                if (not qry.Eof) then
                begin
                  Parent.AddLog('Stamdatenresolving: Achtung! Ersetzte MBT_ID ' + expImpdata.ExpImpIdAlt + ' durch ' + qry.Fields[0].AsString + ' (Passend nur zur ENE)');
                  Parent.AddDefaultValue('MBT_ID', expImpdata.ExpImpIdAlt, qry.Fields[0].AsString);
                  continue;
                end;

                if parent.FDummyStammdatenAnlegen  then
                begin
                  idResolved := IntToStr(ConnectionPool.GetSequenceValue(SEQ_ENCOMMON));
                  AConnection.ExecuteDirect('INSERT INTO ' + ENMBT +
                    '(MBT_ID, MND_ID, ENE_ID, MBT_NAME, MBT_STRASSE, MBT_HAUSNUMMER, MBT_PLZ, MBT_ORT)' + CR_LF +
                    'VALUES ' + CR_LF +
                    '(' + idResolved + ', ' + IntToStr(Mandant_ID) + ',' + eneIdResolved + ', ' + QuotedStr('IMP-DUMMY-MSB-' + eneIdResolved) + ', ' + QuotedStr('DUMMY-STR') + ', 1, 99998, ' + QuotedStr('IMPORT') + ')');
                  dtmBasicsStammdaten.qdsENMBT.Refresh;
                  Parent.AddLog('Dummy-MSB zu EneId = ' + eneIdResolved + ' angelegt (MbtId: ' + idResolved + ', Fremd)');
                  Parent.AddDefaultValue('MBT_ID', expImpdata.ExpImpIdAlt, idResolved);
                  continue;
                end;
                Parent.AddLog('Es konnte gar kein Messstellenbetreiber zu EneId = ' + eneIdResolved + ' gefunden werden');
              end;
            end
            else if ((expImpdata.ExpTableName = ENLPR) and parent.StandardStammdaten) then
            begin
              if (Parent.FStandardStammdaten) then
              begin
                // Fallback für LPR. Es wird dann der eltesten Lastprofil für MND und ENE genommen
                idResolved := Parent.GetValueWithDefaultFallback('ENE_ID', IntToStr(expImpdata.EneId));
                qry.SQL.Clear;
                qry.SQL.Text := 'SELECT LPR_ID FROM ' + ENLPR + ' WHERE ENE_ID = ' + idResolved +
                              '  AND mnd_id = ' + IntToStr(Parent.MandantId) + ' ORDER BY LPR_ID ASC';
                qry.Open();
                if (not qry.Eof) then
                begin
                  Parent.AddLog('Stamdatenresolving: Achtung! Ersetzte LPR_ID ' + expImpdata.ExpImpIdAlt + ' durch ' + qry.Fields[0].AsString + ' (Passend nur zu ENE und MND)');
                  Parent.AddDefaultValue('LPR_ID', expImpdata.ExpImpIdAlt, qry.Fields[0].AsString);
                  continue;
                end;
              end;
            end
            else if ((expImpdata.ExpTableName = ENLIF) and parent.StandardStammdaten) then
            begin
              if (Parent.FStandardStammdaten) then
              begin
                // Fallback für LKIF. Es wird dann der eltesten Lastprofil für MND und ENE genommen
                eneIdResolved := Parent.GetValueWithDefaultFallback('ENE_ID', IntToStr(expImpdata.EneId));
                qry.SQL.Clear;
                qry.SQL.Text := 'SELECT LIF_ID FROM ' + ENLIF + ' WHERE ENE_ID = ' + eneIdResolved +
                              '  AND mnd_id = ' + IntToStr(Parent.MandantId) + ' ORDER BY LIF_ID ASC';
                qry.Open();
                if (not qry.Eof) then
                begin
                  Parent.AddLog('Stamdatenresolving: Achtung! Ersetzte LIF_ID ' + expImpdata.ExpImpIdAlt + ' durch ' + qry.Fields[0].AsString + ' (Passend nur zu ENE und MND)');
                  Parent.AddDefaultValue('LIF_ID', expImpdata.ExpImpIdAlt, qry.Fields[0].AsString);
                  continue;
                end;
                if parent.FDummyStammdatenAnlegen  then
                begin
                  idResolved := IntToStr(ConnectionPool.GetSequenceValue(SEQ_ENCOMMON));
                  AConnection.ExecuteDirect('INSERT INTO ' + ENLIF +
                    '(LIF_ID, MND_ID, ENE_ID, LIF_NAME, LIF_EIGEN)' + CR_LF +
                    'VALUES ' + CR_LF +
                    '(' + idResolved + ', ' + IntToStr(Mandant_ID) + ',' + eneIdResolved + ', ' +
                          QuotedStr('IMP-DUMMY-LIF-' + eneIdResolved) + ', ' + QuotedStr('0') + ')');
                  dtmBasicsStammdaten.qdsENLIF.Refresh;
                  Parent.AddLog('Dummy-LIF zu EneId = ' + eneIdResolved + ' angelegt (LifId: ' + idResolved + ', fremd)');
                  Parent.AddDefaultValue('LIF_ID', expImpdata.ExpImpIdAlt, idResolved);
                  continue;
                end;
                Parent.AddLog('Es konnte gar kein Lieferant zu EneId = ' + eneIdResolved + ' gefunden werden');
              end;
            end
            else if ((expImpdata.ExpTableName = ENNET) and parent.StandardStammdaten) then
            begin
              if (Parent.FStandardStammdaten) then
              begin
                eneIdResolved := Parent.GetValueWithDefaultFallback('ENE_ID', IntToStr(expImpdata.EneId));
                qry.SQL.Clear;
                qry.SQL.Text := 'SELECT NET_ID FROM ' + ENNET + ' WHERE ENE_ID = ' + eneIdResolved +
                              '  AND mnd_id = ' + IntToStr(Parent.MandantId) + ' ORDER BY NET_ID ASC';
                qry.Open();
                if (not qry.Eof) then
                begin
                  Parent.AddLog('Stamdatenresolving: Achtung! Ersetzte NET_ID ' + expImpdata.ExpImpIdAlt + ' durch ' + qry.Fields[0].AsString + ' (Passend nur zu ENE und MND)');
                  Parent.AddDefaultValue('NET_ID', expImpdata.ExpImpIdAlt, qry.Fields[0].AsString);
                  continue;
                end;

                if parent.FDummyStammdatenAnlegen  then
                begin
                  idResolved := IntToStr(ConnectionPool.GetSequenceValue(SEQ_ENCOMMON));
                  qry.SQL.Text := 'SELECT MAX(NET_NR) FROM ' + ENNET + ' WHERE NET_NR LIKE ' + QuotedStr('IMP' + eneIdResolved + '%') +
                                '  AND mnd_id = ' + IntToStr(Parent.MandantId);
                  qry.Open();
                  if (not qry.Fields[0].IsNull) then
                    netDummy := qry.Fields[0].AsString + 'x'
                  else
                    netDummy := 'IMP' + eneIdResolved;

                  AConnection.ExecuteDirect('INSERT INTO ' + ENNET +
                    '(NET_ID, MND_ID, ENE_ID, NET_BEZEICHNUNG, NET_EIGEN, NET_NR, NET_GRUNDVERSORGER)' + CR_LF +
                    'VALUES ' + CR_LF +
                    '(' + idResolved + ', ' + IntToStr(Mandant_ID) + ',' + eneIdResolved + ', ' +
                          QuotedStr('IMP-DUMMY-NET-' + eneIdResolved) + ', ' + QuotedStr('0') +  ',' + QuotedStr(netDummy) + ', ''0'')');
                  dtmBasicsStammdaten.qdsENNET.Refresh;
                  Parent.AddLog('Dummy-Netz "' + netDummy + '" zu EneId = ' + eneIdResolved + ' angelegt (NetId: ' + idResolved + ', fremd, keine NNB)');
                  Parent.AddDefaultValue('NET_ID', expImpdata.ExpImpIdAlt, idResolved);
                  continue;
                end;
              end;
            end;
            if (expImpdata.ExpTableName = ENBIK) and parent.StandardStammdaten then
            begin
               if (expImpdata.RgzIdAlt <> '') then
               begin
                 idResolved := parent.GetValueWithDefaultFallback('RGZ_ID', expImpdata.RgzIdAlt);
                 if (idResolved <> '') then
                 begin
                   qry.SQL.Text := 'SELECT BIK_ID FROM ' + ENBIK + ' WHERE RGZ_ID = ' + idResolved + ' ORDER BY BIK_ID ASC';
                   qry.Open();
                   if not qry.Eof then
                   begin
                     Parent.AddLog('Stamdatenresolving: Achtung! Ersetzte BIK_ID ' + expImpdata.ExpImpIdAlt + ' durch ' + qry.Fields[0].AsString + ' (Passend zu Rgz)');
                     Parent.AddDefaultValue('BIK_ID', expImpdata.ExpImpIdAlt, qry.Fields[0].AsString);
                     continue;
                   end;
                   qry.SQL.Text := 'SELECT BIK_ID FROM ' + ENBIK + ' WHERE MND_ID = ' + IntToStr(Parent.MandantId) + ' AND RGZ_ID IS NOT NULL ORDER BY BIK_ID ASC';
                   qry.Open();
                   if not qry.Eof then
                   begin
                     Parent.AddLog('Stamdatenresolving: Achtung!!! Ersetzte BIK_ID ' + expImpdata.ExpImpIdAlt + ' durch ' + qry.Fields[0].AsString + ' (Erste BIK_ID im Mandant)');
                     Parent.AddDefaultValue('BIK_ID', expImpdata.ExpImpIdAlt, qry.Fields[0].AsString);
                     continue;
                   end;
                 end;
               end
                else
               if (expImpdata.MgbIdAlt <> '') then
               begin
                 idResolved := parent.GetValueWithDefaultFallback('MGB_ID', expImpdata.MgbIdAlt);
                 if (idResolved <> '') then
                 begin
                   qry.SQL.Text := 'SELECT BIK_ID FROM ' + ENBIK + ' WHERE MGB_ID = ' + idResolved + ' ORDER BY BIK_ID ASC';
                   qry.Open();
                   if not qry.Eof then
                   begin
                     Parent.AddLog('Stamdatenresolving: Achtung! Ersetzte BIK_ID ' + expImpdata.ExpImpIdAlt + ' durch ' + qry.Fields[0].AsString + ' (Passend zu Rgz)');
                     Parent.AddDefaultValue('BIK_ID', expImpdata.ExpImpIdAlt, qry.Fields[0].AsString);
                     continue;
                   end;
                   qry.SQL.Text := 'SELECT BIK_ID FROM ' + ENBIK + ' WHERE MND_ID = ' + IntToStr(Parent.MandantId) + ' AND MGB_ID IS NOT NULL ORDER BY BIK_ID ASC';
                   qry.Open();
                   if not qry.Eof then
                   begin
                     Parent.AddLog('Stamdatenresolving: Achtung!!! Ersetzte BIK_ID ' + expImpdata.ExpImpIdAlt + ' durch ' + qry.Fields[0].AsString + ' (Erste BIK_ID im Mandant)');
                     Parent.AddDefaultValue('BIK_ID', expImpdata.ExpImpIdAlt, qry.Fields[0].AsString);
                     continue;
                   end;
                 end;
               end
           end;
           if IsIrrelevantInMandanten(TableName) then
           begin
             AddResolvedValue(expImpdata.ExpImpIdAlt, 'NULL');
             Parent.AddLog(expImpdata.IdColumnName + ' in ' + expImpdata.ExpTableName
                             + ' mit der ID = ' + expImpdata.ExpImpIdAlt + ' konnte nicht aufgelöst werden irrelevanter Wert, mit "NULL" ersetzt.');
           end
           else
          begin
            if (parent.FFehlerIgnorieren) then
            begin
              AddResolvedValue(expImpdata.ExpImpIdAlt, 'NULL');
              Parent.AddLog(expImpdata.IdColumnName + ' in ' + expImpdata.ExpTableName
                             + ' mit der ID = ' + expImpdata.ExpImpIdAlt + ' konnte nicht aufgelöst werden und wurde mit "NULL" ersetzt.');
            end
            else
            begin
              DoResolveError(expImpdata.IdColumnName + ' in ' + expImpdata.ExpTableName
                               + ' mit der ID = ' + expImpdata.ExpImpIdAlt + ' konnte nicht aufgelöst werden.');
            end;
          end;
          end;
        end;
      finally
        qry.Close;
      end;
    end;
  finally
    qry.Free;
    EndResolving;
  end;
end;

procedure TExpImpStammdaten.ResolveNbt(AQry: TSQLQuery; AExpImpData: TExpImpData; AConnection: TecConnection);
var
  I, DummyCount, Schritte: Integer;
  IdNeu, NrNeu, eneResolved: String;
  IsResolved: Boolean;
begin
  IsResolved := False;
  eneResolved := Parent.GetValueWithDefaultFallback('ENE_ID', IntToStr(AExpImpData.EneId));

  if (Parent.FStammdatenMitFallback) then
  begin
    Schritte := 3;
  end
  else
  begin
    Schritte := 1;
  end;

  for I := 0 to Schritte do
  begin
    AQry.SQL.Clear;
    AQry.SQL.Add('SELECT nbt_id, nbt_nr FROM ' + ENNBT);
    AQry.SQL.Add('  WHERE mnd_id = ' + IntToStr(Parent.MandantId));
    AQry.SQL.Add('  AND ene_id = ' + eneResolved);

    // im 2. Versuch ohne nbt_nr probieren.
    if (I < 1) then
    begin
      AQry.SQL.Add('  AND nbt_nr = ' + QuotedStr(AExpImpData.EdiCode));
    end;

    if (I < 2) then
    begin
      if (AExpImpData.VerbundIlnValue <> NULL_STRING) then
      begin
        AQry.SQL.Add('  AND nbt_verbundiln = ' + QuotedStr(AExpImpData.VerbundIlnValue));
      end;
    end;

    if (I < 3) then
    begin
      if (AExpImpData.IsEigenValue <> NULL_STRING) then
      begin
        AQry.SQL.Add('  AND nbt_eigen = ' + QuotedStr(AExpImpData.IsEigenValue));
      end;
    end;

    AQry.SQL.Add('ORDER BY nbt_id');

    AQry.Open;
    if (not AQry.Eof) then
    begin
      IdNeu := AQry.FieldByName('nbt_id').AsString;
      NrNeu := AQry.FieldByName('nbt_nr').AsString;

      AExpImpData.ExpImpIdResolved := IdNeu;

      AddResolvedValue(AExpImpData.ExpImpIdAlt, IdNeu);
      AddResolvedValue(AExpImpData.EdiCode, NrNeu);

      Parent.AddLog('Netzbetreiber aufgelöst im ' + IntToStr(I+1) + '. Schritt: nbt_id alt = ' + AExpImpData.ExpImpIdAlt +
                                      ' wurden durch ' + IdNeu + ' und nbt_nr alt ' + AExpImpData.EdiCode + ' durch ' +
                                      NrNeu + ' aufgelöst.');

      IsResolved := True;
      Break;
    end;
  end;

  if (not IsResolved) then
  begin
    Parent.AddLog('Netzbetreiber mit NBT_ID =  ' + AExpImpData.ExpImpIdAlt + ' und NBT_NR = ' +
                  AExpImpData.EdiCode + ' ENE_ID = ' + eneResolved + ' konnte nicht fachlich aufgelöst werden');
    if (parent.FDummyStammdatenAnlegen) then
    begin
      IdNeu := IntToStr(ConnectionPool.GetSequenceValue(SEQ_ENZAEHLER));

      // TODO: alsc - Insert anpassen
      DummyCount := AConnection.ExecuteDirect('INSERT INTO ' + ENNBT +
        '(NBT_NR, NBT_ID, MND_ID, ENE_ID, NBT_BEZEICHNUNG)' + CR_LF +
        'VALUES ' + CR_LF +
        '(' + QuotedStr('IMP-' + eneResolved) + ', ' + IdNeu + ', ' + IntToStr(Mandant_ID) + ',' + eneResolved + ', ' + QuotedStr('IMP-DUMMY-NBT' + eneResolved) + ')');
      dtmBasicsStammdaten.qdsENNBT.Refresh;
      Parent.AddLog('Dummy-NBT zu EneId = ' + eneResolved + ' angelegt (NbtId: ' + IdNeu + ')');

function TExpImpStammdaten.LegeAstpAn(AConnection: TecConnection; AstpIdAlt: String): String;
var
  I, J: Integer;
  Table: TTableObj;
  Column: TColumnObj;
  IdAlt, Bezeichnung, Schluessel, Standard, Typ, Aktiv, Uebername: String;
  DatenGefunden: Boolean;
begin
  DatenGefunden := False;
  Result := NULL_STRING;
  // String sind bereits in XML quodet. QuotedStr(Schluessel) im INSERT ist nicht nötig.
  for I := 0 to FParent.FTables.Childs.Count - 1 do
  begin
    Table := FParent.FTables.Childs.Items[I];
    if (Table.Tablename = ENASTP) then
    begin
      for J := 0 to Table.Columns.Count - 1 do
      begin
        Column := Table.Columns.Items[J];
        if (Column.Columnname = 'ASTP_ID') then
        begin
          IdAlt := Column.Value;
          if (IdAlt = AstpIdAlt) then
            DatenGefunden := True
          else
            Break;
        end
        else if (Column.Columnname = 'ASTP_BEZEICHNUNG') then
          Bezeichnung := 'IMPORT_' + EntferneHochkomma(Column.Value)
        else if (Column.Columnname = 'ASTP_SCHLUESSEL') then
          Schluessel := Column.Value
        else if (Column.Columnname = 'ASTP_STANDARD') then
          Standard := Column.Value
        else if (Column.Columnname = 'ASTP_TYP') then
          Typ := Column.Value
        else if (Column.Columnname = 'ASTP_AKTIV') then
          Aktiv := Column.Value
        else if (Column.Columnname = 'ASTP_UZG_UEBERNAHME') then
          Uebername := Column.Value;
      end;
    end;
    if DatenGefunden then
      Break;
  end;

  if DatenGefunden then
  begin
    Result := IntToStr(ConnectionPool.GetSequenceValue(SEQ_ENCOMMON));
    AConnection.ExecuteDirect('INSERT INTO ' + ENASTP +
                    '(ASTP_ID, MND_ID, ASTP_BEZEICHNUNG, ASTP_SCHLUESSEL, ASTP_STANDARD, ASTP_TYP, ASTP_AKTIV, ASTP_UZG_UEBERNAHME)' + CR_LF +
                    'VALUES ' + CR_LF +
                    '(' + Result + ', ' + IntToStr(Mandant_ID) + ',' + QuotedStr(Bezeichnung) + ', ' +
                      Schluessel + ', ' + Standard + ', ' + Typ + ', ' + Aktiv + ', ' + Uebername + ')');
  end;
end;

procedure TExpImpStammdaten.GetStammdatenSelect(AQry: TSQLQuery; AExpImpData: TExpImpData;
                                                AMitEdicode: Boolean; AMitLifVerbund: Boolean;
                                                AIsLike: Boolean;
                                                AOhneImsbKme: Boolean);
var
  eneResolved,
  idResolved: String;
begin
//      in KeyColName wird Mandant gespeichert. Direkter Zugriff ist nicht möglich.
//      qry.SQL.Add('SELECT ' + TStammdatenTab(expImpdata).KeyColName + ' FROM ' + TStammdatenTab(expImpdata).TableName);
  AQry.SQL.Clear;

  AQry.SQL.Add('SELECT ' + AExpImpData.IdColumnName + ' FROM ' + AExpImpData.ExpTableName);
  AQry.SQL.Add('  WHERE 1 = 1');

  if ((AExpImpData.EdiCode <> NULL_STRING) and AMitEdicode) then
  begin
    AQry.SQL.Add('  AND ' + AExpImpData.EdiColumnName + ' = ' + QuotedStr(AExpImpData.EdiCode));
  end;

  if (AExpImpData.MndIdOrig <> NULL_STRING) then
  begin
    AQry.SQL.Add('  AND mnd_id = ' + IntToStr(Parent.MandantId));
  end;

  if (AExpImpData.EneId <> NULL_INTEGER) then
  begin
    eneResolved := Parent.GetValueWithDefaultFallback('ENE_ID', IntToStr(AExpImpData.EneId));
    AQry.SQL.Add('  AND ene_id = ' + eneResolved);
  end;

  if (AExpImpData.IsEigenValue <> NULL_STRING) then
  begin
    AQry.SQL.Add('  AND ' + AExpImpData.IsEigenColumnName + ' = ' + QuotedStr(AExpImpData.IsEigenValue));
  end;

  if ((AExpImpData.IsGrundKme = '1') and (AExpImpData.ExpTableName = ENMBT) and (not AOhneImsbKme)) then
  begin
    AQry.SQL.Add('  AND EXISTS(SELECT 1 FROM ' + ENNBT + ' WHERE mbt_id_kme = mbt_id)');
  end;

  if ((AExpImpData.IsGrundMbt = '1') and (AExpImpData.ExpTableName = ENMBT) and (not AOhneImsbKme)) then
  begin
    AQry.SQL.Add('  AND EXISTS(SELECT 1 FROM ' + ENNBT + ' WHERE mbt_id_imsb = mbt_id)');
  end;

  if (AExpImpData.ExpTableName = ENLPR) then
  begin
    if (AExpImpData.NbtIdAlt <> NULL_STRING) then
    begin
      idResolved := parent.GetValueWithDefaultFallback('NBT_ID', AExpImpData.NbtIdAlt);
      if idResolved <> '' then
      begin
        AQry.SQL.Add('  AND nbt_id = ' + idResolved);
      end
      else
      begin
        DoResolveError('NBT_ID in ENLPR = ' + AExpImpData.NbtIdAlt + ' konnte nicht aufgelöst werden');
      end;
    end
    else
    begin
      AQry.SQL.Add('  AND nbt_id IS NULL');
    end;
  end;

  if (AExpImpData.ExpTableName = ENNWS) then
  begin
    if (AExpImpData.NbtIdAlt <> NULL_STRING) then
    begin
      idResolved := parent.GetValueWithDefaultFallback('NBT_ID', AExpImpData.NbtIdAlt);
      if idResolved <> '' then
      begin
        AQry.SQL.Add('  AND nbt_id_neu = ' + idResolved);
      end
      else
      begin
        DoResolveError('NBT_ID_NEU in ENNWS = ' + AExpImpData.NbtIdAlt + ' konnte nicht aufgelöst werden');
      end;
    end
    else
    begin
      AQry.SQL.Add('  AND nbt_id_neu IS NULL');
    end;
  end;

  if (AExpImpData.ExpTableName = ENBIG) then
  begin
    if (AExpImpData.RgzIdAlt <> NULL_STRING) then
    begin
      idResolved := parent.GetValueWithDefaultFallback('RGZ_ID', AExpImpData.RgzIdAlt);
      if idResolved <> '' then
      begin
        AQry.SQL.Add('  AND rgz_id = ' + idResolved);
      end
      else
      begin
        DoResolveError('RGZ_ID in ENBIG = ' + AExpImpData.RgzIdAlt + ' konnte nicht aufgelöst werden');
      end;
    end;
  end;

  if (AExpImpData.ExpTableName = ENBIK) then
  begin
    if (AExpImpData.RgzIdAlt <> NULL_STRING) then
    begin
      idResolved := parent.GetValueWithDefaultFallback('RGZ_ID', AExpImpData.RgzIdAlt);
      if idResolved <> '' then
      begin
        AQry.SQL.Add('  AND rgz_id = ' + idResolved);
      end
      else
      begin
        DoResolveError('RGZ_ID in ENBIK = ' + AExpImpData.RgzIdAlt + ' konnte nicht aufgelöst werden');
      end;
    end
    else
    begin
      AQry.SQL.Add('  AND rgz_id IS NULL');
    end;

    if (AExpImpData.MgbIdAlt <> NULL_STRING) then
    begin
      idResolved := parent.GetValueWithDefaultFallback('MGB_ID', AExpImpData.MgbIdAlt);
      if idResolved <> '' then
      begin
        AQry.SQL.Add('  AND mgb_id = ' + idResolved);
      end
      else
      begin
        DoResolveError('MGB_ID in ENBIK = ' + AExpImpData.MgbIdAlt + ' konnte nicht aufgelöst werden');
      end;
    end
    else
    begin
      AQry.SQL.Add('  AND mgb_id IS NULL');
    end;
  end;

  if ((AExpImpData.ExpTableName = ENLIF) and AMitLifVerbund) then
  begin
    AQry.SQL.Add('  AND lif_verbundunternehmen = 1');
  end;

  if ((AExpImpData.ExpTableName = ENMGB) and AIsLike) then
  begin
    if (AnsiContainsText(AExpImpData.EdiCode, 'MH')) then
    begin
      AQry.SQL.Add('  AND UPPER(MGB_CODE_NR) LIKE ''%MH%''');
    end
    else
    begin
      AQry.SQL.Add('  AND UPPER(MGB_CODE_NR) LIKE ''%ML%''');
    end;
  end;

  if (AExpImpData.ExpTableName = ENSPE) then
  begin
    if (not AIsLike) then
    begin
      AQry.SQL.Add('  AND SPE_KZ_ENTNAHME = ' + QuotedStr(AExpImpData.SpeKzEntnahme));
      AQry.SQL.Add('  AND SPE_KZ_MESSUNG = ' + QuotedStr(AExpImpData.SpeKzMessung));
    end;
  end;

  if (AExpImpData.IntFeld1 <> NULL_STRING) then
  begin
    AQry.SQL.Add('  AND ' + AExpImpData.IntFeld1ColumnName + ' = ' + AExpImpData.IntFeld1);
  end;
  if (AExpImpData.IntFeld2 <> NULL_STRING) then
  begin
    AQry.SQL.Add('  AND ' + AExpImpData.IntFeld2ColumnName + ' = ' + AExpImpData.IntFeld2);
  end;

  AQry.SQL.Add('ORDER BY ' + AExpImpData.IdColumnName);
end;

function TMprdStammdaten.LoadExportData(ANode: IXMLNode; AExportMndId: Integer): Boolean;
var idx, idx2: Integer;
    node, nodeCol: IXMLNode;
    data: TData;
    value: String;
begin
  result := false;
  if ANode = nil then
    exit;
  if not ANode.HasAttribute(TAG_RESOLVE_TABLENAME) then
    exit;
  if ANode.Attributes[TAG_RESOLVE_TABLENAME] <> TableName then
    exit;

  Clear;
  result := true;
  if not ANode.HasChildNodes then
    exit;
  for idx := 0 to ANode.ChildNodes.Count - 1 do
  begin
    node := ANode.ChildNodes[idx];
    if ExportMndIdAbweichend(node, AExportMndId) then
      continue;

    data := TData.Create;
    for idx2 := 0 to node.ChildNodes.Count - 1 do
    begin
      nodeCol := node.ChildNodes[idx2];
      value := TXmlUtils.ReplaceXml2Backend(nodeCol.Text);
      if nodeCol.LocalName = 'MPRD_CODE' then
        data.MprdCode := value
      else
      if nodeCol.LocalName = 'MPRD_ID' then
        data.MprdId := StrToIntDef(value, NULL_INTEGER)
      else
      if nodeCol.LocalName = 'MPRD_STANDARD' then
        data.Standard := StrToIntDef(value, NULL_INTEGER)
      else
      if nodeCol.LocalName = 'MPRD_VERWENDUNG' then
        data.MprdVerwendung := StrToIntDef(value, 0)
    end;
    // Keine zusätzliche Einschränkung auf ExportMndid -> Stammdaten auf Konzern-Ebene
    FResolveDataList.add(data);
  end;
end;

procedure TMprdStammdaten.Resolve(AConnection: TecConnection; AMndId: Integer);
var idx: Integer;
    mprdUtils : TMprVerwendungszweckUtils;
    data: TData;
begin
  if FInResolving then
    exit;
  StartResolving;
  mprdUtils := TMprVerwendungszweckUtils.Create(AConnection);
  try
    for idx := 0 to FResolveDataList.Count - 1 do
    begin
      data := FResolveDataList[idx];
      if data.MprdIdResolved <> NULL_INTEGER then
         Continue;
      if data.Standard <> 1 then // Können vorerst nicht aufgelöst werden!
        continue;
      data.MprdIdResolved := mprdUtils.ErmittleStandardMessprodukt(data.MprdCode, data.MprdVerwendung);
      if data.MprdIdResolved <> NULL_INTEGER then
        AddResolvedValue(IntToStr(data.MprdId), IntToStr(data.MprdIdResolved));
    end;
  finally
    mprdUtils.Free;
    EndResolving;
  end;
end;

        abnInfo.HausNr := TXmlUtils.ReplaceXml2Backend(node.Text);
      end;

      node := TXmlUtils.GetAndCheckChildXmlNode(nodeInfo, TAG_HAUSNRVONZUSATZ, false);
      if node <> nil then
      begin
        abnInfo.HausUnterNr := TXmlUtils.ReplaceXml2Backend(node.Text);
        abnInfo.HausUnterNrEx := TXmlUtils.ReplaceXml2Backend(node.Text);
      end;
    end
  end;

end;

procedure TGpAdrInfoExportManager.PrepareExport();

var idx, idx2: Integer;
    addOn: String;
    gpInfos: TObjectList;
    abnInfos: TObjectList;
    gpInfo1, gpInfo2: TGpInfo;
    abnInfo1, abnInfo2: TAbnInfo;
    DPSpace: TDPSpace;
    rolle: TGSPRolle;
    qry: TSQLQuery;
begin
  FGpInfos.clear;
  DPSpace := TDPSpace.Create(fConnection, 'KIS', fViewContext.GetRolId);
  gpInfos := TObjectList.Create(false);
  abnInfos := TObjectList.Create(false);
  qry := FConnection.CreateSQLQuery;
  try
    // Erst mal nur direkt die GP-Haupt-Adresse, keine Beachtung von abweichenden Adressen oder Adressverwendungen!!
    for idx := 0 to FRoIds.Count - 1 do
    begin
      rolle :=  TGSPRolle.Find(DPSpace, FRoIds[idx]);
      if rolle = nil then
        Continue; // Darf eigentlich nicht vorkommen
      gpInfo1 := TGpInfo.Create;
      gpInfo1.Name := rolle.Partner.Person.Name.AsString;
      gpInfo1.Vorname := rolle.Partner.Person.Vorname.AsString;
      gpInfo1.Strasse := rolle.Partner.Adresse.Strasse.AsString;
      gpInfo1.Ort := rolle.Partner.Adresse.Ort.AsString;
      gpInfo1.Plz := rolle.Partner.Adresse.Plz.AsString;
      gpInfo1.HausNr := rolle.Partner.Adresse.HausNrVon.AsString;
      gpInfo1.HausNrVonZusatz := rolle.Partner.Adresse.HausNrVonZusatz.AsString;
      gpInfo1.HausNrBis := rolle.Partner.Adresse.HausNrBis.AsString;
      gpInfo1.HausNrBisZusatz := rolle.Partner.Adresse.HausNrBisZusatz.AsString;
      gpInfo1.PlzVerwendung := rolle.Partner.Adresse.PlzVerwendung.AsString;
      gpInfo1.Postfach := rolle.Partner.Adresse.Postfach.AsString;
      gpInfo1.HausUnterNrVom := rolle.Partner.Adresse.HausUnternrVon.AsString;
      gpInfo1.HausUnterNrBis := rolle.Partner.Adresse.HausUnterNrBis.AsString;
      gpInfo1.PlzPostfach := rolle.Partner.Adresse.PlzPostfach.AsString;
      gpInfo1.OrtPostfach := rolle.Partner.Adresse.OrtPostfach.AsString;
      gpInfo1.PlzGrosskunde := rolle.Partner.Adresse.PlzGrosskunde.AsString;

      // Vorbelegung 1:1
      gpInfo1.PlzVerwendungEx := gpInfo1.PlzVerwendung;
      gpInfo1.HausNrEx := gpInfo1.HausNr;
      gpInfo1.HausNrVonZusatzEx := gpInfo1.HausNrVonZusatz;
      gpInfo1.HausNrBisEx := gpInfo1.HausNrBis;
      gpInfo1.HausNrBisZusatzEx := gpInfo1.HausNrBisZusatz;
      gpInfo1.HausUnterNrVomEx := gpInfo1.HausUnterNrVom;
      gpInfo1.HausUnterNrBisEx := gpInfo1.HausUnterNrBis;

      gpInfo1.RoId := FRoIds[idx];
      gpInfos.Add(gpInfo1);
      FGpInfos.AddObject(FRoIds[idx], gpInfo1);
    end;

    for Idx := 0 to gpInfos.Count - 1 do
    begin
      gpInfo1 := TGpInfo(gpInfos[idx]);
      if Idx > 0 then
        addOn := Vextend(IntToStr(idx), 2, '0')
      else
        addOn := '';

      if (gpInfo1.name <> '') and (gpInfo1.NameEx = '') then
      begin
        gpInfo1.NameEx :=  'ImpName' + addOn;
        for Idx2 := idx+1 to gpInfos.Count - 1 do
        begin
          gpInfo2 := TGpInfo(gpInfos[idx2]);
          if (gpInfo2.name = gpInfo1.name) and (gpInfo2.NameEx = '') then
            gpInfo2.NameEx := gpInfo1.NameEx;
        end;
      end;

      if (gpInfo1.vorname <> '') and (gpInfo1.vornameEx = '') then
      begin
        gpInfo1.vornameEx := 'ImpVorname' + addOn;
        for Idx2 := idx+1 to gpInfos.Count - 1 do
        begin
          gpInfo2 := TGpInfo(gpInfos[idx2]);
          if (gpInfo2.vorname = gpInfo1.vorname) and (gpInfo2.vornameEx = '') then
            gpInfo2.vornameEx := gpInfo1.vornameEx;
        end;
      end;

      if (gpInfo1.strasse <> '') and (gpInfo1.strasseEx = '') then
      begin
        gpInfo1.strasseEx := 'Importweg' + addOn;
        for Idx2 := idx+1 to gpInfos.Count - 1 do
        begin
          gpInfo2 := TGpInfo(gpInfos[idx2]);
          if (gpInfo2.strasse = gpInfo1.strasse) and (gpInfo2.strasseEx = '') and
             (gpInfo2.Plz = gpInfo1.Plz) then
          begin
            gpInfo2.strasseEx := gpInfo1.strasseEx;
          end;
        end;
      end;

      if (gpInfo1.Ort <> '') and (gpInfo1.OrtEx = '') then
      begin
        gpInfo1.OrtEx := 'ImpOrt' + addOn;
        for Idx2 := idx+1 to gpInfos.Count - 1 do
        begin
          gpInfo2 := TGpInfo(gpInfos[idx2]);
          if (gpInfo2.Ort = gpInfo1.Ort) and (gpInfo2.OrtEx = '') then
            gpInfo2.OrtEx := gpInfo1.OrtEx;
        end;
      end;

      if (gpInfo1.Plz <> '') and (gpInfo1.PlzEx = '') then
      begin
        gpInfo1.PlzEx := VExtend(IntToStr(idx), 5, '9');

        for Idx2 := idx+1 to gpInfos.Count - 1 do
        begin
          gpInfo2 := TGpInfo(gpInfos[idx2]);
          if (gpInfo2.Plz = gpInfo1.Plz) and (gpInfo2.PlzEx = '') then
            gpInfo2.PlzEx := gpInfo1.PlzEx;
        end;
      end;

      if (gpInfo1.PlzGrosskunde <> '') and (gpInfo1.PlzGrosskundeEx = '') then
      begin
        gpInfo1.PlzGrosskundeEx := VExtend(IntToStr(idx), 5, '9');

        for Idx2 := idx+1 to gpInfos.Count - 1 do
        begin
          gpInfo2 := TGpInfo(gpInfos[idx2]);
          if (gpInfo2.PlzGrosskunde = gpInfo1.PlzGrosskunde) and (gpInfo2.PlzGrosskundeEx = '') then
            gpInfo2.PlzGrosskundeEx := gpInfo1.PlzGrosskundeEx;
        end;
      end;

      if (gpInfo1.Postfach <> '') and (gpInfo1.PostfachEx = '') then
      begin
        gpInfo1.PostfachEx := VExtend(IntToStr(idx), length(gpInfo1.Postfach), '9');
        for Idx2 := idx+1 to gpInfos.Count - 1 do
        begin
          gpInfo2 := TGpInfo(gpInfos[idx2]);
          if (gpInfo2.Postfach = gpInfo1.Postfach) and (gpInfo2.PostfachEx = '') then
            gpInfo2.postfachEx := gpInfo1.postfachEx;
        end;
      end;

      if (gpInfo1.OrtPostfach <> '') and (gpInfo1.OrtPostfachEx = '') then
      begin
        gpInfo1.OrtPostfachEx := 'ImpOrt' + addOn;
        for Idx2 := idx+1 to gpInfos.Count - 1 do
        begin
          gpInfo2 := TGpInfo(gpInfos[idx2]);
          if (gpInfo2.OrtPostfach = gpInfo1.OrtPostfach) and (gpInfo2.OrtPostfachEx = '') then
            gpInfo2.OrtPostfachEx := gpInfo1.OrtPostfachEx;
        end;
      end;

      if (gpInfo1.PlzPostfach <> '') and (gpInfo1.PlzPostfachEx = '') then
      begin
        gpInfo1.PlzPostfachEx := VExtend(IntToStr(idx), 5, '9');

        for Idx2 := idx+1 to gpInfos.Count - 1 do
        begin
          gpInfo2 := TGpInfo(gpInfos[idx2]);
          if (gpInfo2.PlzPostfach = gpInfo1.PlzPostfach) and (gpInfo2.PlzPostfachEx = '') then
            gpInfo2.PlzPostfachEx := gpInfo1.PlzPostfachEx;
        end;
      end;
    end;

    qry.SQL.Text := 'SELECT * FROM ' + ENABN + ' WHERE ABN_ID = :ABN_ID';
    qry.DBPrepare;

    // Erst mal nur die Haupt-Adresse
    for idx := 0 to FAbnIds.Count - 1 do
    begin
      SetDBParam(qry.ParamByName('ABN_ID'), StrToInt64Def(FAbnIds[idx], NULL_INT64));
      try
        qry.Open();
        if qry.Eof then
          Continue;

        abnInfo1 := TAbnInfo.Create;
        abnInfo1.AbnId := FAbnIds[idx];
        abnInfo1.Strasse := qry.FieldByName('ABN_STRASSE').AsString;
        abnInfo1.HausNr := qry.FieldByName('ABN_HAUSNR').AsString;
        abnInfo1.HausUnterNr := qry.FieldByName('ABN_HAUSUNTERNR').AsString;
        abnInfo1.Plz := qry.FieldByName('ABN_PLZ').AsString;
        abnInfo1.Ort := qry.FieldByName('ABN_ORT').AsString;

        // 1;1
        abnInfo1.HausNrEx := abnInfo1.HausNr;
        abnInfo1.HausUnterNrEx := abnInfo1.HausUnterNr;
        abnInfos.Add(abnInfo1);
        FAbnInfos.Add(abnInfo1.AbnId, abnInfo1);
      finally
        qry.Close;
      end;
    end;

    for Idx := 0 to abnInfos.Count - 1 do
    begin
      abnInfo1 := TAbnInfo(abnInfos[idx]);
      if Idx > 0 then
        addOn := Vextend(IntToStr(idx), 2, '0')
      else
        addOn := '';

      if (abnInfo1.strasse <> '') and (abnInfo1.strasseEx = '') then
      begin
        for Idx2 := 0 to gpInfos.Count - 1 do
        begin
          gpInfo1 := TGpInfo(gpInfos[idx2]);
          if (gpInfo1.strasse = abnInfo1.strasse)  and
             (gpInfo1.Plz = abnInfo1.Plz) then
          begin
            abnInfo1.strasseEx := gpInfo1.strasseEx;
            break;
          end;
        end;
      end;

      if (abnInfo1.strasse <> '') and (abnInfo1.strasseEx = '') then
        abnInfo1.strasseEx := 'Importweg' + addOn;

      if abnInfo1.PlzEx = '' then
      begin
        for idx2 := 0 to gpInfos.Count - 1 do
        begin
          gpInfo1 := TGpInfo(gpInfos[idx2]);
          if (gpInfo1.Plz = abnInfo1.Plz) and
             (gpInfo1.Ort = abnInfo1.Ort) then
          begin
            abnInfo1.PlzEx := gpInfo1.PlzEx;
            abnInfo1.OrtEx := gpInfo1.OrtEx;
            break;
          end;
        end;
      end;

      if abnInfo1.PlzEx = '' then
          abnInfo1.PlzEx := VExtend(IntToStr(idx), 5, '9');

      if abnInfo1.OrtEx = '' then
          abnInfo1.OrtEx := 'ImpOrt' + addOn;

      for Idx2 := idx+1 to abnInfos.Count - 1 do
      begin
        abnInfo2 := TAbnInfo(abnInfos[idx2]);
        if (abnInfo2.Plz = abnInfo2.Plz) and (abnInfo2.PlzEx = '') then
          abnInfo2.PlzEx := abnInfo1.PlzEx;
        if (abnInfo2.Ort = abnInfo2.Ort) and (abnInfo2.OrtEx = '') then
          abnInfo2.OrtEx := abnInfo1.OrtEx;
        if (abnInfo2.Strasse = abnInfo2.Strasse) and (abnInfo2.StrasseEx = '') then
          abnInfo2.StrasseEx := abnInfo1.StrasseEx;
      end;
    end;
  finally
    DPSpace.Free;
    gpInfos.Free;
    abnInfos.Free;
    qry.Free;
  end;
end;

procedure TImportMapHelper.CheckImpTables;
begin
  if not TableExists(QSKIMP) then
  begin
    ConnectionPool.AdminConnection.ExecuteDirect(
        'CREATE TABLE ' + QSKIMP + ' (' + CR_LF +
        '       IMP_ID                   NUMBER(12) NOT NULL,' + CR_LF +
        '       IMP_DATEINAME            VARCHAR2(500) NULL,' + CR_LF +
        '       IMP_EXPORT_IDENTIFIER    VARCHAR2(500) NULL,' + CR_LF +
        '       IMP_ANGELEGT             DATE NOT NULL,' + CR_LF +
        '       CONSTRAINT QS_KUNDENIMPORT_PK' + CR_LF +
        '       PRIMARY KEY (IMP_ID)' + CR_LF +
        '       USING INDEX' + CR_LF +
        '              TABLESPACE CSEN_SPACE_INDEX' + CR_LF +
        ')' + CR_LF +
        '       TABLESPACE CSEN_SPACE');

    ConnectionPool.AdminConnection.ExecuteDirect(
        'CREATE INDEX CREATOR.QSKIMP_IE_EXPID ON CREATOR.QS_KUNDENIMPORT ' + CR_LF +
        '(' + CR_LF +
        '       IMP_EXPORT_IDENTIFIER    ASC,' + CR_LF +
        '       IMP_DATEINAME            ASC' + CR_LF +
        ')' + CR_LF +
        '       TABLESPACE CSEN_SPACE_INDEX');
  end;

  if not TableExists(QSKIMPM) then
  begin
    ConnectionPool.AdminConnection.ExecuteDirect(
        'CREATE TABLE ' + QSKIMPM + ' (' + CR_LF +
        '       IMPM_ID           NUMBER(12) NOT NULL,' + CR_LF +
        '       IMP_ID            NUMBER(12) NOT NULL,' + CR_LF +
        '       IMPM_TYP          NUMBER(3) NOT NULL,' + CR_LF +       // Nachträglich von Number(1) auf Number(3) geändert. Muss ggf auf internen Umgebungen noch manuell angepasst werden
        '       IMPM_OLD_VALUE    VARCHAR2(200) NOT NULL,' + CR_LF +
        '       IMPM_NEW_VALUE    VARCHAR2(200) NOT NULL,' + CR_LF +
        '       CONSTRAINT QSKIMPM_PK' + CR_LF +
        '       PRIMARY KEY (IMPM_ID)' + CR_LF +
        '       USING INDEX' + CR_LF +
        '              TABLESPACE CSEN_SPACE_INDEX,' + CR_LF +
        '       CONSTRAINT QSKIMPM_FK_QSKIMP' + CR_LF +
        '              FOREIGN KEY (IMP_ID)' + CR_LF +
        '       REFERENCES CREATOR.QS_KUNDENIMPORT' + CR_LF +
        ')' + CR_LF +
        '       TABLESPACE CSEN_SPACE');

    ConnectionPool.AdminConnection.ExecuteDirect(
        'CREATE INDEX CREATOR.QSKIMPM_FI_QSKIMP ON CREATOR.QS_KUNDENIMPORT_MAP' + CR_LF +
        '(' + CR_LF +
        '       IMP_ID    ASC' + CR_LF +
        ')' + CR_LF +
        '       TABLESPACE CSEN_SPACE_INDEX');

    ConnectionPool.AdminConnection.ExecuteDirect(
        '    CREATE UNIQUE INDEX CREATOR.QSKIMPM_AK_TYP ON CREATOR.QS_KUNDENIMPORT_MAP' + CR_LF +
        '(' + CR_LF +
        '       IMP_ID            ASC,' + CR_LF +
        '       IMPM_OLD_VALUE    ASC,' + CR_LF +
        '       IMPM_TYP          ASC' + CR_LF +
        ')' + CR_LF +
        '       TABLESPACE CSEN_SPACE_INDEX');
  end;
end;

procedure TImportMapHelper.Initialize(ADateiName, AExportIdentifier: String; AImpId: TLongId; AAllowCreate: Boolean = true);
var qry: TSqlQuery;
    loadData: Boolean;
begin
  FImpId := AImpId;
  FMeteringCodeMap.Clear;
  FMaloCodeMap.Clear;
  FRollenMap.Clear;
  FDateiname := ADateiName;
  qry := FConnection.CreateSQLQuery;
  try
    loadData := not AImpId.IsNull;
    qry.SQL.Text := 'SELECT * FROM ' + QSKIMP + ' WHERE IMP_ID = ' + AImpId.ToStr;
    if loadData then
      qry.Open();
    if loadData and not qry.Eof then
    begin
      FDateiname := qry.FieldByName('IMP_DATEINAME').AsString;
      qry.SQL.Text := 'SELECT * FROM ' + QSKIMPM + CR_LF +
                      ' WHERE IMP_ID = ' + AImpId.ToStr;
      qry.Open;
      while not qry.Eof do
      begin
        if qry.FieldByName('impm_typ').AsInteger = IMP_MAP_METERINGCODE then
          FMeteringCodeMap.AddObject(qry.FieldByName('impm_old_value').AsString, qry.FieldByName('impm_new_value').AsString)
        else
        if qry.FieldByName('impm_typ').AsInteger = IMP_MAP_MALOCODE then
          FMaloCodeMap.AddObject(qry.FieldByName('impm_old_value').AsString, qry.FieldByName('impm_new_value').AsString)
        else
        if qry.FieldByName('impm_typ').AsInteger = IMP_MAP_ROLLENNR then
          FRollenMap.AddObject(qry.FieldByName('impm_old_value').AsString, qry.FieldByName('impm_new_value').AsString)
        else
        if qry.FieldByName('impm_typ').AsInteger = IMP_MAP_GERAETENR then
          FGeraeteNrMap.AddObject(qry.FieldByName('impm_old_value').AsString, qry.FieldByName('impm_new_value').AsString)
        else
        if qry.FieldByName('impm_typ').AsInteger = IMP_MAP_NELOCODE then
          FNeloCodeMap.AddObject(qry.FieldByName('impm_old_value').AsString, qry.FieldByName('impm_new_value').AsString)
        else
        if qry.FieldByName('impm_typ').AsInteger = IMP_MAP_SRECODE then
          FSreCodeMap.AddObject(qry.FieldByName('impm_old_value').AsString, qry.FieldByName('impm_new_value').AsString)
         else
        if qry.FieldByName('impm_typ').AsInteger = IMP_MAP_TRECODE then
          FTreCodeMap.AddObject(qry.FieldByName('impm_old_value').AsString, qry.FieldByName('impm_new_value').AsString)
         else
        if qry.FieldByName('impm_typ').AsInteger = IMP_MAP_EDIREF then
          FEdiRefMap.AddObject(qry.FieldByName('impm_old_value').AsString, qry.FieldByName('impm_new_value').AsString)
        else
        if qry.FieldByName('impm_typ').AsInteger = IMP_MAP_REC_NR then
          FRecNrMap.AddObject(qry.FieldByName('impm_old_value').AsString, qry.FieldByName('impm_new_value').AsString)
         else
        if qry.FieldByName('impm_typ').AsInteger = IMP_MAP_KONFIGID then
          FKonfigIdMap.AddObject(qry.FieldByName('impm_old_value').AsString, qry.FieldByName('impm_new_value').AsString)
         else
        if qry.FieldByName('impm_typ').AsInteger = IMP_MAP_NWSPAKETID then
          FNwsPaketIdMap.AddObject(qry.FieldByName('impm_old_value').AsString, qry.FieldByName('impm_new_value').AsString)
         else
          raise Exception.Create('Ungültiger Wert ' + qry.FieldByName('impm_typ').AsString + ' für  impm_typ');
        qry.Next;
      end;
    end
     else
    begin
      if not AAllowCreate then
        raise Exception.Create('Keinen Import-Datensatz zu ImpId = ' + AImpId.ToStr + ' gefunden');
      qry.SQL.Text := 'INSERT INTO ' + QSKIMP +  CR_LF +
        '(IMP_ID, IMP_DATEINAME, imp_export_identifier, imp_angelegt) ' +  CR_LF +
          'VALUES' + CR_LF +
          '(' + AImpId.ToStr +', ' + QuotedStr(ADateiName) + ', ' + QuotedStr(AExportIdentifier) + ', Sysdate)';
      qry.ExecSQL();
    end;
  finally
    qry.Free;
  end;
end;

procedure TImportMapHelper.SaveMappings;
var qry: TSqlQuery;
    procedure SaveMap(AMap: TelStringStringMap; ATyp: Integer);
    var idx: Integer;
    begin
      for idx := 0 to AMap.Count - 1 do
      begin
        qry.ParamByName('IMPM_OLD_VALUE').AsString := AMap.Items[idx];
        qry.ParamByName('IMPM_NEW_VALUE').AsString := AMap.Values[qry.ParamByName('IMPM_OLD_VALUE').AsString];
        qry.ParamByName('IMPM_TYP').AsInteger := ATyp;
        qry.ExecSQL();
      end;
    end;

begin
  qry := FConnection.CreateSQLQuery;
  try
    qry.SQL.Text := 'INSERT INTO ' + QSKIMPM +  CR_LF +
               '(IMP_ID, IMPM_ID, IMPM_TYP, IMPM_OLD_VALUE, IMPM_NEW_VALUE)' + CR_LF +
               'SELECT ' + FImpId.ToStr + ', ' +  SEQ_ENKUNDE + '.nextVal, :IMPM_TYP, :IMPM_OLD_VALUE, :IMPM_NEW_VALUE'  +
               ' FROM DUAL ' +  CR_LF +
               'WHERE NOT EXISTS (SELECT 1 FROM ' + QSKIMPM + ' b' + CR_LF +
               '                   WHERE IMP_ID = ' + FImpId.ToStr + CR_LF +
               '                    AND IMPM_TYP = :IMPM_TYP ' + CR_LF +
               '                    AND IMPM_OLD_VALUE = :IMPM_OLD_VALUE)';
    qry.DBPrepare;
    SaveMap(FMeteringCodeMap, IMP_MAP_METERINGCODE);
    SaveMap(FMaloCodeMap, IMP_MAP_MALOCODE);
    SaveMap(FRollenMap, IMP_MAP_ROLLENNR);
    SaveMap(FGeraeteNrMap, IMP_MAP_GERAETENR);
    SaveMap(FNeloCodeMap, IMP_MAP_NELOCODE);
    SaveMap(FSreCodeMap, IMP_MAP_SRECODE);
    SaveMap(FTreCodeMap, IMP_MAP_TRECODE);
    SaveMap(FEdiRefMap, IMP_MAP_EDIREF);
    SaveMap(FRecNrMap, IMP_MAP_REC_NR);
    SaveMap(FKonfigIdMap, IMP_MAP_KONFIGID);
    SaveMap(FNwsPaketIdMap, IMP_MAP_NWSPAKETID);
  finally
    qry.Free;
  end;
end;

function TEneStammdaten.GetDefaultValue(AIdAlt: string; AMndId: Integer): string;
var idx: Integer;
    data: TEneDaten;
    eneTypDefault: Integer;
begin
  eneTypDefault := NULL_INTEGER;
  if FResolveDataList.Count = 0 then
  begin
    // Energieart nicht exportiert !!!
    result := AIdAlt;
    exit;
  end;

  for idx := 0 to FResolveDataList.Count - 1 do
  begin
    data := FResolveDataList[idx];
    if (data.EneIdAlt = AIdAlt) and (data.MndId = IntToStr(AMndId)) then
    begin
      if data.EneIdResolved <> '' then
      begin
        result := data.EneIdResolved;
        exit;
      end
      else
      begin
        eneTypDefault := StrToIntDef(data.EneTyp, NULL_INTEGER);
        break;
      end;
    end;
  end;

  if (eneTypDefault <> NULL_INTEGER) then
  begin
    dtmBasicsStammdaten.qdsENENE.First;
    while not dtmBasicsStammdaten.qdsENENE.Eof do
    begin
      if not dtmBasicsStammdaten.qdsENENEENE_ENERGIEARTTYP.IsNull and
         (dtmBasicsStammdaten.qdsENENEENE_ENERGIEARTTYP.AsInteger = eneTypDefault) then
      begin
        result := dtmBasicsStammdaten.qdsENENEENE_ID.AsString;
        exit;
      end;
      dtmBasicsStammdaten.qdsENENE.Next;
    end;
  end;

  if (StrToIntDef(AIdAlt, NULL_INTEGER) >0) and
     (StrToIntDef(AIdAlt, NULL_INTEGER) <=3) then
  begin
    result := AIdAlt;
    exit;
  end;

  dtmBasicsStammdaten.qdsENENE.First;
  while not dtmBasicsStammdaten.qdsENENE.Eof do
  begin
    if dtmBasicsStammdaten.qdsENENEENE_ENERGIEARTTYP.IsNull then
    begin
      result := dtmBasicsStammdaten.qdsENENEENE_ID.AsString;
      exit;
    end;
    dtmBasicsStammdaten.qdsENENE.Next;
  end;
end;

function TEneStammdaten.LoadExportData(ANode: IXMLNode; AExportMndId: Integer): Boolean;
var idx, idx2: Integer;
    node, nodeCol: IXMLNode;
    data: TEneDaten;
    value: String;
begin
  result := false;
  if ANode = nil then
    exit;
  if not ANode.HasAttribute(TAG_RESOLVE_TABLENAME) then
    exit;
  if ANode.Attributes[TAG_RESOLVE_TABLENAME] <> TableName then
    exit;
  Clear;
  result := true;
  if not ANode.HasChildNodes then
   exit;

  for idx := 0 to ANode.ChildNodes.Count - 1 do
  begin
    node := ANode.ChildNodes[idx];
    if ExportMndIdAbweichend(node, AExportMndId) then
      continue;

    data := TEneDaten.Create;
    for idx2 := 0 to node.ChildNodes.Count - 1 do
    begin
      nodeCol := node.ChildNodes[idx2];
      value := TXmlUtils.ReplaceXml2Backend(nodeCol.Text);
      if nodeCol.LocalName = 'MND_ID' then
        data.MndId := value
      else
      if nodeCol.LocalName = 'ENE_ID' then
        data.EneIdAlt := value
      else
      if nodeCol.LocalName = 'ENE_ENERGIEARTTYP' then
        data.EneTyp := value
      else
      if nodeCol.LocalName = 'ENE_TEXT' then
        data.EneText := value
      else
      if nodeCol.LocalName = 'ENE_EINSPEISUNG' then
        data.EneEinspeisung := value
      else
      if nodeCol.LocalName = 'ENE_MUELL' then
        data.EneMuell := value
      else
      if nodeCol.LocalName = 'ENE_EDM_AKTIV_NETZ' then
        data.EneEdmAktivNetz := value
      else
      if nodeCol.LocalName = 'ENE_EDM_AKTIV_VERTRIEB' then
        data.EneEdmAktivVertieb := value
    end;
    if (data.MndId <> '') and (StrToIntDef(data.MndId, AExportMndId) <> AExportMndId) then
    begin
      data.Free;
      continue;
    end;

    FResolveDataList.add(data);
  end;
end;

procedure TEneStammdaten.Resolve(AConnection: TecConnection; AMndId: Integer);

var idx: Integer;
    data: TEneDaten;

begin
  if FInResolving then
    exit;
  StartResolving;
  try
    for idx := 0 to FResolveDataList.Count - 1 do
    begin
      data := FResolveDataList[idx];
      if data.EneIdResolved <> '' then
        Continue;
       if dtmBasicsStammdaten.qdsENENE.Locate('ENE_ID', data.EneIdAlt, []) then
       begin
         if (dtmBasicsStammdaten.qdsENENEENE_ENERGIEARTTYP.AsString = data.EneTyp) and
            (data.EneEinspeisung  = dtmBasicsStammdaten.qdsENENEENE_EINSPEISUNG.AsString) then
         begin
           data.EneIdResolved := dtmBasicsStammdaten.qdsENENEENE_ID.AsString;
           AddResolvedValue(data.EneIdAlt, data.EneIdResolved);
           Continue;
         end;
       end;

      if (data.EneTyp = IntToStr(ENERGIEARTTYP_STROM)) or
         (data.EneTyp = IntToStr(ENERGIEARTTYP_GAS)) then
      begin
        dtmBasicsStammdaten.qdsENENE.Filter := 'ENE_ENERGIEARTTYP = ' + data.EneTyp;
        dtmBasicsStammdaten.qdsENENE.Filtered := true;
        try
          dtmBasicsStammdaten.qdsENENE.First;
          while not dtmBasicsStammdaten.qdsENENE.Eof do
          begin
            if (data.EneEinspeisung  = dtmBasicsStammdaten.qdsENENEENE_EINSPEISUNG.AsString) then
            begin
              data.EneIdResolved := dtmBasicsStammdaten.qdsENENEENE_ID.AsString;
              AddResolvedValue(data.EneIdAlt, data.EneIdResolved);
              break;
            end;
            dtmBasicsStammdaten.qdsENENE.Next;
          end;
        finally
          dtmBasicsStammdaten.qdsENENE.Filtered := false;
        end;
      end
      else
      begin
        // für undefinierte Energiearttypen
        dtmBasicsStammdaten.qdsENENE.Filter := 'ENE_ENERGIEARTTYP IS NULL';
        dtmBasicsStammdaten.qdsENENE.Filtered := true;
        try
          dtmBasicsStammdaten.qdsENENE.First;
          while not dtmBasicsStammdaten.qdsENENE.Eof do
          begin
            if (data.EneEinspeisung  = dtmBasicsStammdaten.qdsENENEENE_EINSPEISUNG.AsString) and
                (data.EneMuell  = dtmBasicsStammdaten.qdsENENEENE_MUELL.AsString) and
                (data.EneEdmAktivNetz  = dtmBasicsStammdaten.qdsENENEENE_EDM_AKTIV_NETZ.AsString) and
                (data.EneEdmAktivVertieb  = dtmBasicsStammdaten.qdsENENEENE_EDM_AKTIV_VERTRIEB.AsString) then
            begin
              data.EneIdResolved := dtmBasicsStammdaten.qdsENENEENE_ID.AsString;
              AddResolvedValue(data.EneIdAlt, data.EneIdResolved);
              break;
            end;
            dtmBasicsStammdaten.qdsENENE.Next;
          end;
        finally
          dtmBasicsStammdaten.qdsENENE.Filtered := false;
        end;
      end;
    end;
  finally
    EndResolving;
  end;
end;

procedure TGerStammdaten.Resolve(AConnection: TecConnection; AMndId: Integer);
var idx: Integer;
    qry: TSQLQuery;
    ger: TGerData;
    std: String;
begin
  qry := AConnection.CreateSQLQuery;
  if FInResolving then
    exit;
  StartResolving;
  try
    for idx := 0 to FResolveDataList.Count - 1 do
    begin
      ger := FResolveDataList[idx];
      if ger.GerIdResolved <> '' then
         Continue;
      if (ger.GerGeraetegrundtyp <> '') then  // Altlast aus ganz alten Kunden-Exporten
      begin
        std := '';
        qry.SQL.Text := 'SELECT GER_ID, GER_BEZEICHNUNG, GER_STANDARD' + CR_LF +
                         ' FROM ' + ENGER + CR_LF +
                         ' WHERE GER_GERAETEGRUNDTYP = ' + QuotedStr(ger.GerGeraetegrundtyp) + CR_LF +
                         ' ORDER BY GER_ID ASC';
        qry.Open;
        try
          // Erste GerId nehmen, die die gleiche Bezeichnung und den gleichen Grundtyp hat
          while not qry.Eof do
          begin
            if (qry.FieldByName('GER_STANDARD').AsString = '1') then
              std := qry.FieldByName('GER_ID').AsString;
            if SameText(ger.GerBezeichnung, qry.FieldByName('GER_BEZEICHNUNG').AsString) then
              ger.GerIdResolved := qry.FieldByName('GER_ID').AsString;
            qry.Next;
          end;

          if (ger.GerIdResolved = '') then
          begin
            // Standard-GerId zum Grundtyp nehmen
            ger.GerIdResolved := std;
          end;

          if (ger.GerIdResolved <> '') then
            AddResolvedValue(ger.GerIdAlt, ger.GerIdResolved);
        finally
          qry.Close;
        end;
      end;

     if (ger.GerIdResolved = '') then
     begin
        // 0..9 sind normalerweise ausgelieferte Standard-Gerätetypen die nicht vom Benutzer geändert werden können
        ger.GerIdResolved := ger.GerIdAlt;
        AddResolvedValue(ger.GerIdAlt, ger.GerIdResolved);
     end;
    end;


  finally
    qry.Free;
    EndResolving;
  end;
end;

                       IfThen(wfs.WfsBezeichnung <> '', ' AND WFS_BEZEICHNUNG = ' + QuotedStr(wfs.WfsBezeichnung) + CR_LF, '') +

          wfs.WfsIdResolved := qry.FieldByName('WFS_ID').AsString;

        end

                           ' FROM ' + ENWFS + CR_LF +

          if not qry.Eof then
          begin
            wfs.WfsIdResolved := qry.FieldByName('WFS_ID').AsString;
            AddResolvedValue(wfs.WfsIdAlt, wfs.WfsIdResolved);
            Parent.AddLog('WFS_ID alt "' + wfs.WfsIdAlt + '" wurde durch "' + qry.FieldByName('WFS_ID').AsString + '" ohne Bezeichnung und Adapter aufgelöst');

function TSmtpStammdaten.LoadExportData(ANode: IXMLNode; AExportMndId: Integer): Boolean;
var idx, idx2: Integer;
    node, nodeCol: IXMLNode;
    data: TSmtpData;
    value: String;
begin
  result := false;
  if ANode = nil then
    exit;
  if not ANode.HasAttribute(TAG_RESOLVE_TABLENAME) then
    exit;
  if ANode.Attributes[TAG_RESOLVE_TABLENAME] <> TableName then
    exit;
  Clear;
  result := true;
  if not ANode.HasChildNodes then
   exit;

  for idx := 0 to ANode.ChildNodes.Count - 1 do
  begin
    node := ANode.ChildNodes[idx];
    data := TSmtpData.Create;
    for idx2 := 0 to node.ChildNodes.Count - 1 do
    begin
      nodeCol := node.ChildNodes[idx2];
      value := TXmlUtils.ReplaceXml2Backend(nodeCol.Text);
      if nodeCol.LocalName = 'SMTP_ID' then
        data.SmtpIdAlt := StrToIntDef(value, NULL_INTEGER)
      else
      if nodeCol.LocalName = 'SMTP_ART' then
        data.Art := StrToIntDef(value, NULL_INTEGER)
      else
      if nodeCol.LocalName = 'SMTP_BEZEICHNUNG' then
        data.Bezeichnung := value
      else
      if nodeCol.LocalName = 'SMTP_NR' then
        data.Nummer := value
      else
      if nodeCol.LocalName = 'SMTPT_OBIS' then
        data.ObisCodes.Add(value);
    end;
    FResolveDataList.add(data);
  end;
end;

procedure TSmtpStammdaten.Resolve(AConnection: TecConnection; AMndId: Integer);
var resolved, idx, idx2: Integer;
    qry: TSQLQuery;
    smtp, smtpDb: TSmtpData;
    smtpList: TelObjectList<TSmtpData>;
begin
  qry := AConnection.CreateSQLQuery;
  if FInResolving then
    exit;
  StartResolving;
  smtpList := TelObjectList<TSmtpData>.Create(true);
  try
    qry.SQL.Text := 'SELECT smtp.smtp_id, smtp_nr, smtp_bezeichnung, smtp_art, smtpt_obis '  + CR_LF +
                    ' FROM ' + ENSMTP + ' smtp JOIN ' + ENSMTPT + ' smtpt on smtpt.smtp_id = smtp.smtp_id ' + CR_LF +
                    ' WHERE smtp.SMTP_ART = :ART' + CR_LF +
                    ' ORDER BY smtp_id DESC, smtpt_obis DESC';
    qry.DBPrepare;
    for idx := 0 to FResolveDataList.Count - 1 do
    begin
      smtp := FResolveDataList[idx];
      if smtp.SmtpIdNeu <> NULL_INTEGER then
         Continue;
      smtpList.Clear;
      qry.ParamByName('ART').AsInteger := smtp.Art;
      qry.Open;
      try
        smtpDb := nil;
        if not qry.Eof then
        begin
          while not qry.Eof do
          begin
            if (smtpDb = nil) or (smtpDb.SmtpIdNeu <> qry.FieldByName('smtp_id').AsInteger) then
            begin
              smtpDb := TSmtpData.Create(qry.FieldByName('smtp_nr').AsString, qry.FieldByName('smtp_bezeichnung').AsString, qry.FieldByName('smtp_art').AsInteger);
              smtpDb.SmtpIdNeu := qry.FieldByName('smtp_id').AsInteger;
              smtpList.Add(smtpDb);
            end;
            smtpDb.ObisCodes.Add(qry.FieldByName('smtpt_obis').AsString);
            qry.Next;
          end;

          for idx2 := smtpList.Count - 1 downto 0 do
          begin
            smtpDb := smtpList[idx2];
            if smtpDb.ObisCodes.Text <> smtp.ObisCodes.Text then
            begin
              smtpList.Remove(smtpDb);
            end;
          end;

          resolved := NULL_INTEGER;
          if smtpList.Count = 0 then
          begin
            // falls SMTP nicht aufgelöst ist, wird später beim Import den ORA Fehler geworfen
            // ORA-02291: integrity constraint (CREATOR.ENSMA_FK_ENSMTP) violated - parent key not found
            DoResolveError('SMTP_ID für SMTP_ART = ' + IntToStr(smtp.Art) + ' und SMTP_OBIS = ' +
                smtp.ObisCodes.Text + ' konnte nicht aufgelöst werden');
            continue;
          end;

          if smtpList.Count > 1 then
            for idx2 := smtpList.Count - 1 downto 0 do
            begin
              smtpDb := smtpList[idx2];
              if smtpDb.Bezeichnung = smtp.Bezeichnung then
              begin
                resolved := smtpDb.SmtpIdNeu;
                break;
              end;
            end;

          if resolved = NULL_INTEGER then
            resolved := smtpList[0].SmtpIdNeu;

          if resolved <> NULL_INTEGER then
           begin
            smtp.SmtpIdNeu := resolved;
            AddResolvedValue(IntToStr(smtp.SmtpIdAlt), IntToStr(resolved));
          end;
        end;
      finally
        qry.Close;
      end;
    end;
  finally
    qry.Free;
    smtpList.free;
    EndResolving;
  end;
end;

procedure TSmtpStammdaten.WriteExportData(AConnection: TecConnection; AFileStream: TStream);
var qry, qryObis: TSQLQuery;
    idx: Integer;
begin
  if not HasUsedValues then
    exit;
  TXmlUtils.writeXmlStr(AFileStream,'  <' +  TAG_RESOLVE_TABLE + ' ' + TAG_RESOLVE_TABLENAME + '="' + Tablename + '">');
  qry := AConnection.CreateSQLQuery;
  qryObis := AConnection.CreateSQLQuery;
  try
    qry.SQL.Text := 'SELECT smtp.smtp_id, smtp_nr, smtp_bezeichnung, smtp_art, smtpt_obis '  + CR_LF +
                    ' FROM ' + ENSMTP + ' smtp JOIN ' + ENSMTPT + ' smtpt on smtpt.smtp_id = smtp.smtp_id ' + CR_LF +
                    ' WHERE smtp.smtp_id = :SMTP_ID' + CR_LF +
                    ' ORDER BY smtpt_obis DESC';
    qry.DBPrepare;
    for Idx := 0 to FUsedValues.Count - 1 do
    begin
      SetDBParam(qry.ParamByName('SMTP_ID'), StrToInt64Def(GetUsedValue(idx), NULL_INT64));
      try
        qry.Open();
        if not qry.Eof then
        begin
           TXmlUtils.writeXmlStr(AFileStream,
             '    <' + TAG_RESOLVE_TABLE_ITEM + '>');
           TXmlUtils.writeXmlStr(AFileStream, '     <SMTP_ID>' + qry.FieldByName('smtp_id').AsString + '</SMTP_ID>');
           TXmlUtils.writeXmlStr(AFileStream, '     <SMTP_NR>' + qry.FieldByName('smtp_nr').AsString + '</SMTP_NR>');
           TXmlUtils.writeXmlStr(AFileStream, '     <SMTP_BEZEICHNUNG>' + qry.FieldByName('smtp_bezeichnung').AsString + '</SMTP_BEZEICHNUNG>');
           TXmlUtils.writeXmlStr(AFileStream, '     <SMTP_ART>' + qry.FieldByName('smtp_art').AsString + '</SMTP_ART>');

           TXmlUtils.writeXmlStr(AFileStream, '     <SMTPT_OBIS>' + qry.FieldByName('smtpt_obis').AsString + '</SMTPT_OBIS>');
           qry.Next;
           while not qry.Eof do
           begin
             TXmlUtils.writeXmlStr(AFileStream, '     <SMTPT_OBIS>' + qry.FieldByName('smtpt_obis').AsString + '</SMTPT_OBIS>');
             qry.Next;
           end;
           TXmlUtils.writeXmlStr(AFileStream,
               '    </' + TAG_RESOLVE_TABLE_ITEM + '>');

        end;
      finally
        qry.Close();
      end;
    end;
  finally
    qry.free;
    qryObis.Free;
  end;
  TXmlUtils.writeXmlStr(AFileStream, '  </' + TAG_RESOLVE_TABLE + '>');
end;

procedure TKkpStammdaten.Resolve(AConnection: TecConnection; AMndId: Integer);
var idx: Integer;
    qry: TSQLQuery;
    kkp: TKkpData;
begin
  qry := AConnection.CreateSQLQuery;
  if FInResolving then
    exit;
  StartResolving;
  try
    for idx := 0 to FResolveDataList.Count - 1 do
    begin
      kkp := FResolveDataList[idx];
      if kkp.KkpIdResolved <> '' then
         Continue;

       qry.SQL.Text := 'SELECT KKP_ID ' + CR_LF +
                       ' FROM ' + ENKKP + CR_LF +
                       ' WHERE 1=1 ' + CR_LF +
                       IfThen(kkp.KkpBezeichnung <> '', 'AND KKP_BEZEICHNUNG = ' + QuotedStr(kkp.KkpBezeichnung) + CR_LF, '') +
                       ' ORDER BY KKP_ID DESC';
      qry.Open;
      try
        if not qry.Eof then
        begin
          kkp.KkpIdResolved := qry.FieldByName('KKP_ID').AsString;
          AddResolvedValue(kkp.KkpIdAlt, kkp.KkpIdResolved);
        end
        else
        begin
          qry.SQL.Clear;
          qry.SQL.Text := 'SELECT KKP_ID ' + CR_LF +
                 ' FROM ' + ENKKP + CR_LF +
                 ' WHERE hie_id = ' + IntToStr(HierarchieManager.GetHieIdByCS2Mandant(EnergyMandantToCS2(AMndId))) + CR_LF +
                 ' ORDER BY KKP_ID DESC';
          qry.Open;
          if not qry.Eof then
          begin
            kkp.KkpIdResolved := qry.FieldByName('KKP_ID').AsString;
            AddResolvedValue(kkp.KkpIdAlt, kkp.KkpIdResolved);
            Parent.AddLog('KKP_ID alt "' + kkp.KkpIdAlt + '" wurde durch "' + qry.FieldByName('KKP_ID').AsString + '" ohne Bezeichnung aufgelöst. MBT_ID wurde nicht beachtet!');
          end
          else
          begin
            if (not Parent.FFehlerIgnorieren) then
            begin
              raise Exception.Create('Kein Kommunikationsprofil für HIE_ID =  "'
                                + IntToStr(HierarchieManager.GetHieIdByCS2Mandant(EnergyMandantToCS2(AMndId)))
                                + '" gefunden. Bitte legen Sie ENKKP an.');
            end;
          end;
        end;
      finally
        qry.Close;
      end;
    end;
  finally
    qry.Free;
    EndResolving;
  end;
end;

procedure TTurnusStammdaten.AddUsedTurNr(AMndId: Integer; AKeyValue: String);
var tur: TTurData;
begin
  if (FQryTurNr = nil) then
  begin
    FQryTurNr := ConnectionPool.MainConnection.CreateSQLQuery;
    FQryTurNr.SQL.Text := 'SELECT TUR_ID FROM ' + ENTUR + CR_LF +
                          'WHERE TUR_NR = :TUR_NR ' + CR_LF +
                          ' AND MND_ID = :MND_ID' + CR_LF +

  FQryTurNr.ParamByName('MND_ID').AsInteger := AMndId;
  FQryTurNr.ParamByName('TUR_NR').AsInteger := StrToIntDef(AKeyValue, NULL_INTEGER);
  try

    if not FQryTurNr.Eof then
      AddUsedTurId(AMndId, FQryTurNr.Fields[0].AsString)
    else
     FParent.AddLog('Keinen Turnus zu MndId=' + IntToStr(AMndId)  + ', Art=' + QuotedStr(IntToStr(FTurArt)) + ' und TUR_NR=' + AKeyValue + ' vorhanden');
  finally
    FQryTurNr.Close;
  end;
end;

  AColumnName := Uppercase(AColumnName);
  AKeyValue := EntferneHochkomma(AKeyValue);

    exit;

  if not FColumnNames.ContainsKey(AColumnName) then

  if FColumnNames.Values[AColumnName] = 'TUR_ID' then
    AddUsedTurId(AMndId, AKeyValue)
  else
    AddUsedTurNr(AMndId, AKeyValue)
end;

function TTurnusStammdaten.GetResolvedTurNr(ANrAlt: String): String;
var tur, turFound: TTurData;
    qry: TSqlQuery;
begin
  turFound := nil;
  for tur in FResolveDataList do
   if tur.TurNrAlt = ANrAlt then
   begin
     turFound := tur;
     break;
   end;


  if turFound = nil then
  begin
    // Für Fälle, in denen die Tur-Stammdaten nicht exportiert wurden
    // Sonderlocke: Falls TUR_NR existiert, dann diese als "Default" verwenden
    result := ReadDBTableValue(FParent.Connection, 'ENTUR', 'TUR_NR', 'TUR_ART = ' + IntToStr(FTurArt) + ' AND MND_ID = ' + IntToStr(Mandant_ID) + ' AND TUR_NR = ' + ANrAlt, '');

    if (result <> '') then
      FParent.AddLog('Verwende vorhandene TurnusNr ' + ANrAlt + ' (Art: ' + IntToStr(FTurArt) + ') als Mapping ohne Überprüfung der Termine!!')
     else
    begin
      // Sonderlocke: Falls TUR_NR nicht existiert, dann die kleinste Nr der Art als "Default" verwenden
      qry := FParent.Connection.CreateSQLQuery;
      try
        qry.Sql.Text := 'SELECT MIN(TUR_NR) as TUR_NR FROM ' + ENTUR + ' WHERE TUR_ART = ' + IntToStr(FTurArt) + ' AND MND_ID = ' + IntToStr(Mandant_ID);
        qry.Open();
        if not qry.Eof
          and not qry.FieldByName('TUR_NR').IsNull then
        begin
          result := qry.FieldByName('TUR_NR').AsString;
          FParent.AddLog('Verwende minimale vorhandene TurnusNr ' + result + ' als Mapping für ' +  ANrAlt + ' (Art: ' + IntToStr(FTurArt) + ') ohne Überprüfung der Termine!!');
        end;
      finally
        qry.free;
      end;
    end;

    if result <> '' then
    begin
      turFound := TTurData.Create;
      turFound.TurNrAlt := ANrAlt;
      turFound.TurNrNeu := Result;
      turFound.IsDefault := true;
      FResolveDataList.Add(turFound);
    end
  end;

  if turFound = nil then
    result := ''
  else
    result := turFound.TurNrNeu;
end;

function TTurnusStammdaten.GetResolvedValue(AColumnName, AIdAlt: String): String;
begin
  AColumnName := Uppercase(AColumnName);
  result := '';
  if not FColumnNames.ContainsKey(AColumnName) then
    raise Exception.Create('Interner Fehler: ' + AColumnName + ' ist nicht in diesen TurnusStammdaten registriert');
  if FColumnNames.Values[AColumnName] = 'TUR_ID' then
    result := GetResolvedTurId(AIdAlt)
  else
    result := GetResolvedTurNr(AIdAlt)
end;

function TTurnusStammdaten.LoadExportData(ANode: IXMLNode; AExportMndId: Integer): Boolean;
var idx, idx2: Integer;
    node, nodeCol: IXMLNode;
    data: TTurData;
    value: String;
begin
  result := false;
  if ANode = nil then
    exit;
  if not ANode.HasAttribute(TAG_RESOLVE_TABLENAME) then
    exit;
  if ANode.Attributes[TAG_RESOLVE_TABLENAME] <> TableName then
    exit;
  Clear;
  result := true;
  if not ANode.HasChildNodes then
   exit;

  for idx := 0 to ANode.ChildNodes.Count - 1 do
  begin
    node := ANode.ChildNodes[idx];
    if ExportMndIdAbweichend(node, AExportMndId) then
      continue;

    data := TTurData.Create;
    for idx2 := 0 to node.ChildNodes.Count - 1 do
    begin
      nodeCol := node.ChildNodes[idx2];
      value := TXmlUtils.ReplaceXml2Backend(nodeCol.Text);
      if nodeCol.LocalName = 'TUR_ID' then
        data.TurIdAlt := value
      else
      if nodeCol.LocalName = 'TUR_NR' then
        data.TurNrAlt := value
      else
      if nodeCol.LocalName = 'TUE_VALUES' then
        data.TueEl.CommaText := value
    end;

    FResolveDataList.add(data);
  end;
end;

procedure TTurnusStammdaten.Resolve(AConnection: TecConnection; AMndId: Integer);

var j, turNr, turId, idx: Integer;
    data: TTurData;
    qryInsert,
    qryTue, qryTur: TSQLQuery;
    found: Boolean;

begin
  if FInResolving then
    exit;
  StartResolving;
  qryTur := AConnection.CreateSQLQuery;
  qryTue := AConnection.CreateSQLQuery;
  qryInsert := AConnection.CreateSQLQuery;
  try
    qryTur.SQL.Text := 'SELECT TUR_NR, TUR_ID ' + CR_LF +
                       ' FROM ' + ENTUR + ' tur ' + CR_LF +
                       'WHERE tur_art = ' + IntToStr(FTurArt) + CR_LF +
                       ' AND mnd_id = ' + IntToStr(AMndId) + CR_LF +
                       ' AND :TUECNT = (SELECT count(*)' + CR_LF +
                       '                  FROM ' + ENTUE + ' tue ' + CR_LF +
                       '                WHERE tue.tur_id = tur.tur_id)' + CR_LF +
                       'ORDER BY 1 ASC';
    qryTur.DBPrepare;

    qryTue.SQL.Text := 'SELECT tue_wert FROM ' + ENTUE + ' WHERE tur_id = :TUR_ID';
    qryTue.DBPrepare;

    for idx := 0 to FResolveDataList.Count - 1 do
    begin
      data := FResolveDataList[idx];
      found := true;
      if data.TurIdNeu <> '' then
        Continue;

      qryTur.ParamByName('TUECNT').AsInteger := data.TueEl.Count;
      qryTur.Open();
      try
        if qryTur.Eof then
          continue;
        qryTue.ParamByName('TUR_ID').AsInteger := qryTur.FieldByName('TUR_ID').AsInteger;

        qryTue.Open;
        if qryTue.Eof then
          Continue;

        while not qryTue.Eof do
        begin
          if data.TueEl.IndexOf(qryTue.FieldByName('tue_wert').AsString) <0 then
          begin
            found := false;
            break;
          end;
          qryTue.Next;
        end;
        if found then
        begin
          data.TurIdNeu := qryTur.FieldByName('TUR_ID').AsString;
          data.TurNrNeu := qryTur.FieldByName('TUR_NR').AsString;
        end
         else
        begin
          try
            qryInsert.SQL.Text := 'SELECT MAX(TUR_ID) + 1 FROM ' + ENTUR;
            qryInsert.Open();
            turId := qryInsert.Fields[0].AsInteger;
            qryInsert.SQL.Text := 'SELECT MAX(TUR_NR) + 1 FROM ' + ENTUR;
            qryInsert.Open();
            turNr := qryInsert.Fields[0].AsInteger;

            qryInsert.SQL.Clear;
            qryInsert.SQL.Add('INSERT INTO ' + ENTUR + ' (tur_id, mnd_id, tur_nr, tur_art, tur_anzahlmonate_abs, tur_standard, tur_vg_rlm) ');
            qryInsert.SQL.Add('VALUES (' + IntToStr(turId) + ', ' + IntToStr(AMndId) + ', ' + IntToStr(turNr) + ', ' + IntToStr(FTurArt) +', NULL, 0, 0)');
            qryInsert.ExecSQL;

            for j := 0 to data.TueEl.Count - 1 do
            begin
              qryInsert.SQL.Clear;
              qryInsert.SQL.Add('INSERT INTO ' + ENTUE + ' (tur_id, tue_wert, mnd_id) ');
              qryInsert.SQL.Add('VALUES (' + IntToStr(turId) + ', ' + QuotedStr(data.TueEl[j]) + ', ' + IntToStr(AMndId) + ')');
              qryInsert.ExecSQL
            end;
            data.TurIdNeu := IntToStr(turId);
            data.TurNrNeu := IntToStr(turNr);
          except
            on e: Exception do
              FParent.AddLog('Fehler beim Anlegen eines neuen Turnus: ' + e.message);
          end;
        end;
      finally
        qryTur.Close;
        qryTue.Close;
      end;
    end;
  finally
    EndResolving;
    qryTur.Free;
    qryTue.Free;
    qryInsert.Free;
  end;
end;

procedure TTurnusStammdaten.WriteExportData(AConnection: TecConnection; AFileStream: TStream);
var qry: TSQLQuery;
    idx: Integer;
    List: TStringList;
begin
  if not HasUsedValues then
    exit;

  TXmlUtils.writeXmlStr(AFileStream,'  <' +  TAG_RESOLVE_TABLE + ' ' + TAG_RESOLVE_TABLENAME + '="' + TableName + '">');
  qry := AConnection.CreateSQLQuery;
  list := TStringList.Create;
  try
    qry.SQL.Text := 'SELECT tur.tur_id, tur.tur_nr, tue_wert ' + CR_LF +
                   ' FROM ' + ENTUR + ' tur join ' + ENTUE + ' tue ON tur.tur_id = tue.tur_id AND tur.mnd_id = tue.mnd_id' + CR_LF +
                   ' WHERE tur.tur_id = :TUR_ID' + CR_LF +
                   ' AND tur.tur_art = ' + IntToStr(FTurArt);
    qry.DBPrepare;
    for Idx := 0 to FUsedValues.Count - 1 do
    begin
      SetDBParam(qry.ParamByName('TUR_ID'), StrToInt64Def(GetUsedValue(idx), NULL_INT64));
      try
        qry.Open();
        if not qry.Eof then
        begin
          TXmlUtils.writeXmlStr(AFileStream, '    <' + TAG_RESOLVE_TABLE_ITEM + '>');
          TXmlUtils.writeXmlStr(AFileStream, '    <' + TAG_EXPORT_MND + '>' + IntToStr(GetUsedValueMndId(idx)) + '</' + TAG_EXPORT_MND + '>');
          TXmlUtils.writeXmlStr(AFileStream, '    <TUR_ID>' + qry.FieldByName('TUR_ID').AsString + '</TUR_ID>');
          TXmlUtils.writeXmlStr(AFileStream, '    <TUR_NR>' + qry.FieldByName('TUR_NR').AsString + '</TUR_NR>');
          list.Clear;
          while not qry.Eof do
          begin
            list.Add(qry.FieldByName('TUE_WERT').AsString);
            qry.Next;
          end;
          TXmlUtils.writeXmlStr(AFileStream, '    <TUE_VALUES>' + list.CommaText + '</TUE_VALUES>');
          TXmlUtils.writeXmlStr(AFileStream, '    </' + TAG_RESOLVE_TABLE_ITEM + '>');
        end;
      finally
        qry.Close();
      end;
    end;
  finally
    list.Free;
    qry.free;
  end;
  TXmlUtils.writeXmlStr(AFileStream, '  </' + TAG_RESOLVE_TABLE + '>');
end;

function TZvfStammdaten.GetDefaultValue(AIdAlt: string; AMndId: Integer): string;
var idx: Integer;
    zvfData: TZvfData;
    qry: TsqlQuery;
    eneResolved : String;
begin
  result := FDefaultValue;
  for idx := 0 to FResolveDataList.Count - 1 do
  begin
     zvfData := FResolveDataList[idx];
     if zvfData.ZvfNrAlt = AIdAlt then
     begin
       //Erste SZTP-Nr mit gleicher Energieart wählen
       qry := ConnectionPool.MainConnection.CreateSQLQuery;
       try
          eneResolved := Parent.GetValueWithDefaultFallback('ENE_ID', IntToStr(zvfData.EneIdAlt));
          qry.SQL.Text := 'SELECT ZVF_NR FROM ' + ENZVF + CR_LF +
                          ' WHERE MND_ID = ' + IntToStr(AMndId) + CR_LF +
                          ' AND ENE_ID = ' + eneResolved + CR_LF +
                          ' AND ZVF_EDICODE = '+ QuotedStr('E02'); // SLP
          qry.Open;
          if not qry.Eof then
            result := qry.FieldByName('ZVF_NR').AsString;
          break;
       finally
         qry.Free;
       end;
     end;
  end;
  AddResolvedValue(AIdAlt, result);
end;

function TZvfStammdaten.LoadExportData(ANode: IXMLNode; AExportMndId: Integer): Boolean;
var idx, idx2: Integer;
    node, nodeCol: IXMLNode;
    data: TZvfData;
    value: String;
begin
  result := false;
  if ANode = nil then
    exit;
  if not ANode.HasAttribute(TAG_RESOLVE_TABLENAME) then
    exit;
  if ANode.Attributes[TAG_RESOLVE_TABLENAME] <> TableName then
    exit;
  Clear;
  result := true;
  if not ANode.HasChildNodes then
   exit;

  for idx := 0 to ANode.ChildNodes.Count - 1 do
  begin
    node := ANode.ChildNodes[idx];
    if ExportMndIdAbweichend(node, AExportMndId) then
      continue;
    data := TZvfData.Create;
    for idx2 := 0 to node.ChildNodes.Count - 1 do
    begin
      nodeCol := node.ChildNodes[idx2];
      value := TXmlUtils.ReplaceXml2Backend(nodeCol.Text);
      if nodeCol.LocalName = 'MND_ID' then
        data.MndId := value
      else
      if nodeCol.LocalName = 'ZVF_NR' then
        data.ZvfNrAlt := value
      else
      if nodeCol.LocalName = 'ENE_ID' then
        data.EneIdAlt := StrToIntDef(value, NULL_INTEGER)
      else
      if nodeCol.LocalName = 'ZVF_EDICODE' then
        data.EdiCode := value
      else
      if nodeCol.LocalName = 'ZVF_BEZEICHNUNG' then
        data.Bezeichnung := value
    end;
    if (data.MndId <> '') and (StrToIntDef(data.MndId, AExportMndId) <> AExportMndId) then
    begin
      data.Free;
      continue;
    end;
    FResolveDataList.add(data);
  end;
end;

procedure TZvfStammdaten.Resolve(AConnection: TecConnection; AMndId: Integer);
var idx: Integer;
    qry: TSQLQuery;
    zvfData: TZvfData;
    eneResolved: String;
    zvfNrErledigt: TelIntegerArrayList;

begin
  if FInResolving then
    exit;

  StartResolving;
  zvfNrErledigt := TelIntegerArrayList.Create;
  qry := AConnection.CreateSQLQuery;
  try
    for idx := 0 to FResolveDataList.Count - 1 do
    begin
      zvfData := FResolveDataList[idx];
      if zvfData.ZvfNrResolved <> '' then
         Continue;

       eneResolved := '';
       if zvfData.EneIdAlt <> NULL_INTEGER then
         eneResolved := Parent.GetValueWithDefaultFallback('ENE_ID', IntToStr(zvfData.EneIdAlt))
       else
       begin
         Parent.AddLog('Resolving ZVF-Nr ' + zvfData.ZvfNrAlt + ': EneId ' + IntToStr(zvfData.EneIdAlt) + ' kann nicht aufgelöst werden --> EneId = 1 wird versende');
         eneResolved := '1';
       end;

       qry.SQL.Text := 'SELECT ZVF_NR ' + CR_LF +
                       ' FROM ' + ENZVF + CR_LF +
                       ' WHERE MND_ID = ' + IntToStr(AMndId) + CR_LF +
                       '  AND ENE_ID = ' + eneResolved + CR_LF +
                       '  AND ZVF_EDICODE = ' + QuotedStr(zvfData.EdiCode);
      qry.Open;
      try
        if not qry.Eof then
        begin
          zvfData.ZvfNrResolved := qry.FieldByName('ZVF_NR').AsString;
          AddResolvedValue(zvfData.ZvfNrAlt, zvfData.ZvfNrResolved);
        end;
      finally
        qry.Close;
      end;
      zvfNrErledigt.Add(StrToInt(zvfData.ZvfNrAlt));
    end;

    qry.SQL.Text := 'SELECT ZVF_NR' + CR_LF +
                     ' FROM ' + ENZVF + CR_LF +
                     ' WHERE MND_ID = ' + IntToStr(AMndId);
    qry.Open;
    while not qry.eof do
    begin
      if (zvfNrErledigt.IndexOf(qry.FieldByName('ZVF_NR').AsInteger) < 0) then
      begin
        AddResolvedValue(qry.FieldByName('ZVF_NR').AsString, qry.FieldByName('ZVF_NR').AsString);
        zvfNrErledigt.Add(qry.FieldByName('ZVF_NR').AsInteger);
      end;

      qry.next;
    end;
  finally
    qry.Free;
    zvfNrErledigt.free;
    EndResolving;
  end;
end;
