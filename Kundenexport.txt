function AnonymisiereDaten(AColumn: TcolumnObj; ATyp:TAnonymisierungsDatenTyp): string;
begin
  result := AColumn.Value;
  case ATyp of
    pdName:                 result := QuotedStr('ImpName');
    pdVorname:              result := QuotedStr('ImpVorname');
    pdStrasse:              result := QuotedStr('Importweg');
    pdHausnr:               result := QuotedStr('3');
    pdHausUnterNr:          result := QuotedStr('a');
    pdOrt:                  result := QuotedStr('ImpOrt');
    pdPlz:                  result := QuotedStr('99990');
    pdSenderName:           result := QuotedStr('ImpSenderName');
    pdSenderOrt:            result := QuotedStr('ImpSenderOrt');
    pdSenderStrasse:        result := QuotedStr('ImportSenderweg');
    pdSenderHausNr:         result := QuotedStr('35');
    pdSenderPlz:            result := QuotedStr('99909');
    pdEmpfaengerName:       result := QuotedStr('ImpEmpfaengerName');
    pdEmpfaengerOrt:        result := QuotedStr('ImpEmpfaengerOrt');
    pdEmpfaengerStrasse:    result := QuotedStr('ImportEmpfaengerweg');
    pdEmpfaengerHausNr:     result := QuotedStr('35');
    pdEmpfaengerPlz:        result := QuotedStr('99909');
    pdAnsprechpartner:      result := QuotedStr('ImpAsp');
    pdAnsprechpartnerEmail: result := QuotedStr('ImpAspEmail');
    pdTelNr:                result := QuotedStr('0123456789');
    pdMail:                 result := QuotedStr('Email@Import.de');
  end;
end;

function TKundenExport.ExportKundenListe(AConnection: TecConnection; AViewContext: TViewContext; AKndIds: TelSimpleLongIdList; AMndId: Integer;
                      AFileName: String; AProdukteExportiert: Boolean; AExportOptionen: TExportOptionen): boolean;
begin
  Result := False;
  FExportOptionen := AExportOptionen;
  FWartenThread := TThreadedProgress.Create('Exportiere Kunden', true, 0);
  FEckdaten := TExportEckdaten.Create(AMndId, true);
  FGpAdrInfoManager := TGpAdrInfoExportManager.Create(AConnection, AViewContext);
  try
    fConnection := aConnection;
    FSdManager := TStammdatenExportImportManager.Create(fConnection, nil); // beim Export werden ATables nicht benötig
    FEckdaten.ExportMndId := AMndId;
    if AProdukteExportiert then
      FEckdaten.ExportMitPrd := 1
    else
      FEckdaten.ExportMitPrd := 0;
    fFileName := aFileName;
    Init(AMndId);

    if AKndIds.Count > 0 then
    begin
      ClearIds;
      FSdManager.Clear;
      FuelleKndIdListe(AKndIds);
      DoExport(FExportOptionen.DoGZip);
    end;
  finally
    FreeAndNil(FWartenThread);
    FreeAndNil(FSdManager);
    FreeAndNil(FEckdaten);
    FreeAndNil(FGpAdrInfoManager);
  end;
end;

procedure TKundenExport.DoExport(ADoGZip: Boolean);
var
  idx: Integer;
  mndId: Integer;
  MndMasterIds: TExportMndMasterIds;
  fStream: TFileStream;
  dbgInfo: String;
begin
  CheckModule;
  fNewLine := False;
  if ADoGZip then
  begin
    fStream := TFileStream.Create(FFileName, fmCreate or fmShareDenyWrite);
    fXmlStream := TJclGZIPCompressionStream.Create(fStream);
  end
   else
  begin
    fStream := nil;
    fXmlStream := TFileStream.Create(FFileName, fmCreate or fmShareDenyWrite);
  end;
  try
    WriteXmlStr('<?xml version="1.0" encoding="windows-1252"?>');
    fNewLine := True;
    WriteXmlStr('<ROOT xmlns="NS" >');
    WriteXmlStr('<' + XML_TAG_TABLES + ' ExportVersion="1.1">');
    for mndId in FMndMasterIds.KeyCollection do
    begin
      MndMasterIds := FMndMasterIds.Values[mndId];
      DoExportDaten(MndMasterIds);
    end;
    WriteXmlStr('</' + XML_TAG_TABLES + '>');

    FEckdaten.WriteExportData(fXmlStream);
    FSdManager.WriteExportData(fXmlStream);
    FGpAdrInfoManager.WriteExportData(fXmlStream);

    if FExportLog.Count > 0 then
    begin
      dbgInfo := '';
      for idx := 0 to FExportLog.Count - 1 do
        dbgInfo := dbgInfo +  FExportLog[idx] + CR_LF;
       TXmlUtils.WriteXmlStr(fXmlStream, '<!-- ExportLog:' + CR_LF + dbgInfo + '-->', true);
    end;

    if FExportOptionen.IsExportXmlDebug then
    begin
      dbgInfo := '';
      for mndId in FMndMasterIds.KeyCollection do
      begin
        if dbgInfo <> '' then
          dbgInfo := dbgInfo + CR_LF + CR_LF;
        dbgInfo := dbgInfo + FMndMasterIds.Values[mndId].GetDebugInfos;
      end;
      TXmlUtils.WriteXmlStr(fXmlStream, '<!-- DebugInfo Initial-SelectedIds:' + CR_LF + dbgInfo + '-->', true);
    end;

    FExportOptionen.WriteToXml(fXmlStream);
    WriteXmlStr('</ROOT>');
  finally
    FreeAndNil(fXmlStream);
    FreeAndNil(fStream);
  end;

procedure TKundenExport.WriteChildTable(table: TTableObj);
var
  i: Integer;
  column: TcolumnObj;
  addOn: String;
  expObj: TExpImpTableObj;
  id: TLongId;
begin
  WriteXmlStr(' <Table Tablename="' + table.Tablename + '" ExportMndId=' + table.ExportMndId + addon + '>');
  addOn := '';
  if FExportOptionen.IsExportXmlDebug then
  begin
    expObj := FExportObjectManager.GetItemByIdentifier(table.Tablename);
    if (expObj <> nil) then
    begin
      column := table.GetColumnByName(expObj.IdFieldName);
      if (column <> nil) then
      begin
        id := TLongId.FromStr(column.Value);
        addon := expObj.IdDebugInfo[id];
        TXmlUtils.WriteXmlStr(fXmlStream, '<!-- DebugInfo:' + CR_LF + addOn + '-->', true);
      end;
    end;
  end;
  for i := 0 to table.Columns.Count - 1 do
  begin
    column := table.Columns[i];
    WriteXmlStr('    <Column CName="' + column.Columnname + '">' + column.Value + '</Column>');
  end;
  WriteXmlStr(' </Table>');
end;

  i: Integer;
  neueIdsSelektiert: Boolean;
  MndIds: TelSimpleLongIdList;

  Qry := fConnection.CreateSQLQuery;

    FExportMndId := AMndMasterIds.MndId;

    InitSelects(FExportMndId);

    FExportObjectManager.ObjENKZB.DoInitialSelectByKndIds(fConnection, AMndMasterIds.MndId, AMndMasterIds.KndIds);
    CheckIsCanceled;

    FExportObjectManager.ObjENKZE.DoInitialSelectByKndIds(fConnection, AMndMasterIds.MndId, AMndMasterIds.KndIds);

    FExportObjectManager.ObjEDINV.DoInitialSelectByZptIds(fConnection, AMndMasterIds.MndId, AMndMasterIds.ZptIds);

    CheckIsCanceled;

    FExportObjectManager.ObjEDQUO.DoInitialSelectByMlkIds(fConnection, AMndMasterIds.MndId, AMndMasterIds.MlkIds);
    CheckIsCanceled;

    FExportObjectManager.ObjEDORD.DoInitialSelectByZptIds(fConnection, AMndMasterIds.MndId, AMndMasterIds.ZptIds);
    CheckIsCanceled;

    FExportObjectManager.ObjEDORD.DoInitialSelectByMlkIds(fConnection, AMndMasterIds.MndId, AMndMasterIds.ZptIds);
    CheckIsCanceled;

    FExportObjectManager.ObjEDUTZP.DoInitialSelectByZptIds(fConnection, AMndMasterIds.MndId, AMndMasterIds.ZptIds);
    CheckIsCanceled;

    FExportObjectManager.ObjEDUTZP.DoInitialSelectByMlkIds(fConnection, AMndMasterIds.MndId, AMndMasterIds.MlkIds);
    CheckIsCanceled;

    FExportObjectManager.ObjENLOKB.DoInitialSelectByAbnIds(fConnection, AMndMasterIds.AbnIds);

    CheckIsCanceled;

    FExportObjectManager.ObjENNWM.DoInitialSelectByMlkIds(fConnection, AMndMasterIds.MndId, AMndMasterIds.MlkIds);
    CheckIsCanceled;

    ExportTabelle(AMndMasterIds.KndIds, fSelectENKND, ENKND, 'knd_id', AMndMasterIds.KndIds);
    CheckIsCanceled;

    ExportTabelle(AMndMasterIds.KndIds, fSelectENMESUeberKndId, ENMES, 'knd_id', AMndMasterIds.KndIds);
    CheckIsCanceled;

    if (TDBTableInfo.TableExists(fENZPTTR)) then
    begin
      ExportTabelle(AMndMasterIds.KndIds, fSelectENZPTTR, fENZPTTR, 'zpt_id', AMndMasterIds.ZptIds);
    end;

    ExportTabelle(AMndMasterIds.KndIds, 'SELECT * FROM ' + ENMLU + ' WHERE ZPT_ID_MELO = :zpt_id', ENMLU, 'zpt_id', AMndMasterIds.ZptIds);

    ExportTabelle(AMndMasterIds.KndIds, fSelectENMES, ENMES, 'zpt_id', AMndMasterIds.ZptIds);

      cntNeu := FZpgIds.Count - cntAlt;
      neueZpgs := cntNeu > 0;
      cnt := cnt + cntNeu;
    end;

       ExportTabelle(AMndMasterIds.KndIds, fSelectENZPG_ZPG, ENZPG, 'zpg_id', FZpgIds);

procedure TKundenExport.FillTable(ATable: TTableObj; AQry: TSQLQuery);
var
  i: Integer;
  id: TLongId;
  column: TcolumnObj;
  msg, value, addon: String;
  expObj: TExpImpTableObj;
begin
  value := NULL_STRING;
  if AQry.Eof then
    exit;

  expObj := FExportObjectManager.GetItemByIdentifier(ATable.Tablename);

  for i := 0 to AQry.FieldList.Count - 1 do
  begin
    if AQry.Fields[i].IsNull then
      Continue;

    column := TcolumnObj.Create(AQry.FieldList.Fields[i].FieldName, AQry.FieldList.Fields[i].DataType, AQry.FieldList.Fields[i].DataSize);
    value := TXmlUtils.ReplaceBackend2Xml(AQry.Fields[i].AsString, column.FieldType);
    column.Value:= value;
    ATable.Columns.Add(column);
    FSdManager.AddUsedValue(FExportMndId, column.Columnname, column.FieldType, value);

    if (ATable.Tablename = ENNWMN
    ) then
    begin
      if (column.Columnname = 'NWMN_EMPFAENGER_TYP') then
      begin
        if (column.Value = IntToStr(NWMN_EMPFAENGER_TYP_LIEFERANT)) then
        begin
          FSdManager.AddUsedValue(FExportMndId, 'LIF_ID', ftLargeint, AQry.FieldByName('NWMN_EMPFAENGER_ID').AsString);
        end
        else if (column.Value = IntToStr(NWMN_EMPFAENGER_TYP_NETZBETREIBER)) then
        begin
          FSdManager.AddUsedValue(FExportMndId, 'NBT_ID', ftLargeint, AQry.FieldByName('NWMN_EMPFAENGER_ID').AsString);
        end
        else if (column.Value = IntToStr(NWMN_EMPFAENGER_TYP_MESSSTELLENBETREIBER)) then
        begin
          FSdManager.AddUsedValue(FExportMndId, 'MBT_ID', ftLargeint, AQry.FieldByName('NWMN_EMPFAENGER_ID').AsString);
        end
        else if (column.Value = IntToStr(NWMN_EMPFAENGER_TYP_UEBERTRAGUNGSNETZBETREIBER)) then
        begin
          //ENMPT wird aktuell weder exportiert noch importiert
          FSdManager.AddUsedValue(FExportMndId, 'MPT_ID', ftLargeint, AQry.FieldByName('NWMN_EMPFAENGER_ID').AsString);
        end;
      end;
    end;

    if (ATable.Tablename = ENNWS) and (column.Columnname = 'NBT_ID_NEU') then
      FSdManager.AddUsedValue(FExportMndId, 'NBT_ID', ftLargeint, value);

    if expObj <> nil then
       expObj.AnonymisiereFeldWert(column);

    if AnsiEndsText('_SENDER_NAME', column.Columnname) then
      value := AnonymisiereDaten(column, pdSenderName)
    else
    if AnsiEndsText('_SENDER_STRASSE', column.Columnname) then
      value := AnonymisiereDaten(column, pdSenderStrasse)
    else
    if AnsiEndsText('_SENDER_ORT', column.Columnname) then
      value := AnonymisiereDaten(column, pdSenderOrt)
    else
    if AnsiEndsText('_SENDER_PLZ', column.Columnname) then
      value := AnonymisiereDaten(column, pdSenderPlz)
    else
    if AnsiEndsText('_SENDER_HAUSNR', column.Columnname) then
      value := AnonymisiereDaten(column, pdSenderHausNr)
    else
    if AnsiEndsText('_EMPF_NAME', column.Columnname) then
      value := AnonymisiereDaten(column, pdEmpfaengerName)
    else
    if AnsiEndsText('_EMPF_STRASSE', column.Columnname) then
      value := AnonymisiereDaten(column, pdEmpfaengerStrasse)
    else
    if AnsiEndsText('_EMPF_ORT', column.Columnname) then
      value := AnonymisiereDaten(column, pdEmpfaengerOrt)
    else
    if AnsiEndsText('_EMPF_PLZ', column.Columnname) then
      value := AnonymisiereDaten(column, pdEmpfaengerPlz)
    else
    if AnsiEndsText('_EMPF_HAUSNR', column.Columnname) then
      value := AnonymisiereDaten(column, pdEmpfaengerHausNr)
    else
    if (AnsiEndsText('_STRASSE', column.Columnname) or
        AnsiEndsText('_SUCHF_STRASSE', column.Columnname) or
        AnsiEndsText('_SUCHFELDSTRASSE', column.Columnname)) then
      value := AnonymisiereDaten(column, pdStrasse)
    else
    if AnsiEndsText('_HAUSNR', column.Columnname) or
       AnsiEndsText('SUCHF_HAUSNR', column.Columnname) then
      value := AnonymisiereDaten(column, pdHausnr)
    else
    if AnsiEndsText('_PLZ', column.Columnname) or
       AnsiEndsText('_SUCHF_PLZ', column.Columnname) then
      value := AnonymisiereDaten(column, pdPlz)
    else
    if (AnsiEndsText('_ORT', column.Columnname) or
        AnsiEndsText('_SUCHF_ORT', column.Columnname) or
        AnsiEndsText('_SUCHFELDORT', column.Columnname)) then
      value := AnonymisiereDaten(column, pdOrt)
    else
    if AnsiEndsText('_HAUSUNTERNR', column.Columnname) then
      value := AnonymisiereDaten(column, pdHausUnterNr)
    else
    if (AnsiEndsText('_LS_NAME', column.Columnname) or
        AnsiEndsText('_SUCHF_NAME', column.Columnname) or
        AnsiEndsText('_KND_NAME', column.Columnname) or
        AnsiEndsText('ANS_NAME', column.Columnname)) then
      value := AnonymisiereDaten(column, pdName)
    else
    if (AnsiEndsText('_LS_VORNAME', column.Columnname) or
        AnsiEndsText('_KND_VORNAME', column.Columnname) or
        AnsiEndsText('_SUCHF_VORNAME', column.Columnname) or
        AnsiEndsText('ANS_VORNAME', column.Columnname)) then
      value := AnonymisiereDaten(column, pdVorname)
    else
    if AnsiEndsText('_ANSPRECHPARTNER', column.Columnname) then
      value := AnonymisiereDaten(column, pdAnsprechpartner)
    else
    if AnsiEndsText('ASP_EMAIL', column.Columnname) then
      value := AnonymisiereDaten(column, pdAnsprechpartnerEmail)
    else
    if AnsiEndsText('_EMAIL', column.Columnname) then
      value := AnonymisiereDaten(column, pdMail)
    else
    if AnsiEndsText('_TELEFON', column.Columnname) then
      value := AnonymisiereDaten(column, pdTelNr)
    else
    if ('EPR_BEZEICHNUNG' =  column.Columnname) then
      value := QuotedStr('EPR_BEZEICHNUNG ' + IntToStr( ConnectionPool.GetSequenceValue(SEQ_ENCOMMON)));

    // NBT_NR wird für die Auflösung verwendet, wenn das als QuotedStr gespeichert wird,
    // kommt in SQL = AND NBT_NR = '''200001'''
    if (column.Columnname = 'NBT_NR') then
    begin
      value := EntferneHochkomma(value);
    end;
    column.Value:= value;

    if (ATable.Tablename = ENKND) then
    begin
      if (column.Columnname = 'KND_KUNDENINFO') then
        column.Value:= 'NULL'
      else if (column.Columnname = 'KND_STEUERNUMMER') then
        column.Value:= 'NULL'
    end
    else if (ATable.Tablename = ENKZB) then
    begin
      if (column.Columnname = 'KZB_KONTAKTMIT') then
        column.Value:= QuotedStr('IMPORT')
    end
    else if (ATable.Tablename = ENKZE) then
    begin
      if (column.Columnname = 'KZE_KONTAKTMIT') then
        column.Value:= QuotedStr('IMPORT')
    end;

    if (ATable.Tablename = ENKNV) and (column.Columnname = 'NBT_NR') then
      FSdManager.AddUsedNbtNr(FExportMndId, value, FieldAsInteger(AQry.FieldByName('KNV_ENERGIEART')));

    if (column.Columnname = 'RO_ID') then
      FGpAdrInfoManager.RegisterExportRoId(column.Value);
    if (column.Columnname = 'ABN_ID') then
      FGpAdrInfoManager.RegisterExportAbnId(column.Value);

    if (column.Columnname = 'MND_ID') And FexportOptionen.IsExportXmlDebug then
    begin
      if expObj <> nil then
      begin
        id := AQry.FieldByName(expObj.IdFieldName).AsLongId;
        addon := CR_LF + expObj.IdFieldName + ': ' + id.ToStr + CR_LF + expObj.IdDebugInfo[id];
      end
       else
      addon := '';

      if (ATable.ExportMndId <> AQry.FieldList.Fields[i].AsString) then
      begin
        msg := '1: Abweichender Mandant (1): ' + ATable.Tablename + ' MndIdIst: ' + AQry.FieldList.Fields[i].AsString + ' <-> MndIdSoll ' + ATable.ExportMndId + addon;
        if DebugHook > 0 then
          ShowErrorDialog(msg);
        AddToExportLog(msg);
      end;

      if (FExportMndId <> AQry.FieldList.Fields[i].AsInteger) then
      begin
        msg := '2: Abweichender Mandan (2): ' + ATable.Tablename + ' MndIdIst: ' + AQry.FieldList.Fields[i].AsString + ' <-> MndIdSoll ' + IntToStr(FExportMndId) + addon;
        if DebugHook > 0 then
          ShowErrorDialog(msg);
        AddToExportLog(msg);
      end;
    end;

   if (AQry.FieldList.Fields[i].FieldName = 'ZSTEP_LISTE') then
     LoadZstepListe(AQry, ATable);
  end;
end;

function TKundenExport.AddIdToList(AListe: TelSimpleLongIdList; AId: TLongId): Boolean;
begin
  Result := True;
  if AId.IsNull then
  begin
    Result := False;
    exit;
  end;

  if (not AListe.Contains(AId)) then
  begin
    AListe.Add(AId);
    Result := False;
  end;
end;

begin
  Result := True;
  if (AListe.IndexOf(AValue) = -1) then

    AListe.Add(AValue);
    Result := False;
  end;
end;

function TKundenExport.FillChildrenWithPreparedQuery(Qry: TSQLQuery; ATableName: String;
   aIdValueList: TDictionary<String, TLongId>;
   AMndKndIdListe: TelSimpleLongIdList): Integer;
var
  Child: TTableObj;
  i: Integer;
  value: TLongId;
begin
  CheckIsCanceled;
  result := 0;
  Qry.Close;
  if (Qry.Params.Count > 0) then
  begin
    for i := 0 to Qry.Params.Count - 1 do
    begin
      if (aIdValueList.TryGetValue(Qry.Params[i].Name, value)) then
      begin
        setDBParam(Qry.Params[i], value);
      end;
    end;
  end;

  Qry.Open;
  while not Qry.Eof do
  begin
    CheckIsCanceled;
    if (CheckObSchonExportiert(ATableName, Qry, AMndKndIdListe)) then
    begin
      Qry.Next;
      continue;
    end;
    result := result + 1;
    Child := TTableObj.Create;
    child.ExportMndId := IntToStr(FExportMndId);
    child.Tablename := ATableName;
    try
      FillTable(Child, Qry);
      WriteChildTable(Child);
    finally
      Child.Free;
    end;
    Qry.Next;
  end;
end;

    przQuery.SQL.Add('prz.prz_id');
    przQuery.SQL.Add(', vfm.knd_id');
    przQuery.SQL.Add(', rpr.prz_idstorno');
    przQuery.SQL.Add(', prz.prz_absstornoprz_id');
    przQuery.SQL.Add('FROM ' + ENPRZ + ' prz');
    przQuery.SQL.Add('JOIN ' + ENVFM + ' vfm ON vfm.prz_id = prz.prz_id');
    przQuery.SQL.Add('LEFT JOIN ' + ENRPR + ' rpr ON rpr.prz_id = prz.prz_id');
    przQuery.SQL.Add('  AND rpr.knd_id = vfm.knd_id');
    przQuery.SQL.Add('WHERE');
    przQuery.SQL.Add('prz.prz_id <> 0');
    przQuery.SQL.Add('AND NVL(prz.prz_proforma, 0) = 0');
    przQuery.SQL.Add('AND vfm_belegdatum >= ' + dtmLib32SystemZeit.getDatabaseDateString(FExportOptionen.StichtagVon));
    if (FExportOptionen.StichtagBis <> NULL_DATE) then
      przQuery.SQL.Add('AND vfm_belegdatum <= ' + dtmLib32SystemZeit.getDatabaseDateString(FExportOptionen.StichtagBis));
    przQuery.SQL.Add('AND vfm.knd_id = :knd_id');

    lieferscheinPrzQry.SQL.Add('SELECT DISTINCT');

    lieferscheinPrzQry.SQL.Add(', prz.prz_absstornoprz_id');

    begin
      przIdList.Clear;
      if (not ANurLieferscheinProzesse) then

        SetStatus('Exportiere Rechnungsdaten (MndId: ' + IntToStr(FExportMndId) + ' / KndId: ' + AMndKndIdListe.Items[i].ToStr + ')');
        przQuery.Open;
        while (not przQuery.Eof) do
        begin
          przId := przQuery.FieldByName('prz_id').AsLongId;
          przIdStorno := przQuery.FieldByName('prz_idstorno').AsLongId;
          przIdAbschlagStorno := przQuery.FieldByName('prz_absstornoprz_id').AsLongId;

            przIdList.Add(przId);

            if (not przIdStornoList.Contains(przIdStorno)) then

        end;
        przQuery.Close;
      end;

function TKundenExport.ExportTabelle(AMndKndIdListe: TelSimpleLongIdList; select: String; tablename: String; idFeldname1: String;

var
  i: Integer;
  keyValueList: TDictionary<String, TLongId>;
begin
  CheckIsCanceled;
  SetAddonStatus(tablename);
  result := 0;
  keyValueList := TDictionary<String, TLongId>.Create;
  try
    qryUltimate.SQL.Clear;
    qryUltimate.SQL.Add(select);
    qryUltimate.DBPrepare;
    for i := 0 to idListe1.Count - 1 do
    begin
      CheckIsCanceled;
      keyValueList.Clear;

      end;

         FillChildrenWithPreparedQuery(qryUltimate, tablename, keyValueList, AMndKndIdListe);
    end;
  finally
    keyValueList.Free;
  end;
end;

procedure TKundenExport.ExportTabelle(AExportObj: TExpImpTableObj);
var
  Child: TTableObj;
  qry: TSQLQuery;
  sql, baseSql: String;
  ids: TelSimpleLongIdList;
begin
  if AExportObj.IdsSelektiert.Count = 0 then
    exit;
  if not AExportObj.CanDoExport then
    exit;

  SetAddonStatus(AExportObj.TableName + ' (' + IntToStr(AExportObj.IdsSelektiert.Count) + ' Dts)');
  qry := fConnection.CreateSQLQuery;
  ids := TelSimpleLongIdList.Create(AExportObj.IdsSelektiert.Count);
  try
    ids.AddRange(AExportObj.IdsSelektiert);
    baseSql := 'SELECT * ' + CR_LF +
           ' FROM ' + AExportObj.TableName + CR_LF +
           ' WHERE ' + AExportObj.DefaultFieldName + ' ' + BASISIDLISTTAG;
    sql :=  GetSqlMitIds(baseSql, ids);
    while sql <> '' do
    begin
      qry.SQL.Text := sql;
      qry.Open();
      while not qry.Eof do
      begin
        Child := TTableObj.Create;
        child.ExportMndId := IntToStr(FExportMndId);
        child.Tablename := AExportObj.TableName;
        try
          FillTable(Child, Qry);
          WriteChildTable(Child);
        finally
          Child.Free;
        end;
        qry.Next;
      end;
      sql :=  GetSqlMitIds(baseSql, ids);
    end;
  finally
    qry.Free;
    ids.Free;
  end;
end;

            if AMasterIds.AddAbnId(qryLokByMlkId.Fields[1].AsLongId, 'qryLokByMlkId.Fields[1], MlkId = ' + AMasterIds.MlkIds[i].ToStr) then
               result := true;

            // ZPT_ID_MELO
            if not qryLokByMlkId.Fields[2].IsNull and
               AMasterIds.AddZptId(qryLokByMlkId.Fields[2].AsLongId, 'qryLokByMlkId.Fields[2], MlkId = ' + AMasterIds.MlkIds[i].ToStr) then
               result := true;

            // MLK_ID
            if not qryLokByMlkId.Fields[3].IsNull and
               AMasterIds.AddMlkId(qryLokByMlkId.Fields[3].AsLongId, 'qryLokByMlkId.Fields[3], MlkId = ' + AMasterIds.MlkIds[i].ToStr) then
               result := true;

            // TRE_ID
            if not qryLokByMlkId.Fields[4].IsNull and
               AMasterIds.AddTreId(qryLokByMlkId.Fields[4].AsLongId, 'qryLokByMlkId.Fields[4], MlkId = ' + AMasterIds.MlkIds[i].ToStr) then
               result := true;

             qryLokByMlkId.Next;
          end;
        finally
          qryLokByMlkId.Close;
        end;
      end;
    end;

    // Lokationsbündel nachselektieren
    if AMasterIds.HasNewLokbIdsSinceLastReset then
    begin
      for i := AMasterIds.LastLokbIdsCount  to AMasterIds.LokbIds.Count - 1 do
      begin
        qryLok.ParamByName('LOKB_ID').AsLongId := AMasterIds.LokbIds[i];
        try
          // ZPT_ID_MELO, MLK_ID, NLK_ID, TRE_ID
          qryLok.Open();
          while not qryLok.Eof do
          begin
            // ZPT_ID_MELO
            if not qryLok.Fields[0].IsNull and
               AMasterIds.AddZptId(qryLok.Fields[0].AsLongId, 'qryLok.Fields[0], LokbId = ' + AMasterIds.LokbIds[i].ToStr) then
               result := true;

            // MLK_ID
            if not qryLok.Fields[1].IsNull and
               AMasterIds.AddMlkId(qryLok.Fields[1].AsLongId, 'qryLok.Fields[1], LokbId = ' + AMasterIds.LokbIds[i].ToStr) then
               result := true;

            // TRE_ID
            if not qryLok.Fields[2].IsNull and
               AMasterIds.AddTreId(qryLok.Fields[2].AsLongId, 'qryLok.Fields[2], LokbId = ' + AMasterIds.LokbIds[i].ToStr) then
               result := true;

             qryLok.Next;
          end;
        finally
          qryLok.Close;
        end;
      end;
    end;

    if AMasterIds.HasNewZptIdsSinceLastReset then
    begin
      // restliche AbnIds der ZptIds selektieren!
      for i := AMasterIds.LastZptIdsCount to AMasterIds.ZptIds.Count - 1 do
      begin
        qryAbnIds.ParamByName('ZPT_ID').AsLongId := AMasterIds.ZptIds[i];
        try
          qryAbnIds.Open();
          if not qryAbnIds.Eof then
            AMasterIds.AddAbnId(qryAbnIds.Fields[0].AsLongId, 'qryAbnIds.Fields[0], ZptId = ' + AMasterIds.ZptIds[i].ToStr)
        finally
          qryAbnIds.Close;
        end;
      end;
    end;
  end;

procedure TKundenExport.AddEexIdsByFkLinkId(AFkId: TLongId; AFkTyp: Integer);
var expEpzObj, expEexObj, expLinkObj: TExpImpTableObj;
    key: String;
begin
  if AFkId.IsNull then
    exit;

  if not FExportOptionen.DoEexExport then
    exit;

  key := IntToStr(AFkTyp) + '#' + AFkId.ToStr;

  if (FEexSelectErledigt <> nil) then
  begin
    if (FEexSelectErledigt.IndexOf(key) >= 0) then
      exit;
  end
   else
  begin
    FEexSelectErledigt := TStringList.Create;
    FEexSelectErledigt.Duplicates := dupIgnore;
  end;
  FEexSelectErledigt.Add(key);

  expEpzObj := FExportObjectManager.GetItemByIdentifier(CREPZ);
  expEexObj := FExportObjectManager.GetItemByIdentifier(CREEX);

  expLinkObj := FExportObjectManager.GetItemByIdentifier(CREPZL);
  if FQrySelectEpzl = nil then
  begin
    FQrySelectEpzl := fConnection.CreateSQLQuery;
    FQrySelectEpzl.SQL.Text :=
      'SELECT EPZ_ID, EPZL_ID' + CR_LF +
      'FROM '+ CREPZL + CR_LF +
      'WHERE EPZL_FKTYP = :EPZL_FKTYP' + CR_LF +
      'AND EPZL_FKID = :EPZL_FKID';
    FQrySelectEpzl.DBPrepare;
  end;

  FQrySelectEpzl.ParamByName('EPZL_FKTYP').AsInteger := AFkTyp;
  FQrySelectEpzl.ParamByName('EPZL_FKID').AsLongId := AFkId;
  try
    FQrySelectEpzl.Open;
    while not FQrySelectEpzl.Eof do
    begin
      expLinkObj.SelectIdForExport(FQrySelectEpzl.fieldByName('EPZL_ID').AsLongId, 'EPZL_FKTYP ' + IntToStr(AFkTyp) + ' / EPZL_FKID ' + AFkId.ToStr);
      expEpzObj.SelectIdForExport(FQrySelectEpzl.fieldByName('EPZ_ID').AsLongId, 'EPZL_FKTYP ' + IntToStr(AFkTyp) + ' / EPZL_FKID ' + AFkId.ToStr);
      FQrySelectEpzl.Next;
    end;
  finally
    FQrySelectEpzl.Close;
  end;

  expLinkObj := FExportObjectManager.GetItemByIdentifier(CREEL);
  if FQrySelectEel = nil then
  begin
    FQrySelectEel := fConnection.CreateSQLQuery;
    FQrySelectEel.SQL.Text :=
      'SELECT EPZ_ID, eel.EEX_ID, eel.EEL_ID' + CR_LF +
      'FROM '+ CREEL + ' eel JOIN ' + CREEX + ' eex on eex.eex_id = eel.eex_id' + CR_LF +
      'WHERE EEL_FKTYP = :EEL_FKTYP' + CR_LF +
      'AND EEL_FKID = :EEL_FKID';
    FQrySelectEpzl.DBPrepare;
  end;

  FQrySelectEel.ParamByName('EEL_FKTYP').AsInteger := AFkTyp;
  FQrySelectEel.ParamByName('EEL_FKID').AsLongId := AFkId;
  try
    FQrySelectEel.Open;
    while not FQrySelectEpzl.Eof do
    begin
      expLinkObj.SelectIdForExport(FQrySelectEel.fieldByName('EEL_ID').AsLongId, 'EEL_FKTYP ' + IntToStr(AFkTyp) + ' / EEL_FKID ' + AFkId.ToStr);
      expEpzObj.SelectIdForExport(FQrySelectEel.fieldByName('EPZ_ID').AsLongId, 'EEL_FKTYP ' + IntToStr(AFkTyp) + ' / EEL_FKID ' + AFkId.ToStr);
      expEexObj.SelectIdForExport(FQrySelectEel.fieldByName('EEX_ID').AsLongId, 'EEL_FKTYP ' + IntToStr(AFkTyp) + ' / EEL_FKID ' + AFkId.ToStr);
      FQrySelectEel.Next;
    end;
  finally
    FQrySelectEel.Close;
  end;
end;

procedure TKundenExport.CheckNbtIdZuNbtNr(AMndIdStr, ANbtNrStr: String);
var
  Qry: TSQLQuery;
  NbtId: TLongId;
begin
  Qry := fConnection.CreateSQLQuery;
  try
    // In NbtNrList merken, um nicht bei jeder NBT_NR ID erneut zu laden
    FNbtNrAusKnvList.Add(AnbtNrStr);

    // NBT_ID laden
    Qry.SQL.Add('SELECT nbt_id FROM ' + ENNBT);
    Qry.SQL.Add('WHERE mnd_id = ' + AMndIdStr);
    Qry.SQL.Add('AND nbt_nr = ' + QuotedStr(ANbtNrStr));
    Qry.Open;
    if (not Qry.Eof) then
    begin
      NbtId := Qry.FieldByName('nbt_id').AsLongId;
      if (not NbtId.IsNull and (not FNBTIds.Contains(NbtId))) then
      begin
        // in ID List speichern
        FNBTIds.Add(NbtId);
      end;
    end;
  finally
    Qry.Free;
  end;
end;

// Daten müssen mandantscharf selektiert werden, deswegen zuerst wird Selekt initialisiert,
// nach der Initialisierung werden die Listen mit NBT_ID und LIF_ID befüllt
function TKundenExport.GetSelectENMPID(ASelect: String): String;
var
  NbtIdsStr, LifIdsStr, MbtIdsStr: String;
begin
  NbtIdsStr := ConvertLongIdListToSeparatedList(FNBTIds, ', ');
  LifIdsStr := ConvertLongIdListToSeparatedList(FLifIds, ', ');
  MbtIdsStr := ConvertLongIdListToSeparatedList(FMbtIds, ', ');

  // Strings können auch leer sein, dann kommt ORA-Fehler bei AND ((lif_id IN (26)) OR (nbt_id IN ()))
  if ((NbtIdsStr <> NULL_STRING) or (LifIdsStr <> NULL_STRING) or (MbtIdsStr <> NULL_STRING)) then
  begin
    if (NbtIdsStr = NULL_STRING) then
      NbtIdsStr := '0';
    if (LifIdsStr = NULL_STRING) then
      LifIdsStr := '0';
    if (MbtIdsStr = NULL_STRING) then
      MbtIdsStr := '0';

    Result := ASelect + CR_LF + 'AND ((lif_id IN (' + LifIdsStr + ')) OR (nbt_id IN (' + NbtIdsStr + ')) OR (mbt_id IN (' + MbtIdsStr + ')))';
  end
  else
  begin
    Result := ASelect;
  end;
end;

procedure TKundenExport.LoadZstepListe(AQry: TSQLQuery; ATable: TTableObj);
var
  InStream: TStream;
  ZipStream: TJclGZIPDecompressionStream;
  JsonStr: UTF8String;
  I: Integer;
begin
  InStream := CreateStreamFromBlob(AQry.FieldByName('ZSTEP_LISTE') );
  try
    ZipStream := TJclGZIPDecompressionStream.Create(InStream);
    try
      SetLength(JsonStr, ZipStream.OriginalDataSize);
      if JsonStr <> '' then
      begin
        ZipStream.Read(PAnsiChar(JsonStr)^, Length(JsonStr));
        for I := 0 to ATable.Columns.Count - 1 do
        begin
          if (TcolumnObj(ATable.Columns.Items[I]).Columnname = 'ZSTEP_LISTE') then
          begin
            TcolumnObj(ATable.Columns.Items[I]).FieldType := ftBlob;
            TcolumnObj(ATable.Columns.Items[I]).Value := UTF8ToString(JsonStr);
          end;
        end;
      end;
    finally
      ZipStream.Free;
    end;
  finally
    InStream.Free;
  end;
end;
